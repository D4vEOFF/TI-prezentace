\tikzset{
    dfsvertex/.style={
        circle,
        draw=black,
        thick,
        minimum size=8mm,
        inner sep=0pt,
        font=\small
    },
    dfsedge/.style={
        thick,
        ->,
        >=stealth,
        shorten >=12pt,
        shorten <=12pt
    },
    dfstreeedge/.style={
        very thick,
        ->,
        >=stealth,
        draw=dfstree,
        shorten >=12pt,
        shorten <=12pt
    },
    dfsbox/.style={
        draw=black,
        rounded corners=2mm,
        fill=white,
        inner sep=2mm,
        font=\small
    }
}
