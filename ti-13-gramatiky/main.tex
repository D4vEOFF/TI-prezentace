    \subsection{Definice gramatiky}

    \begin{frame}{Definice formální gramatiky}
        \begin{defblock}[Formální gramatika]
            \markdarkred{Formální gramatikou} nazýváme uspořádanou čtveřici $G=(V,T,P,S)$, kde
            \begin{itemize}
                \item $V$ je množina tzv. \textbf{neterminálů},
                \item $T$ je množina tzv. \textbf{terminálů},
                \item $S\in V$ je počáteční symbol a
                \item $P$ je množina \textbf{přepisovacích pravidel} tvaru $\alpha\to\omega$ (řetězec $\alpha$ se přepíše na řetězec $\omega$), přičemž $\alpha$ obsahuje alespoň jeden neterminál. 
            \end{itemize}
        \end{defblock}
    \end{frame}
