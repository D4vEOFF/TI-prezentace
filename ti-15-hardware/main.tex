    \subsection{CPU}

    \begin{frame}{Základní části počítače}

        \begin{itemize}
            \visible<2->{\item \textbf{Datová část} provádí výpočty a zpracovává data.}

            \visible<3->{\item \textbf{Řídicí část} určuje, co se má kdy stát.}

            \visible<4->{\item \textbf{Paměť} uchovává programy, instrukce a~data.}

            \visible<5->{\item \textbf{Vstupní zařízení} předávají počítači data z~okolí.}

            \visible<6->{\item \textbf{Výstupní zařízení} předávají výsledky zpět uživateli nebo jinému zařízení.}
        \end{itemize}

        \vspace{4mm}

        \visible<7->{
            \begin{tinfoframe}
                \footnotesize
                V této prezentaci se zaměříme hlavně na
                \textbf{procesor} a \textbf{paměťový subsystém},
                tedy na části, které přímo souvisejí s vykonáváním programu.
            \end{tinfoframe}
        }

    \end{frame}

    \begin{frame}{Základní části počítače}

        \centering

        \input{images/zakladni-casti-pocitace.tex}

        \vspace{4mm}

        {\small
        Program, data a instrukce se přesouvají mezi procesorem,

        }

    \end{frame}

    \subsection{Von Neumannova architektura}

    \begin{frame}{Von Neumannova architektura}

        \small
        \begin{itemize}
            \visible<2->{\item Program i data jsou uloženy ve \textbf{společné paměti}.}

            \visible<3->{\item Procesor postupně načítá \textbf{instrukce} z paměti.}

            \visible<4->{\item Instrukce určují, jaká data se mají zpracovat a kam uložit výsledek.}

            \visible<5->{\item Základní cyklus: $\text{fetch} \rightarrow \text{decode} \rightarrow \text{execute}$.}
        \end{itemize}

        \vspace{1mm}

        \visible<6->{
            \begin{tinfoframe}
                \footnotesize
                Paměť zde nerozlišuje, zda uložené bity představují
                \textbf{data}, nebo \textbf{instrukce}. Rozhoduje až způsob,
                jakým je procesor použije.
            \end{tinfoframe}
        }

        \centering
        \input{images/von-neumannova-architektura.tex}

    \end{frame}

    \begin{frame}{Von Neumannova vs. harvardská architektura}

        \small

        \begin{itemize}
            \visible<2->{\item \textbf{Von Neumannova architektura} používá \textbf{společnou paměť} pro instrukce i data.}

            \visible<3->{\item \textbf{Harvardská architektura} má \textbf{oddělenou paměť} pro instrukce a pro data.}

            \visible<4->{\item Oddělení pamětí umožňuje načítat \textbf{instrukce a data současně}.}

            \visible<5->{\item V praxi se často používá \textbf{kombinace obou přístupů}.}
        \end{itemize}

        \vspace{2mm}

        \visible<6->{
            \begin{tinfoframe}
                \footnotesize
                Navenek se dnešní počítače obvykle chovají jako
                \textbf{von Neumannovy}, ale uvnitř procesoru mohou mít
                například \textbf{oddělenou instrukční a datovou cache}.
            \end{tinfoframe}
        }

    \end{frame}

    \begin{frame}{Von Neumannova vs. harvardská architektura}

    \end{frame}

        \small

        \begin{itemize}
            \visible<2->{\item \textbf{Aritmeticko-logická jednotka (ALU)} provádí aritmetické a logické operace.}

            \visible<4->{\item \textbf{Registry} uchovávají aktuálně používaná data přímo uvnitř procesoru.}

            \visible<5->{\item Procesor komunikuje s pamětí a zařízeními pomocí \textbf{sběrnic}.}

    \begin{frame}{Instrukční cyklus}

        \small

        \begin{itemize}
            \visible<2->{\item Procesor zpracovává program jako posloupnost \textbf{strojových instrukcí}.}
            \visible<3->{\item Každá instrukce projde několika základními kroky.}
            \visible<4->{\item Po dokončení jedné instrukce procesor pokračuje další instrukcí.}
        \end{itemize}

        \vspace{2mm}

        \visible<5->{
            \begin{tinfoframe}
                \footnotesize
                Instrukční cyklus je základní „pracovní rytmus“ procesoru:
                \textbf{načti instrukci}, \textbf{zjisti její význam},
                \textbf{zpracuj data} a \textbf{ulož výsledek}.
            \end{tinfoframe}
        }

        \vspace{3mm}

        \visible<6->{
        \centering

        \input{images/instrukcni-cyklus.tex}
        }

    \end{frame}

\begin{frame}{Registry}

    \small

    \begin{itemize}
        \visible<2->{\item \textbf{Registry} jsou malé, velmi rychlé paměti přímo uvnitř CPU.}
        \visible<3->{\item Uchovávají dočasná data, adresy i informace o stavu výpočtu.}
        \visible<4->{\item Procesor s registry pracuje mnohem rychleji než s hlavní pamětí.}
    \end{itemize}

    \vspace{1mm}

    \visible<5->{
    \centering

    \input{images/registry.tex}
    }

    \begin{itemize}
        \item \textbf{PC} (program counter) ukazuje na adresu další instrukce v paměti.
        \item \textbf{IR} (instruction register) uchovává právě prováděnou instrukci.
        \item \textbf{SP} (stack pointer) ukazuje na vrchol zásobníku v paměti.
        \item \textbf{FLAGS} uchovává příznaky, které informují o výsledku poslední operace (např. přetečení, záporný výsledek, nulový výsledek).
    \end{itemize}

\end{frame}

\begin{frame}{Cache}

    \begin{itemize}
        \visible<2->{\item \textbf{Cache} je malá a velmi rychlá paměť mezi \textbf{CPU} a \textbf{RAM}.}
        \visible<3->{\item Uchovává kopie \textbf{často používaných dat a instrukcí}.}
        \visible<4->{\item Jejím cílem je zkrátit dobu, po kterou procesor čeká na data z hlavní paměti.}
    \end{itemize}

    \vspace{1mm}

    \visible<5->{
    \centering

    \input{images/cache.tex}
    }
    \visible<6->{\scriptsize
    \begin{multicols}{2}
        \begin{enumerate}
            \visible<7->{\item CPU požádá o~data}
            \visible<8->{\item Cache ověří, zda je obsahuje}
            \visible<9->{\item Při cache miss se data načtou z~RAM}
            \visible<10->{\item Data se uloží do cache a~předají CPU}
        \end{enumerate}
    \end{multicols}}

\end{frame}

\begin{frame}{Random Access Memory (RAM)}

    \small

    \begin{itemize}
        \visible<2->{\item \textbf{RAM} (\textit{Random Access Memory}) je \textbf{hlavní pracovní paměť} počítače.}

        \visible<3->{\item Ukládají se do ní \textbf{běžící programy}, jejich \textbf{data} a \textbf{mezivýsledky výpočtů}.}

        \visible<4->{\item Je \textbf{rychlejší} než externí úložiště, ale \textbf{pomalejší} než registry nebo cache.}

        \visible<5->{\item Je \textbf{volatilní} — po vypnutí napájení se její obsah ztratí.}
    \end{itemize}

    \vspace{1mm}

    \visible<6->{
        \begin{tinfoframe}
            \footnotesize
            Z pohledu programu se RAM chová jako \textbf{lineárně adresovaná paměť}:
            každé paměťové místo má svou \textbf{adresu} a procesor může
            z dané adresy \textbf{číst} nebo na ni \textbf{zapisovat}.
        \end{tinfoframe}
    }

    \centering

    \input{images/random-access-memory-ram.tex}
\end{frame}

\begin{frame}{Externí paměti: SSD a HDD}

    \small

    \begin{itemize}
        \visible<2->{\item \textbf{HDD} ukládá data magneticky na rotující plotny.}
        \visible<3->{\item \textbf{SSD} ukládá data elektronicky do flash paměti.}
        \visible<4->{\item SSD nemá pohyblivé části, proto má rychlejší náhodný přístup.}
        \visible<5->{\item HDD bývá výhodný pro velké kapacity za nižší cenu.}
    \end{itemize}

    \vspace{2mm}

    \visible<6->{
    \centering

    \input{images/externi-pameti-ssd-a-hdd.tex}
    }

    \vspace{1mm}

    \visible<7->{
        \begin{tinfoframe}
            \footnotesize
            SSD je obvykle rychlejší při spouštění systému a aplikací,
            zatímco HDD se často používá tam, kde rozhoduje hlavně
            \textbf{velká kapacita} a \textbf{nízká cena za GB}.
        \end{tinfoframe}
    }

\end{frame}

\begin{frame}{Paměťová hierarchie}

    \small

    \begin{itemize}
        \visible<2->{\item Počítač nepoužívá jen jeden druh paměti, ale \textbf{celou hierarchii pamětí}.}

        \visible<3->{\item Jednotlivé úrovně se liší hlavně \textbf{rychlostí}, \textbf{kapacitou} a \textbf{cenou}.}

        \visible<4->{\item Čím je paměť blíže procesoru, tím bývá \textbf{rychlejší}, ale také \textbf{menší} a \textbf{dražší}.}
    \end{itemize}

            \footnotesize
            Paměťová hierarchie umožňuje spojit \textbf{vysokou rychlost}
            blízko procesoru s~\textbf{velkou kapacitou} na~nižších úrovních.

    }
