    \begin{frame}{Regulární výrazy formálně}
        \visible<2->{\begin{defblock}[Regulární výrazy a jejich hodnoty]
            \visible<3->{Mějme konečnou abecedu $\Sigma$. Pak \textbf{množinu všech regulárních výrazů nad abecedou $\Sigma$} značíme $\regex(\Sigma)$ a~definujeme ji následoně:}
            \begin{itemize}
                \visible<4->{\item $\lambda,a$, kde $a\in\Sigma$, jsou regulární výrazy, tj. $\lambda,a\in\regex(\Sigma)$. Jejich hodnoty jsou $L(\lambda)=\set{\lambda}$ a~$L(a)=\set{a}$.}
                \visible<5->{\item Jsou-li $\alpha,\beta\in\regex(\Sigma)$, pak také
                \[\alpha+\beta,\quad\alpha\beta,\quad \alpha^{*}\quad\text{a}\quad (\alpha)\]
                jsou v~$\regex(\Sigma)$ a platí
                \begin{align*}
                    L(\alpha+\beta)&=L(\alpha)\cup L(\beta),\quad L(\alpha\beta)=L(\alpha)L(\beta)\\
                    L(\alpha^{*})&=L(\alpha)^{*},\quad L((\alpha))=L(\alpha).
                \end{align*}}
            \end{itemize}
        \end{defblock}}
    \end{frame}

    \begin{frame}{Příklady regulárních jazyků}
        \visible<2->{\begin{tinfoframe}
            \begin{center}
                Operace $*$ má nejvyšší prioritu, nižší konkatenace (zřetězení) a nejnižší sjednocení $+$.
            \end{center}
        \end{tinfoframe}}

        \visible<3->{Mějme abecedu $\Sigma=\set{0,1}$.}

        \visible<4->{\begin{center}
            \begin{tabular}{c|c}
            Jazyk & Regulární výraz \\
            \hline
            \visible<5->{všechna slova nad $\Sigma$} & \visible<6->{$(0+1)^*$ \\}
            \visible<6->{slova končící na $01$} & \visible<8->{$(0+1)^*01$ \\}
            \visible<7->{slova obsahující právě jednu $1$} & \visible<10->{$0^*10^*$ \\}
            \visible<11->{slova se sudým počtem jedniček} & \visible<12->{$0^*(10^*10^*)^*$ \\}
            \visible<13->{slova bez dvou sousedních jedniček} & \visible<14->{$(0+10)^*(\lambda+1)$}
            \end{tabular}
        \end{center}}
    \end{frame}

    \begin{frame}{Kleeneho věta}
        \visible<2->{\begin{thmblock}[Kleeneho]
            \begin{enumerate}
                \visible<3->{\item Každý jazyk reprezentovaný konečným automatem lze zapsat jako regulární výraz.}
                \visible<4->{\item Každý jazyk popsaný regulárním výrazem můžeme zapsat jako NFA (a tedy i~DFA).}
            \end{enumerate}
        \end{thmblock}}

        \visible<5->{Jinými slovy: regulární výrazy a konečné automaty popisují tutéž třídu jazyků.}
    \end{frame}

    \begin{frame}{Kleeneho věta: náznak důkazu}
        \visible<2->{Důkaz má dvě části:}

        \visible<3->{\begin{enumerate}
            \visible<4->{\item Každý regulární výraz lze převést na konečný automat.}
            \begin{itemize}
                \visible<5->{\item Pro $\lambda$ a $a\in\Sigma$ umíme sestrojit jednoduchý automat.}
                \visible<6->{\item Operace $+$, zřetězení a $^*$ odpovídají konstrukcím nad automaty (viz další snímek).}
            \end{itemize}

            \visible<7->{\item Každý konečný automat lze převést na regulární výraz.}
            \begin{itemize}
                \visible<8->{\item Postupně eliminujeme stavy automatu.}
                \visible<9->{\item Hrany označujeme regulárními výrazy.}
                \visible<10->{\item Nakonec zůstane výraz popisující všechna přijímaná slova.}
            \end{itemize}
        \end{enumerate}}
    \end{frame}
