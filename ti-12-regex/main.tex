    \section{Regulární výrazy}

    \titleframe{Regulární výrazy}

    \subsection{Rekapitulace}

    \begin{frame}{Rekapitulace}
        \begin{itemize}
            \visible<2->{\item Již víme, že formální jazyky lze popsat}
            \visible<3->{\begin{itemize}
                \visible<3->{\item charakteristickou vlastností a}
                \visible<4->{\item (ne)deterministickým konečným automatem.}
            \end{itemize}}
            \visible<5->{\begin{tinfoframe}
                \begin{center}
                    Strukturu jazyka lze popsat tzv. \textbf{regulárním výrazem}.
                \end{center}
            \end{tinfoframe}}
            \visible<6->{\item Zkratka \textbf{RegEx} = \textit{regular expression}.}
        \end{itemize}
    \end{frame}

    \subsection{Formální definice RegExu}

    \begin{frame}{Regulární výrazy formálně}
        \visible<2->{\begin{defblock}[Regulární výrazy a jejich hodnoty]
            \visible<3->{Mějme konečnou abecedu $\Sigma$. Pak \textbf{množinu všech regulárních výrazů nad abecedou $\Sigma$} značíme $\regex(\Sigma)$ a~definujeme ji následoně:}
            \begin{itemize}
                \visible<4->{\item $\lambda,a$, kde $a\in\Sigma$, jsou regulární výrazy, tj. $\lambda,a\in\regex(\Sigma)$. Jejich hodnoty jsou $L(\lambda)=\set{\lambda}$ a~$L(a)=\set{a}$.}
                \visible<5->{\item Jsou-li $\alpha,\beta\in\regex(\Sigma)$, pak také
                \[\alpha\mid\beta,\quad\alpha\beta,\quad \alpha^{*}\quad\text{a}\quad (\alpha)\]
                jsou v~$\regex(\Sigma)$ a platí
                \begin{align*}
                    L(\alpha\mid\beta)&=L(\alpha)\cup L(\beta),\quad L(\alpha\beta)=L(\alpha)L(\beta)\\
                    L(\alpha^{*})&=L(\alpha)^{*},\quad L((\alpha))=L(\alpha).
                \end{align*}}
            \end{itemize}
        \end{defblock}}
    \end{frame}

    \subsection{Příklady regulárních jazyků}

    \begin{frame}{Příklady regulárních jazyků}
        \visible<2->{\begin{tinfoframe}
            \begin{center}
                Operace $*$ má nejvyšší prioritu, nižší konkatenace (zřetězení) a nejnižší sjednocení $\mid$.
            \end{center}
        \end{tinfoframe}}

        \visible<3->{Mějme abecedu $\Sigma=\set{0,1}$.}

        \visible<4->{\begin{center}
            \begin{tabular}{c|c}
            Jazyk & Regulární výraz \\
            \hline
            \visible<5->{všechna slova nad $\Sigma$} & \visible<6->{$(0\mid1)^*$ \\}
            \visible<7->{slova končící na $01$} & \visible<8->{$(0\mid1)^*01$ \\}
            \visible<9->{slova obsahující právě jednu $1$} & \visible<10->{$0^*10^*$ \\}
            \visible<11->{slova se sudým počtem jedniček} & \visible<12->{$0^*(10^*10^*)^*$ \\}
            \visible<13->{slova bez dvou sousedních jedniček} & \visible<14->{$(0\mid10)^*(\lambda\mid1)$}
            \end{tabular}
        \end{center}}
    \end{frame}

    \section{Kleeneho věta}

    \titleframe{Kleeneho věta}

    \subsection{Formulace Kleeneho věty}

    \begin{frame}{Kleeneho věta}
        \visible<2->{\begin{thmblock}[Kleeneho]
            \begin{enumerate}
                \visible<3->{\item Každý jazyk reprezentovaný konečným automatem lze zapsat jako regulární výraz.}
                \visible<4->{\item Každý jazyk popsaný regulárním výrazem můžeme zapsat jako NFA (a tedy i~DFA).}
            \end{enumerate}
        \end{thmblock}}

        \visible<5->{\textbf{Jinými slovy:} regulární výrazy a konečné automaty popisují tutéž třídu jazyků.}
    \end{frame}

    \subsection{Náznak důkazu}

    \begin{frame}{Kleeneho věta: náznak důkazu}
        \visible<2->{Důkaz má dvě části:}

        \visible<3->{\begin{enumerate}
            \visible<4->{\item Každý regulární výraz lze převést na konečný automat.}
            \begin{itemize}
                \visible<5->{\item Pro $\lambda$ a $a\in\Sigma$ umíme sestrojit jednoduchý automat.}
                \visible<6->{\item Operace $\mid$, zřetězení a $^*$ odpovídají konstrukcím nad automaty (viz další snímek).}
            \end{itemize}

            \visible<7->{\item Každý konečný automat lze převést na regulární výraz.}
            \begin{itemize}
                \visible<8->{\item Postupně eliminujeme stavy automatu.}
                \visible<9->{\item Hrany označujeme regulárními výrazy.}
                \visible<10->{\item Nakonec zůstane výraz popisující všechna přijímaná slova.}
            \end{itemize}
        \end{enumerate}}
    \end{frame}

    \section{Další příklady regulárních výrazů}

    \begin{frame}{Další příklady regulárních výrazů}
    \visible<2->{Mějme abecedu $\Sigma=\{0,1\}$.}

    \small
    \begin{itemize}
        \item \visible<3->{$\alpha_1=0^*1^*$}
        \visible<4->{\[
            L(\alpha_1)=\set{0^m1^n : m,n\geq 0}
        \]
        slova, ve kterých jsou nejdříve nuly a potom jedničky.}

        \item \visible<5->{$\alpha_2=(0\mid1)^*01$}
        \visible<6->{\[
            L(\alpha_2)=\set{w01 : w\in\Sigma^*}
        \]
        slova končící na $01$.}

        \item \visible<7->{$\alpha_3=(0\mid1)^*00(0\mid1)^*$}
        \visible<8->{\[
            L(\alpha_3)=\set{u00v : u,v\in\Sigma^*}
        \]
        slova obsahující podslovo $00$.}

        \item \visible<9->{$\alpha_4=0^*10^*10^*$}
        \visible<10->{\[
            L(\alpha_4)=\set{w\in\Sigma^* : |w|_1=2}
        \]
        slova obsahující právě dvě jedničky.}

        \item \visible<11->{$\alpha_5=((0\mid1)(0\mid1))^*$}
        \visible<12->{\[
            L(\alpha_5)=\set{w\in\Sigma^* : |w| \text{ je sudé}}
        \]}
    \end{itemize}
    \end{frame}

    \section{RegEx v praxi}

    \begin{frame}{Pozor: RegEx v praxi}
        \pause
        Formální regulární výrazy popisují \textbf{právě regulární jazyky}.

        \pause
        \medskip
        Praktické RegEx enginy ale často obsahují rozšíření, například
        \textbf{zpětné reference}.

        \[
        \texttt{\^{}([01]*)\#\textbackslash 1\$}
        \]

        popisuje jazyk
        \[
        L=\set{w\#w : w\in\{0,1\}^*}.
        \]

        \pause
        \markdarkred{Tento jazyk ale není regulární!}
    \end{frame}
