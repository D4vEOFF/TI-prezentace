% Popisek varianty (jen pro tento náhled)
\newcommand{\popisek}[2]{%
    \begin{frame}[plain]
        \begin{tikzpicture}[remember picture, overlay]
            \fill[white] (current page.north west) rectangle (current page.south east);
            \node[align=center, text width=0.8\paperwidth] at (current page.center) {%
                {\Huge\textcolor{maincolor}{#1}\par}
                \vskip1em
                {\large #2\par}
            };
        \end{tikzpicture}
    \end{frame}
    \addtocounter{framenumber}{-1}%
}

% Vysázení jedné varianty pro dva různé předměty
% #1 = klíč šablony, #2 = nadpis, #3 = popis
\newcommand{\ukazvariantu}[3]{%
    \popisek{#2}{#3}%
    \setbeamertemplate{title page}[#1]
    % Nejdelší reálný název tématu -- zkouška, zda se vejde
    \title[Ukázka]{Teoretická informatika}
    \subtitle{Problém 3-SAT, problém nezávislé množiny a~kliky}
    \begin{frame}[plain]\titlepage\end{frame}
    \addtocounter{framenumber}{-1}%
    \title[Ukázka]{Matematika}
    \subtitle{Derivace a~její geometrický význam}
    \begin{frame}[plain]\titlepage\end{frame}
    \addtocounter{framenumber}{-1}%
}
