\subsection{Formální neuron}

\begin{frame}{Formální neuron}


    \visible<2->{\obrazek[0.4\textheight]{images/formalni-neuron.tex}}



    \visible<3->{\begin{timportantframe}
        Formální neuron nejprve spočítá vážený součet vstupů
        a~poté na něj aplikuje přenosovou funkci.
    \end{timportantframe}}

\end{frame}

\begin{frame}{Výpočet formálního neuronu}

    \visible<2->{\begin{itemize}
        \item Pro \(i\)-tý datový záznam máme vektor
        \[
            \mathbf{x}_i
            =
            (x_{i1},x_{i2},\ldots,x_{ip}).
        \]
        \visible<3->{a~vstupní váhy neuronu $w_1,w_2,\ldots,w_p$.
        \item Z~toho spočítáme výstupní hodnotu
        \[
            \hat{y}_i=f\left(\sum_{j=1}^{p} w_j x_{ij}+\theta\right),
        \]
        kde}

        \visible<4->{\begin{itemize}
            \item \(\theta\) je tzv. \markdarkred{práh} (\markdarkred{bias}), který umožňuje posunout rozhodování neuronu a
            \visible<5->{\item $f$ je tzv. \markdarkred{přenosová funkce} (anglicky \markdarkred{activation function}, viz později).}
        \end{itemize}}
    \end{itemize}}

\end{frame}


\begin{frame}{Váhy a~prahová hodnota}

    \visible<2->{\begin{itemize}
        \item Váha \(w_j\) určuje význam vstupu \(x_{ij}\).

        \visible<3->{\item Pokud je \(w_j>0\), vyšší hodnota vstupu \(x_{ij}\) výstup spíše zvyšuje.}

        \visible<4->{\item Pokud je \(w_j<0\), vyšší hodnota vstupu \(x_{ij}\) výstup spíše snižuje.}

        \visible<5->{\item Prahová hodnota \(\theta\) určuje, jak velký musí být vážený součet,
              aby došlo k~aktivaci neuronu.}
    \end{itemize}}

\end{frame}

\subsection{Neuronová síť}

\begin{frame}{Formální definice neuronové sítě}
    \footnotesize

    \visible<2->{\begin{defblock}[Formální neuronová síť]
        Neuronovou síť budeme chápat jako uspořádanou sedmici
        \[
            \mathcal{N}=(V,E,I,O,w,\theta,f),
        \]
        kde
        \begin{itemize}
            \item $V$ je konečná množina \markdarkred{neuronů},
            \item $E$ je množina orientovaných hran (spojů),
            \item $I$ je množina \markdarkred{vstupních neuronů},
            \item $O$ je množina \markdarkred{výstupních neuronů},
            \item $\mapping{w}{E}{\R}$ je tzv. \markdarkred{váhová funkce},
            \item $\mapping{\theta}{V\setminus I}{\R}$ je tzv. \markdarkred{prahová funkce} a
            \item $f$ je přenosová funkce.
        \end{itemize}
    \end{defblock}}

\end{frame}

\subsection{Dopředná neuronová síť}

\begin{frame}{Dopředná neuronová síť}

    \visible<2->{\markdarkred{Dopředná neuronová síť} (anglicky \markdarkred{feed-forward neural network}) je taková neuronová síť, v~níž se informace
    šíří \textbf{pouze jedním směrem}: od vstupních neuronů k~výstupním neuronům.}


    \visible<3->{\obrazek[0.4\textheight]{images/dopredna-neuronova-sit.tex}}

\end{frame}

\begin{frame}{Dopředný výpočet ve vrstvách}

    \visible<2->{\begin{itemize}
        \item Označme $\mathbf{a}^{(0)}=\mathbf{x}_i$.
        \visible<3->{\item Pro každou další vrstvu $\mathbf{a}^{(\ell+1)}$ postupně počítáme
        \[
            a_i^{(\ell+1)}=f\left(\sum_{j=1}^{n_{\ell}}w_{ji}^{(\ell+1)}a_j^{(\ell)}+\theta_i^{(\ell+1)}\right)
        \]}
        \visible<4->{\item Výstup poslední vrstvy je výstupem celé neuronové sítě:
        \[
            \hat{\mathbf{y}}_i=\mathbf{a}^{(L)}.
        \]}
        \visible<5->{\item Pokud má síť pouze jeden výstupní neuron, píšeme jednoduše $\hat{y}_i$.}
    \end{itemize}}

\end{frame}

\begin{frame}{Dopředný výpočet ve vrstvách}
    \obrazek[0.4\textheight]{images/dopredny-vypocet-ve-vrstvach.tex}
\end{frame}

\subsection{Perceptron}

\begin{frame}{Perceptron}

    \visible<2->{\begin{tinfoframe}
        Perceptron je nejjednodušší neuronová síť tvořená vstupní vrstvou
        a~jedním výstupním neuronem.
    \end{tinfoframe}}



    \visible<3->{\begin{itemize}
        \item Na vstupu je vektor
        \[
            \mathbf{x}_i=(x_{i1},x_{i2},\ldots,x_{ip}),
        \]
        kde jednotlivé znaky jsou ze souborů $\mathbf{X}_1,\dots,\mathbf{X}_p$.

        \visible<4->{\item Každý vstup \(x_{ij}\) má svou váhu \(w_j\).}

        \visible<5->{\item Výstupem je jedna hodnota \(\hat{y}_i\).}

        \visible<6->{\item Perceptron se používá především pro jednoduchou binární klasifikaci.}
    \end{itemize}}

\end{frame}

\begin{frame}{Výpočet perceptronu}

    \visible<2->{\obrazek[0.4\textheight]{images/vypocet-perceptronu.tex}}

\end{frame}

\begin{frame}{Perceptron jako klasifikátor}

    \visible<2->{U~binární klasifikace může perceptron rozhodovat například pomocí \markdarkred{skokové přenosové funkce}
    \[
        f(z)=
        \begin{cases}
            1, & \text{pokud } z\geqslant 0,\\
            0, & \text{pokud } z<0.
        \end{cases}
    \]}



    \visible<3->{\begin{itemize}
        \item Spočítáme hodnotu
        \[
            z_i=\sum_{j=1}^{p}w_jx_{ij}+\theta.
        \]

        \visible<4->{\item Pokud \(z_i\geqslant 0\), zařadíme záznam do jedné třídy.}

        \visible<5->{\item Pokud \(z_i<0\), zařadíme záznam do druhé třídy.}
    \end{itemize}}

\end{frame}

\begin{frame}{Geometrická interpretace perceptronu}

    \visible<2->{Pro dva vstupní znaky \(x_{i1}\) a~\(x_{i2}\) má rozhodovací hranice tvar
    \[
        w_1x_{i1}+w_2x_{i2}+\theta=0.
    \]}



    \visible<3->{\begin{center}
        \obrazek[0.4\textheight]{images/geometricka-interpretace-perceptronu.tex}
    \end{center}}



    \visible<4->{\begin{timportantframe}
        Perceptron vytváří lineární rozhodovací hranici.
        V~rovině je to přímka, ve vyšších dimenzích nadrovina.
    \end{timportantframe}}

\end{frame}

\subsection{Přenosová funkec}

\begin{frame}{Proč potřebujeme přenosovou funkci?}

    Přenosová funkce \(f\) určuje, jak se vnitřní potenciál neuronu převede na jeho výstup.



    \visible<2->{\[
        a_i^{(\ell+1)}
        =
        f\left(
            \sum_{j=1}^{n_{\ell}}
            w_{ji}^{(\ell+1)}a_j^{(\ell)}
            +
            \theta_i^{(\ell+1)}
        \right)
    \]}



    \visible<3->{\begin{timportantframe}
        Pokud by všechny přenosové funkce byly lineární, pak by i~celá
        vícevrstvá síť představovala pouze lineární model.
        Nelineární přenosové funkce umožňují modelovat složitější vztahy.
    \end{timportantframe}}

\end{frame}

\begin{frame}{Skoková funkce a~sigmoida}
    
    \begin{columns}[T]

        \begin{column}{0.48\textwidth}\visible<2->{
            \[
                f(x)=
                \begin{cases}
                    1, & x\geqslant 0,\\
                    0, & x<0.
                \end{cases}
            \]

            \begin{center}
                \obrazek[0.4\textheight]{images/skokova-funkce-a-sigmoida-1.tex}
            \end{center}

            \visible<3->{\begin{tnoticebox}{Skoková funkce}
                Vhodná pro jednoduché binární rozhodování.
            \end{tnoticebox}}
        }\end{column}

        \begin{column}{0.48\textwidth}\visible<4->{
            \[
                f(x)=\frac{1}{1+e^{-x}}
            \]

            \begin{center}
                \obrazek[0.4\textheight]{images/skokova-funkce-a-sigmoida-2.tex}
            \end{center}

            \begin{tnoticebox}{Sigmoida}
                Hladká funkce s~výstupem v~intervalu \((0,1)\).
            \end{tnoticebox}
        }\end{column}
    \end{columns}

\end{frame}


\begin{frame}{Další přenosové funkce}

    \visible<2->{\begin{itemize}
        \item \markdarkred{ReLU} (zkratka pro \markdarkred{Rectified Linear Unit}) se často používá ve skrytých vrstvách sítí s~větším počtem vrstev.
        \[\operatorname{ReLU}(x)=\max\set{0,x},\quad x\in\R.\]
        \visible<3->{\item Funkce \markdarkred{softmax} se často používá pro vícetřídní klasifikaci na výstupní vrstvě. Výstupy lze chápat jako skóre jednotlivých tříd. Máme-li celkově $k$ tříd, pak
        \[\sigma_i(\mathbf{z})=\frac{e^{z_i}}{\sum_{j=1}^{p}e^{z_j}},\quad \mathbf{z}\in(z_1,\dots,z_p)\in\R^{p}, \quad 1\leqslant i\leqslant k.\]}
        \visible<4->{\item U~regresních úloh se často používá \markdarkred{identita}: $\operatorname{id}(x)=x$ pro $x\in\R$.}
    \end{itemize}}



    \visible<5->{\begin{timportantframe}
        Skryté vrstvy obvykle potřebují nelineární přenosovou funkci. Na různých vrstvách se tak mohou používat různé přenosové funkce.
    \end{timportantframe}}

\end{frame}

\subsection{Vícevrstvý perceptron}

\begin{frame}{Vícevrstvý perceptron}

    \markdarkred{Vícevrstvý perceptron} (anglicky \markdarkred{multi-layer perceptron}, zkráceně MLP) je dopředná neuronová síť tvořená vstupní vrstvou, jednou nebo více skrytými vrstvami a~výstupní vrstvou.



    \visible<2->{\begin{center}
        \obrazek[0.4\textheight]{images/vicevrstvy-perceptron.tex}
    \end{center}}



    \visible<3->{Obecně pracujeme s~výstupními statistickými soubory $\mathbf{Y}_1,\mathbf{Y}_2,\ldots,\mathbf{Y}_k$. Pro \(i\)-tý záznam tedy očekáváme výstup $\mathbf{y}_i=(y_{i1},y_{i2},\ldots,y_{ik})$ a~síť počítá odhad
    \[
        \hat{\mathbf{y}}_i=(\hat{y}_{i1},\hat{y}_{i2},\ldots,\hat{y}_{ik}).
    \]}

\end{frame}

\begin{frame}{Výpočet ve vrstvách}

    Vstupní vrstva obsahuje hodnoty znaků \(i\)-tého vektoru dat:
    \[
        a_j^{(0)}=x_{ij},
        \qquad
        j\in\set{1,\ldots,p}.
    \]



    \visible<2->{Hodnota \(i\)-tého neuronu v~následující vrstvě se vypočítá jako
    \[
        a_i^{(\ell+1)}
        =
        f^{(\ell+1)}
        \left(
            \sum_{j=1}^{n_\ell}
            w_{ji}^{(\ell+1)}a_j^{(\ell)}
            +
            \theta_i^{(\ell+1)}
        \right).
    \]}



    \visible<3->{\begin{enumerate}
        \item Nejprve známe hodnoty ve vstupní vrstvě.
        \visible<4->{\item Z~nich vypočítáme hodnoty neuronů v~první skryté vrstvě.}
        \visible<5->{\item Stejným způsobem pokračujeme vrstvu po vrstvě.}
        \visible<6->{\item Hodnoty poslední vrstvy tvoří výstup celé sítě.}
    \end{enumerate}}

\end{frame}

    \end{timportantframe}}

\end{frame}

\subsection{Trénovací data}

\begin{frame}{Trénovací množina}
    \footnotesize

    \visible<2->{\begin{itemize}
        \item Uvažujme vstupní statistické soubory $\mathbf{X}_1,\mathbf{X}_2,\ldots,\mathbf{X}_p$
        a~výstupní statistické soubory $\mathbf{Y}_1,\mathbf{Y}_2,\ldots,\mathbf{Y}_k$.
        \visible<3->{\item Jednotlivé statistické soubory mají tvar
        \[
            \begin{aligned}
                \mathbf{X}_j
                &=
                (x_{1j},x_{2j},\ldots,x_{nj}),
                \qquad
                j\in\{1,\ldots,p\},\\
                \mathbf{Y}_r
                &=
                (y_{1r},y_{2r},\ldots,y_{nr}),
                \qquad
                r\in\{1,\ldots,k\}.
            \end{aligned}
        \]}
        \visible<4->{\item Pro \(i\)-tý datový záznam sestavíme vstupní a~výstupní vektor
        \[
            \mathbf{x}_i
            =
            (x_{i1},x_{i2},\ldots,x_{ip}),
            \qquad
            \mathbf{y}_i
            =
            (y_{i1},y_{i2},\ldots,y_{ik}).
        \]}
    \end{itemize}}



    \visible<5->{\begin{timportantframe}
        Trénovací množinu tvoří dvojice vstupních dat a~jejich očekávaných
        výstupů:
        \[
            D
            =
            \set{
                (\mathbf{x}_i,\mathbf{y}_i)
                \mid
                i\in\{1,\ldots,n\}
            }.
        \]
    \end{timportantframe}}

\end{frame}

\subsection{Měření chyby}

\begin{frame}{Měření chyby neuronové sítě}
    \footnotesize

    \visible<2->{Pro vstup \(\mathbf{x}_i\) neuronová síť $\mathcal{N}$ vypočítá odhad
    \[
        \hat{\mathbf{y}}_i
        =
        (\hat{y}_{i1},\hat{y}_{i2},\ldots,\hat{y}_{ik}).
    \]}



    \visible<3->{Chybu \(r\)-tého výstupu pro \(i\)-tý vektor dat můžeme vyjádřit jako
    \[
        e_{ir}=y_{ir}-\hat{y}_{ir}.
    \]}



    \visible<4->{\begin{itemize}
        \item Součet čtverců chyb (anglicky \markdarkred{sum of squared errors}) definujeme jako
        \[
            \SSE_{\mathbf{X}_1,\ldots,\mathbf{X}_p}^{\mathbf{Y}_1,\ldots,\mathbf{Y}_k}(\mathcal{N})
            =
            \sum_{i=1}^{n}
            \sum_{r=1}^{k}
            \left(y_{ir}-\hat{y}_{ir}\right)^2.
        \]
        \visible<5->{\item Střední kvadratická chyba (anglicky \markdarkred{mean squared error}) je
        \[
            \MSE_{\mathbf{X}_1,\ldots,\mathbf{X}_p}^{\mathbf{Y}_1,\ldots,\mathbf{Y}_k}(\mathcal{N})
            =
            \frac{1}{nk}
            \sum_{i=1}^{n}
            \sum_{r=1}^{k}
            \left(y_{ir}-\hat{y}_{ir}\right)^2
            =
            \frac{1}{nk}\SSE_{\mathbf{X}_1,\ldots,\mathbf{X}_p}^{\mathbf{Y}_1,\ldots,\mathbf{Y}_k}(\mathcal{N}).
        \]}
    \end{itemize}}

\end{frame}

\subsection{Zpětná propagace}

\begin{frame}{Zpětná propagace chyby}

    \begin{tinfoframe}
        Z~hodnoty chyby víme, jak přesný byl výstup celé sítě.
        Potřebujeme však určit, jak jednotlivé váhy a~prahové hodnoty
        k~této chybě přispěly.
    \end{tinfoframe}



    \visible<2->{\obrazek[0.2\textheight]{images/zpetna-propagace-chyby.tex}}



    \visible<3->{\begin{timportantbox}{Základní myšlenka}
        Zpětná propagace chyby (anglicky
        \markdarkred{backpropagation}) postupuje od výstupní vrstvy
        směrem ke vstupní vrstvě a~určuje, jak změna každé váhy
        a~prahové hodnoty ovlivní výslednou chybu.
    \end{timportantbox}}

\end{frame}

\subsection{Minimalizace chyby}

\begin{frame}{Minimalizace chyby gradientním sestupem}

    \begin{tinfobox}{Připomenutí}
        Gradientní sestup upravuje parametry modelu ve směru, ve kterém
        se hodnota chybové funkce zmenšuje.
    \end{tinfobox}



    \visible<2->{Váhu spoje z~\(j\)-tého neuronu v~\((\ell-1)\)-ní vrstvě do
    \(i\)-tého neuronu v~\((\ell)\)-ní vrstvě upravíme podle vztahu
    \[
        w_{ji}^{(\ell)}
        \gets
        w_{ji}^{(\ell)}
        -
        \eta
        \frac{\partial E}
             {\partial w_{ji}^{(\ell)}}.
    \]}



    \visible<3->{Obdobně upravíme prahovou hodnotu neuronu:
    \[
        \theta_i^{(\ell)}
        \gets
        \theta_i^{(\ell)}
        -
        \eta
        \frac{\partial E}
             {\partial \theta_i^{(\ell)}}.
    \]}

\end{frame}

\begin{frame}{Minimalizace chyby gradientním sestupem}

    \visible<2->{\begin{center}
        \obrazek[0.4\textheight]{images/minimalizace-chyby-gradientnim-sestupem.tex}
    \end{center}}
    


    \visible<3->{\begin{timportantframe}
        Parametr \(\eta>0\) se nazývá rychlost učení (anglicky \markdarkred{learning rate}).
        Určuje velikost změny parametrů během jednoho kroku učení.
    \end{timportantframe}}



    \visible<4->{\begin{tnoticebox}{Úloha zpětné propagace}
        Zpětná propagace efektivně vypočítá derivace chybové funkce
        podle všech vah a~prahových hodnot neuronové sítě.
        Gradientní sestup následně tyto parametry upraví.
    \end{tnoticebox}}
\end{frame}

\subsection{Hluboké učení}

\begin{frame}{Hluboké učení}

    \visible<2->{\markdarkred{Hluboké učení} (anglicky \markdarkred{deep learning}) je oblast
    strojového učení, která využívá neuronové sítě s~větším počtem
    skrytých vrstev.}



    \visible<3->{\begin{itemize}
        \item Mělká síť má obvykle malý počet skrytých vrstev.

        \visible<4->{\item Hluboká síť obsahuje mnoho skrytých vrstev.}

        \visible<5->{\item Každá vrstva postupně vytváří nový popis vstupních dat.}

        \visible<6->{\item Díky tomu může síť zachycovat složitější vztahy v~datech.}
    \end{itemize}}



    \visible<7->{\begin{timportantbox}{Hlavní rozdíl}
        \textbf{Nejde o~jiný princip výpočtu neuronu}, ale o~rozsáhlejší síť,
        která skládá velké množství jednoduchých výpočtů.
    \end{timportantbox}}

\end{frame}

\subsection{Batch a~epoch}

\begin{frame}{Batch}

    \visible<2->{\begin{itemize}
        \item Mějme trénovací množinu $D=\set{(\mathbf{x}_i,\mathbf{y}_i)\mid i\in\{1,\ldots,n\}}$.
        \item \markdarkred{Batch} neboli dávka je menší část trénovací množiny, na které v~daném kroku spočítáme chybu a~upravíme parametry sítě.
    \end{itemize}}



    \visible<3->{Pokud má batch velikost \(m\), může mít například tvar
    \[
        B
        =
        \set{
            (\mathbf{x}_{i_1},\mathbf{y}_{i_1}),
            \ldots,
            (\mathbf{x}_{i_m},\mathbf{y}_{i_m})
        }
        \subseteq
        D,
    \]
    kde $i_1,\dots,i_m\in\set{1,\dots,n}$ jsou vybrané indexy.}



    \visible<4->{\begin{tnoticebox}{Proč nepoužívat vždy celý soubor?}
        U~velkých trénovacích množin by výpočet chyby nad celým souborem
        mohl být pomalý. Proto se často používají menší dávky dat.
    \end{tnoticebox}}

\end{frame}

\begin{frame}{Epoch}

    \visible<2->{\begin{itemize}
        \item Jedna epocha (anglicky \markdarkred{epoch}) znamená jedno úplné
        zpracování celé trénovací množiny.
        \visible<3->{\item Trénovací množinu si můžeme rozdělit na batche
        \[
            D
            =
            \mathcal{B}_1
            \cup
            \mathcal{B}_2
            \cup
            \cdots
            \cup
            \mathcal{B}_s.
        \]}
        \visible<4->{\item Během jedné epochy síť postupně zpracuje
        \[
            \mathcal{B}_1,\mathcal{B}_2,\ldots,\mathcal{B}_s.
        \]}
    \end{itemize}}



    \visible<5->{\begin{timportantbox}{Vztah batch a~epoch}
        Jeden batch odpovídá jednomu kroku učení.
        Jedna epocha odpovídá zpracování všech batchů.
    \end{timportantbox}}

\end{frame}

\begin{frame}{Batch a~epoch při trénování modelu}

    \visible<2->{\obrazek[0.4\textheight]{images/batch-a-epoch-pri-trenovani-modelu.tex}}



    \visible<3->{\begin{enumerate}
        \item Trénovací množinu \(D\) rozdělíme na \textbf{batche}.
        \item Pro každý batch spočítáme výstupy sítě a~chybu.
        \item Pomocí zpětné propagace upravíme váhy a~prahové hodnoty.
        \item Po zpracování všech batchů skončí jedna \textbf{epocha}.
        \item Celý postup opakujeme po zvolený počet epoch.
    \end{enumerate}}

\end{frame}

\begin{frame}{Jak volit počet epoch a~velikost batche?}
    
    \begin{columns}[T]

        \begin{column}{0.48\textwidth}\visible<2->{
            \begin{center}
                \obrazek[0.4\textheight]{images/jak-volit-pocet-epoch-a-velikost-batche-1.tex}
            \end{center}

            \visible<3->{\begin{itemize}
                \item Příliš málo epoch:
                síť se nestačí naučit vztahy v~datech.

                \visible<4->{\item Příliš mnoho epoch:
                roste riziko tzv \markdarkred{přeučení}.}

                \visible<5->{\item Sledujeme proto chybu na validační množině.}

                \visible<6->{\item Trénování ukončíme, pokud se validační chyba
                delší dobu nezlepšuje.}
            \end{itemize}}
        }\end{column}

        \begin{column}{0.48\textwidth}\visible<7->{
            \begin{center}
                \obrazek[0.4\textheight]{images/jak-volit-pocet-epoch-a-velikost-batche-2.tex}
            \end{center}

            \visible<8->{\begin{itemize}
                \item Menší batch znamená více a~méně přesných
                aktualizací parametrů.

                \visible<9->{\item Větší batch znamená méně aktualizací
                a~větší nároky na paměť.}

                \visible<10->{\item Vhodnou velikost volíme podle validačních
                výsledků a~výpočetních možností.}
            \end{itemize}}
        }\end{column}
    \end{columns}

\end{frame}

\begin{frame}{Jak volit počet epoch a~velikost batche?}
    \begin{timportantbox}{Praktické pravidlo}
        Neexistuje univerzálně nejlepší počet epoch ani velikost batche.
        Hodnoty volíme experimentálně podle výsledků na validační množině;
        počet epoch lze řídit pomocí předčasného ukončení
        (anglicky \markdarkred{early stopping}).
    \end{timportantbox}
\end{frame}

\subsection{Přeučení}

\begin{frame}{Přeučení modelu}

    \markdarkred{Přeučení} (anglicky \markdarkred{overfitting}) nastává tehdy,
    když se model \textbf{příliš přizpůsobí trénovacím datům} a~ztratí schopnost
    dobře pracovat s~novými daty.



    \visible<2->{\begin{itemize}
        \item Model se nenaučí pouze obecné vztahy, ale také šum a~náhodné zvláštnosti trénovací množiny.

        \visible<3->{\item Na trénovacích datech může mít velmi malou chybu.}

        \visible<4->{\item Na validačních nebo testovacích datech se však chyba začne zvyšovat.}
    \end{itemize}}



    \visible<5->{\begin{timportantbox}{Kdy k~přeučení často dochází?}
        Například při příliš dlouhém trénování, příliš složitém modelu,
        malém množství trénovacích dat nebo při slabé regularizaci.
    \end{timportantbox}}

\end{frame}

\begin{frame}{Přeučení v~průběhu trénování}

    \visible<2->{\obrazek[0.4\textheight]{images/preuceni-v-prubehu-trenovani.tex}}



    \visible<3->{\begin{tnoticebox}{Interpretace}
        Trénovací chyba obvykle s~počtem epoch klesá.
        Pokud však validační chyba začne růst, model se pravděpodobně
        začíná přizpůsobovat trénovacím datům až příliš.
    \end{tnoticebox}}

\end{frame}

\subsection{Hopfieldova síť}

\begin{frame}{Hopfieldova síť}

    \markdarkred{Hopfieldova síť} je tvořena \(n\) neurony, které jsou navzájem
    propojeny (tvoří úplný neorientovaný graf $K_n$). Síť tedy není rozdělená na vrstvy.



    \visible<2->{\obrazek[0.4\textheight]{images/hopfieldova-sit.tex}}

\end{frame}

\begin{frame}{Popis a~stav síťě}
    \footnotesize

    \visible<2->{\begin{itemize}
        \item Všechny neurony jsou vstupní i~výstupní.
        \visible<3->{\item Na Hopfieldovu síť lze popsat jako uspořádanou dvojici $\mathcal{H}_n=(N,w)$, kde $\mapping{w}{V\times V}{\R}$ je váhová funkce a~$|N|=n$.}
        \visible<4->{\item Pro jednotlivé váhy hran platí
        \[w_{ij}=w_{ji}\quad\text{a}\quad w_{ii}=0.\]}
        \visible<5->{\item Pracujeme se skokovou přenosovou funkcí.}
        \visible<6->{\item Pracujeme se vstupními (trénovacími) soubory $\mathbf{X}_1,\dots,\mathbf{X}_n$, kde
        \[\mathbf{x}_i=(x_{i1},\dots,x_{in}),\quad x_{ij}\in\set{-1,1}.\]}
        \visible<7->{\item Stav sítě $\mathcal{H}_n$ můžeme reprezentovat vektorem $\mathbf{s}=(s_1,\dots,s_n)$.}
    \end{itemize}}

    \visible<8->{\begin{timportantframe}
        Hopfieldův model představuje \textbf{učení bez učitele}, neboť zde nemáme výstupní hodnoty. Fungováním představuje model asociativní paměti.
    \end{timportantframe}}
\end{frame}

\begin{frame}{Aktualizace stavu Hopfieldovy sítě}

    \visible<2->{\begin{itemize}
        \item Označme číslem $t\in\N$ krok výpočtu sítě a~$\mathbf{s}_i^{(t)}$ stav $i$-tého neuronu v~kroku výpočtu $t$.
        \visible<3->{\item Pro \(i\)-tý neuron v~kroku $t+1$ spočítáme hodnotu
        \[
            s_i^{(t+1)}=f\left(\sum_{j=1}^{n}w_{ji}s_j^{(t)}+\theta_i\right),
        \]
        kde $f$ je skoková funkce.
        \item Výpočet skončít ve chvíli, kdy pro všechna $i$ je $s_i^{(t+1)}=s_i^{(t)}$.}
    \end{itemize}}



    \visible<4->{\begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{tnoticebox}{Synchronní aktualizace}
                Všechny neurony se přepočítají současně.
                Tato varianta je jednoduchá, ale může vést k~oscilacím.
            \end{tnoticebox}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{timportantbox}{Asynchronní aktualizace}
                Neurony se přepočítávají postupně.
                Tato varianta lépe souvisí s~konvergencí sítě.
            \end{timportantbox}
        \end{column}
    \end{columns}}

\end{frame}

\subsection{Hopfieldova síť jako asociativní paměť}

\begin{frame}{Hopfieldova síť jako asociativní paměť}

    \visible<2->{Cílem není určit třídu ani vypočítat číselný výstup.
    Síť má z~neúplného nebo poškozeného vstupu \textbf{obnovit některý
    z~uložených vzorů}.}


    \visible<3->{\obrazek[0.28\textheight]{images/hopfieldova-sit-jako-asociativni-pamet.tex}}



    \visible<4->{\begin{timportantframe}
        Během vybavování se nemění váhy sítě. Mění se pouze stav
        \[
            \mathbf{s}^{(0)},\mathbf{s}^{(1)},\mathbf{s}^{(2)},\ldots
        \]
        dokud se síť neustálí v~některém stabilním stavu.
    \end{timportantframe}}

\end{frame}

\subsection{Učení Hopfieldovy sítě}

\begin{frame}{Učení Hopfieldovy sítě}
    \small

    \visible<2->{Do sítě s~$n$ neurony ukládáme $p$ vzorů $\mathbf{x}_1,\dots,\mathbf{x}_p$, přičemž $r$-tý vzor je $\mathbf{x}_r=(x_{r1},\dots,x_{rn})$.}



    \visible<3->{\begin{tinfobox}{Hebbovo pravidlo}
        Váhu mezi \(i\)-tým a~\(j\)-tým neuronem určíme vztahem
        \[
            w_{ij}
            =
            \frac{1}{n}
            \sum_{r=1}^{p}
            x_{ri}x_{rj},
            \qquad i\neq j,
        \]
        a~současně položíme $w_{ii}=0$.
    \end{tinfobox}}



    \visible<4->{\begin{itemize}
        \item Pokud mají dva neurony ve vzorech často stejnou hodnotu,
              jejich váha bude spíše kladná.

        \visible<5->{\item Pokud mají často opačnou hodnotu, jejich váha bude spíše záporná.}

        \visible<6->{\item Váhy tedy zachycují vztahy mezi jednotlivými složkami uložených vzorů.}
    \end{itemize}}

\end{frame}

\begin{frame}{Energetická funkce}

    \visible<2->{Pro stav $\mathbf{s}=(s_1,\dots,s_n)$ definujeme energetickou funkci
    \[
        E(\mathbf{s})
        =
        -\frac{1}{2}
        \sum_{i=1}^{n}
        \sum_{j=1}^{n}
        w_{ij}s_is_j
        -
        \sum_{i=1}^{n}
        \theta_i s_i.
    \]}



    \visible<3->{\begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item Stav sítě si můžeme představit jako bod v~krajině.

                \visible<4->{\item Energetická funkce říká, jak „vysoko“ se daný stav nachází.}

                \visible<5->{\item Stabilní stavy odpovídají lokálním minimům energie.}
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}\visible<5->{
    \obrazek[0.4\textheight]{images/energeticka-funkce.tex}
        }\end{column}
    \end{columns}}

\end{frame}
