\subsection{Co je lineární regrese?}

\begin{frame}{Co je lineární regrese?}
    \pause
    Lineární regrese hledá lineární funkci
    \[
        f(x)=ax+b,
    \]
    která \markdarkred{co nejlépe popisuje vztah} mezi statistickými soubory
    \[
        \mathbf X=(x_1,\ldots,x_n),
        \qquad
        \mathbf Y=(y_1,\ldots,y_n).
    \]
    \pause

    \begin{itemize}
        \item Hodnota $f(x_i)$ je předpověď modelu pro vstup $x_i$.\pause
        \item Koeficient $a$ určuje směrnici přímky.\pause
        \item Koeficient $b$ určuje její posunutí.
    \end{itemize}
    \pause
    \begin{tinfoframe}[\textwidth]
        Cílem není najít přímku procházející všemi body, ale přímku s~co nejmenší celkovou chybou.
    \end{tinfoframe}
\end{frame}

\subsection{Predikce a rezidua}

\begin{frame}{Predikce a rezidua}
    \pause
    Pro každé pozorování budeme značit
    \[
        \hat y_i=f(x_i)
    \]
    jako předpovězenou hodnotu a
    \[
        e_i=y_i-\hat y_i
    \]
    nazýváme \markdarkred{reziduum}.
    \pause
    \vspace{0.2cm}

    \begin{columns}[T]
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item $e_i>0$: bod leží nad přímkou,
                \item $e_i<0$: bod leží pod přímkou,
                \item $e_i=0$: bod leží přesně na přímce.
            \end{itemize}
        \end{column}
        \pause
        \begin{column}{0.50\textwidth}
            \centering
            \begin{tikzpicture}[x=0.68cm,y=0.42cm,>=stealth]
                \draw[->] (0,0) -- (7,0) node[right] {$x$};
                \draw[->] (0,0) -- (0,6) node[above] {$y$};

                \draw[red!75!black,thick] (0.7,0.8) -- (6.5,5.1);

                    \fill[red!75!black] (\x,\yh) circle (1.5pt);
                }

                \node[gray] at (5.8,3.55) {$e_i$};
            \end{tikzpicture}
        \end{column}
    \end{columns}
\end{frame}

\subsection{Měření chyby}

\begin{frame}{Jak měřit celkovou chybu modelu?}
    \begin{itemize}
        \item Pouhý součet reziduí není vhodný:
        \[
            \sum_{i=1}^n e_i
        \]\pause
        protože kladné a~záporné chyby se mohou vyrušit.
        \item Proto používáme čtverce reziduí:
        \[
            e_i^2=(y_i-f(x_i))^2.
        \]
    \end{itemize}
\end{frame}

\subsection{Součet čtvercových chyb a střední kvadratická chyba}

\begin{frame}{SSE a~MSE}
    \begin{defblock}
        Nechť jsou dány statistické soubory $\mathbf X=(x_1,\ldots,x_n)$ a $\mathbf Y=(y_1,\ldots,y_n)$ a~funkce $\mapping{f}{\R}{\R}$. Pak
        \begin{itemize}
            \item \markdarkred{součet čtvercových chyb} (Error Sum of Squares) funkce $f$ definujeme jako
            \[
                \SSE_{\mathbf X,\mathbf Y}(f)
                =
                \sum_{i=1}^n
                \left(y_i-f(x_i)\right)^2
                =
                \sum_{i=1}^n e_i^{2}
                .
            \]
            \item \markdarkred{střední kvadratickou chybu} (Mean Square Error) funkce $f$ definujeme jako
            \[
                \MSE_{\mathbf X,\mathbf Y}(f)
                =
                \frac1n\SSE_{\mathbf X,\mathbf Y}(f).
            \]
        \end{itemize}
    \end{defblock}
\end{frame}

\subsection{Metoda nejmenších čtverců}

\begin{frame}{Metoda nejmenších čtverců}
    \pause
    \begin{itemize}
        \item Metoda nejmenších čtverců vybírá takovou funkci $f$, pro kterou je hodnota
        \[
            \sum_{i=1}^n
            \left(y_i-f(x_i)\right)^2
        \]
        nejmenší.\pause
        \item Pro lineární regresi tedy hledáme dvojici
        \[
            (a,b)=\argmin_{(a,b)\in\mathbb R^{2}}
            \sum_{i=1}^n
            (y_i-ax_i-b)^2.
        \]\pause
        \item Takovou optimální přímku budeme nazývat \markdarkred{regresní přímka souborů $\mathbf{X}$ a $\mathbf{Y}$}, nebo též jejich \markdarkred{lineární model}.
    \end{itemize}
    \pause
    \begin{tinfoframe}
        Minimalizace $\SSE$ a~$\MSE$ vede ke stejnému modelu, protože se liší pouze násobkem $\frac1n$. Proto budeme dále používat pro zjednodušení situace $\SSE$.
    \end{tinfoframe}
\end{frame}

\begin{frame}{Souvislost s aritmetickým průměrem}
    \pause
    V prezentaci o statistice jsme ukázali, že aritmetický průměr minimalizuje součet čtvercových odchylek:
    \[
        \average{\mathbf{X}}
        =
        \argmin_{c\in\mathbb R}
        \overbrace{\sum_{i=1}^n (x_i-c)^2}^{\SSE_{\mathbf X,\mathbf Y}(x\mapsto c)}.
    \]

    \pause

    \begin{tinfoframe}
        Aritmetický průměr je tedy nejlepší konstantní odhad hodnot souboru $\mathbf X$ ve~smyslu čtvercové chyby.
    \end{tinfoframe}

    \pause

    Lineární regrese tuto myšlenku zobecňuje:
    \begin{itemize}
        \item místo jedné konstanty hledáme přímku,
        \item místo odchylek od čísla měříme odchylky od hodnot $f(x_i)$.
    \end{itemize}
\end{frame}


\begin{frame}{Nejlepší konstantní model}
    \pause
    Nejprve uvažujme pouze konstantní model
    \[
        f(x)=b.
    \]
    pro soubory $\mathbf{X},\mathbf{Y}$. Chceme minimalizovat výraz
    \[
        \SSE_{\mathbf{X},\mathbf{Y}}(f)=\sum_{i=1}^n
        (y_i-b)^2.
    \]

    \pause

    Podle tvrzení o~aritmetickém průměru platí
    \[
        b=\average{\mathbf{Y}},\quad \text{a tedy}\;f(x)=\average{\mathbf{Y}}.
    \]
    \pause
    \begin{tinfoframe}[10cm]
        \begin{center}
            Tento model později porovnáme s~regresní přímkou.
        \end{center}
    \end{tinfoframe}
\end{frame}

\subsection{Hledání nejlepší přímky}

\begin{frame}{Hledání nejlepší přímky}
    \pause
    Pro lineární model
    \[
        f_{a,b}(x)=ax+b
    \]
    se soubory $\mathbf{X},\mathbf{Y}$ definujeme chybovou funkci
    \[
        E(a,b)=\SSE_{\mathbf{X},\mathbf{Y}}(f_{a,b})=\sum_{i=1}^n(y_i-ax_i-b)^2.
    \]
    \pause
    \begin{timportantframe}[\textwidth]
        Cílem je najít dvojici $(a,b)$ takovou, aby hodnota $E(a,b)$ byla nejmenší.
    \end{timportantframe}
    \pause
    Výpočet rozdělíme na dva kroky:
    \pause
    \begin{enumerate}
        \item najdeme nejlepší $b$ pro pevně zvolené $a$,\pause
        \item potom najdeme nejlepší směrnici $a$.
    \end{enumerate}
\end{frame}


\begin{frame}{Nejlepší hodnota $b$ pro pevné $a$}
    \pause
    \begin{itemize}
        \item Pro pevné $a$ můžeme psát
        \[
            E(a,b)
            =
            \sum_{i=1}^n(y_i-ax_i-b)^2
            =
            \sum_{i=1}^n
            \left((y_i-ax_i)-b\right)^2.
        \]\pause
        \item To je opět úloha hledání nejlepší konstanty pro soubor hodnot
        \[
            \mathbf{Y}-a\mathbf{X}=(y_1-ax_1,\dots,y_n-ax_n).
        \]\pause
        \item Proto dostáváme
        \[
            b=\average{\mathbf{Y}-a\mathbf{X}}=\average{\mathbf{Y}}-a\average{\mathbf{X}}.
        \]
    \end{itemize}
    
\end{frame}


\begin{frame}{Přímka procházející bodem průměrů}
    \pause
    \begin{itemize}
        \item Po dosazení
        \[
            b=\average{\mathbf{Y}}-a\average{\mathbf{X}}
        \]\pause
        lze model zapsat ve tvaru
        \[
            f(x)=\average{\mathbf{Y}}+a(x-\average{\mathbf{X}}).
        \]\pause
        \item Odtud ihned plyne
        \[
            f(\average{\mathbf{X}})=\average{\mathbf{Y}}.
        \]
    \end{itemize}
    \pause
    \begin{tnoticeframe}[11.5cm]
        \begin{center}
            Každý "kandidát" na~nejlepší přímku prochází bodem $(\average{\mathbf{X}},\average{\mathbf{Y}})$.
        \end{center}
    \end{tnoticeframe}

\end{frame}


\begin{frame}{Přímky procházející bodem $(\average{\mathbf{X}},\average{\mathbf{Y}})$}
    \centering
    \begin{tikzpicture}[x=0.85cm,y=0.55cm,>=stealth]
        \draw[->] (0,0) -- (7.2,0) node[right] {$x$};
        \draw[->] (0,0) -- (0,6.2) node[above] {$y$};

        \foreach \x/\y in {1/1.4,2/2.0,3/2.7,4/3.1,5/4.4,6/4.7}
            \fill[blue!70!black] (\x,\y) circle (2pt);

        \node[orange!90!black,above left] at (4.5,1.6) {$(\average{\mathbf{X}},\average{\mathbf{Y}})$};

        \draw[gray!60,thick] (0.8,2.3) -- (6.7,3.8);
        \draw[gray!60,thick] (0.8,4.0) -- (6.7,2.1);
        \draw[red!75!black,thick] (0.8,0.9) -- (6.7,5.2);

        \node[red!75!black] at (5.8,5.5) {nejlepší sklon?};
    \end{tikzpicture}

    \begin{tinfoframe}[\textwidth]
        Zbývá zvolit směrnici $a$ tak, aby výsledná přímka měla nejmenší chybu $E(a,b)$.
    \end{tinfoframe}
\end{frame}


\begin{frame}{Odvození směrnice $a$}
    \pause
    Po dosazení
    \[
        b=\average{\mathbf{Y}}-a\average{\mathbf{X}}
    \]
    dostaneme funkci jedné proměnné:
    \[
        E(a)=E(a,\average{\mathbf{Y}}-a\average{\mathbf{X}})
        =
        \sum_{i=1}^n
        \left(y_i-ax_i-(\average{\mathbf{Y}}-a\average{\mathbf{X}})\right)
        =
        \sum_{i=1}^n
        \left(
        (y_i-\average{\mathbf{Y}})
        -
        a(x_i-\average{\mathbf{X}})
        \right)^2.
    \]

    \pause

    Po roznásobení:
    \[
        E(a)
        =
        \sum_{i=1}^n
        (y_i-\average{\mathbf{Y}})^2 -\boxed{a}\cdot 2\sum_{i=1}^n
        (x_i-\average{\mathbf{X}})(y_i-\average{\mathbf{Y}})+\boxed{a^2}\sum_{i=1}^n (x_i-\average{\mathbf{X}})^2.
    \]
\end{frame}


\begin{frame}{Vzorec pro směrnici regresní přímky}
    \pause
    \begin{itemize}
        \item Funkce $E(a)$ je kvadratická funkce v proměnné $a$.\pause
        \item Minimum nastává pro
        \[
            a=
            \frac{
            \sum_{i=1}^n
            (x_i-\average{\mathbf{X}})(y_i-\average{\mathbf{Y}})
            }{
            \sum_{i=1}^n
            (x_i-\average{\mathbf{X}})^2
            }
            =
            \frac{
            \frac1n\sum_{i=1}^n
            (x_i-\average{\mathbf{X}})(y_i-\average{\mathbf{Y}})
            }{
            \frac1n\sum_{i=1}^n
            (x_i-\average{\mathbf{X}})^2
            }
            =
            \boxed{\frac{\Cov(\mathbf{X},\mathbf{Y})}{\Var(\mathbf{X})}}.
        \]\pause
        \item Jakmile známe $a$, dopočítáme
        \[
            \boxed{
            b=\average{\mathbf{Y}}-a\average{\mathbf{X}}
            }.
        \]
    \end{itemize}
    \pause
    \begin{talertframe}[10cm]
        \begin{center}
            \textbf{Podmínka:} hodnoty $x_i$ nesmějí být všechny stejné.
        \end{center}
    \end{talertframe}
\end{frame}

\subsection{Tvrzení o nejlepší regresní přímce}

\begin{frame}{Tvrzení o nejlepší regresní přímce}
    \begin{propblock}
        Nechť jsou dány statistické soubory $\mathbf{X},\mathbf{Y}$. Potom \markdarkred{existuje právě jedna} lineární funkce
        \[
            f(x)=ax+b,
        \]
        která minimalizuje součet čtvercových chyb, tj. 
        \[
            f=\argmin_{\ell\;\text{lineární}}\;\SSE_{\mathbf{X},\mathbf{Y}}(\ell)
        \]
        a~její koeficienty jsou
        \[
            a=\frac{\Cov(\mathbf{X},\mathbf{Y})}{\Var(\mathbf{X})},
            \qquad
            b=\average{\mathbf{Y}}-a\average{\mathbf{X}}.
        \]
    \end{propblock}
\end{frame}


\begin{frame}{Geometrická interpretace výsledku}
    \pause
    \begin{itemize}
        \item Regresní přímka prochází bodem
        \[
            (\average{\mathbf{X}},\average{\mathbf{Y}}).
        \]\pause
        \item Minimalizuje součet čtverců svislých odchylek bodů od přímky.
    \end{itemize}
    \pause
    \begin{talertframe}
        Směr regrese je podstatný:
        \[
            \text{regrese }\mathbf{Y}\text{ podle }\mathbf{X}
            \quad\neq\quad
            \text{regrese }\mathbf{X}\text{ podle }\mathbf{Y}.
        \]
        Výsledné přímky nejsou stejné.
    \end{talertframe}
\end{frame}

\subsection{Směrnice regresní přímky a korelace}

\begin{frame}{Směrnice regresní přímky a korelace}
    \pause
    \begin{itemize}
        \item Korelační koeficient je
        \[
            \rho_{\mathbf{X}\mathbf{Y}}
            =
            \frac{\Cov(\mathbf X,\mathbf Y)}
            {\sigma_\mathbf{X}\sigma_\mathbf{Y}}.
        \]\pause
        \item Proto $\Cov(\mathbf{X},\mathbf{Y})=\rho_{\mathbf{X}\mathbf{Y}}\sigma_\mathbf{X}\sigma_\mathbf{Y}$.\pause
        \item Po dosazení do vzorce pro $a$ dostaneme
        \[
            a=\frac{\Cov(\mathbf X,\mathbf Y)}{\Var(\mathbf X)}=\frac{\rho_{\mathbf{X}\mathbf{Y}}\sigma_\mathbf{X}\sigma_\mathbf{Y}}{\sigma^{2}_\mathbf{X}}=\rho_{\mathbf{X}\mathbf{Y}}\frac{\sigma_\mathbf{Y}}{\sigma_\mathbf{X}}.
        \]\pause
        \item Regresní přímku můžeme tedy vyjádřit jako
        \[
            \boxed{
            f(x)=
            \average{\mathbf{Y}}+
            \rho_{\mathbf{X}\mathbf{Y}}
            \frac{\sigma_\mathbf{Y}}{\sigma_\mathbf{X}}
            (x-\average{\mathbf{X}})
            }.
        \]
    \end{itemize}
    
\end{frame}


\begin{frame}{Co ze vztahu se směrnicí plyne?}
    \pause
    Ze vztahu
    \[
        a=
        \rho_{\mathbf{X}\mathbf{Y}}
        \frac{\sigma_\mathbf{Y}}{\sigma_\mathbf{X}}
    \]
    plyne:
    \pause
    \begin{itemize}
        \item znaménko směrnice je stejné jako znaménko korelace,\pause
        \item pokud $\rho_{\mathbf{X}\mathbf{Y}}>0$, přímka je rostoucí,\pause
        \item pokud $\rho_{\mathbf{X}\mathbf{Y}}<0$, přímka je klesající,\pause
        \item pokud $\rho_{\mathbf{X}\mathbf{Y}}=0$, potom $a=0$.
    \end{itemize}

    \begin{corblock}
        Jestliže $\rho_{\mathbf{X}\mathbf{Y}}=0$, nejlepší lineární model je $f(x)=\average{\mathbf{Y}}$.
    \end{corblock}
\end{frame}

\subsection{Korelace jako směrnice po standardizaci}

\begin{frame}{Korelace jako směrnice po standardizaci}
    \pause
    Definujme standardizované hodnoty souborů $\mathbf{X}$ a~$\mathbf{Y}$
    \[
        u_i=
        \frac{x_i-\average{\mathbf{X}}}{\sigma_\mathbf{X}},
        v_i=
        \frac{y_i-\average{\mathbf{Y}}}{\sigma_\mathbf{Y}}
        \quad
        \text{a}
        \quad
        \mathbf{U}=\set{u_1,\dots,u_n},\mathbf{V}=\set{v_1,\dots,v_n},
    \]\pause
    Pro standardizované soubory $\mathbf{U}$ a~$\mathbf{V}$ má regresní přímka tvar
    \[
        \boxed{
        v=\rho_{\mathbf{X}\mathbf{Y}}u
        }.
    \]
    \pause
    \begin{timportantframe}
        Korelační koeficient je směrnicí regresní přímky mezi standardizovanými hodnotami.
    \end{timportantframe}
    \pause
    \begin{itemize}
        \item Korelační koeficient nemá jednotky.\pause
        \item Změna jednotek nezmění korelační koeficient.
    \end{itemize}
\end{frame}

\subsection{Minimální chyba a korelace}

\begin{frame}{Minimální chyba a korelace}
    \pause
    \begin{propblock}[Bez důkazu]
        Pro soubory $\mathbf{X},\mathbf{Y}$ a jejich regresní přímku $f$ platí
        \[
            \MSE_{\mathbf{X},\mathbf{Y}}(f)
            =
            \sigma_\mathbf{Y}^2(1-\rho_{\mathbf{X}\mathbf{Y}}^2).
        \]
    \end{propblock}

    \pause

    \begin{itemize}
        \item Pokud $|\rho_{\mathbf{X}\mathbf{Y}}|=1$, potom
        \[
            \MSE_{\mathbf{X},\mathbf{Y}}(f)=0.
        \]
        Všechny body leží přesně na jedné přímce.
        \pause
        \item Pokud $\rho_{\mathbf{X}\mathbf{Y}}=0$, potom
        \[
            \MSE_{\mathbf{X},\mathbf{Y}}(f)=\sigma_\mathbf{Y}^2=\Var(\mathbf{Y}).
        \]
        Nejlepší lineární model se nezlepší oproti konstantnímu modelu.
    \end{itemize}
\end{frame}

\section{Koeficient determinace}

\titleframe{Koeficient determinace}

\subsection{Definice}

\begin{frame}{Koeficient determinace}
    \pause
    Chybu regresní přímky porovnáme s~chybou konstantního modelu
    \[
        f(x)=\average{\mathbf{Y}}.
    \]
    \pause
    \begin{defblock}[Koeficient determinace]
        Nechť jsou dány soubory $\mathbf X=(x_1,\ldots,x_n)$ a $\mathbf Y=(y_1,\ldots,y_n)$, kde $\sigma_\mathbf{Y}>0$ a~dále $f$ je jejich regresní přímka. \markdarkred{Koeficient determinace} modelu $f$ definujeme jako
        \[
            R^2=1-\frac{\MSE_{\mathbf X,\mathbf Y}(f)}{\MSE_{\mathbf X,\mathbf Y}(x\mapsto\average{\mathbf{Y}})}=1-\frac{\MSE_{\mathbf X,\mathbf Y}(f)}{\Var{\mathbf{Y}}}.
        \]
    \end{defblock}
    \pause
    Pro jednoduchou lineární regresi platí
    \[
        R^2=\rho_{\mathbf{X}\mathbf{Y}}^2.
    \]
\end{frame}

\begin{frame}{Příklad}
    \pause
    Uvažujme statistické soubory
    \[
        \mathbf X=(1,2,3,4,5),
        \qquad
        \mathbf Y=(2,3,5,4,6).

    \pause
    Dvojice $(x_i,y_i)$ představují jednotlivá pozorování:
    \[
        \begin{array}{c|ccccc}
            i   & 1 & 2 & 3 & 4 & 5 \\ \hline
            x_i & 1 & 2 & 3 & 4 & 5 \\
            y_i & 2 & 3 & 5 & 4 & 6
        \end{array}
    \]
    \pause
    \centering
    \begin{tikzpicture}
        \begin{axis}[
            width=7cm,

            };
        \end{axis}
    \end{tikzpicture}
\end{frame}

\begin{frame}{Příklad}{Výpočet regresní přímky}
    \pause
    Nejprve spočítáme aritmetické průměry:
    \[
        \average{\mathbf{X}}
        =
        \frac{1+2+3+4+5}{5}
        =
        3,\quad
        \average{\mathbf{Y}}
        =
        \frac{2+3+5+4+6}{5}
        =
        4.
    \]
    \pause
    Směrnice regresní přímky je
    \[
        a
        =
        \frac{
            \sum_{i=1}^5
            (x_i-\average{\mathbf{X}})(y_i-\average{\mathbf{Y}})
        }{
            \sum_{i=1}^5
            (x_i-\average{\mathbf{X}})^2
        }
        =\frac{
            4+1+0+0+4
        }{
            4+1+0+1+4
        }
        =
        \frac9{10}
        =
        0{,}9.
    \]
    \pause
    Absolutní člen je
    \[
        b
        =
        \average{\mathbf{Y}}-a\average{\mathbf{X}}
        =
        4-0{,}9\cdot 3
        =
        1{,}3.
    \]
    \pause
    \begin{timportantframe}[9cm]
        \begin{center}
            Hledaná regresní přímka je $f(x)=0{,}9x+1{,}3$.
        \end{center}
    \end{timportantframe}
\end{frame}

\begin{frame}{Příklad}{Předpovědi a rezidua}
    \pause
    Pro regresní model
    \[
        f(x)=0{,}9x+1{,}3
    \]
    vypočítáme předpovědi $\hat y_i=f(x_i)$ a~rezidua $e_i=y_i-\hat y_i$.
    \pause
    \[
        \begin{array}{c|ccccc}
            i
                & 1 & 2 & 3 & 4 & 5 \\ \hline
            x_i
                & 1 & 2 & 3 & 4 & 5 \\
            y_i
                & 2 & 3 & 5 & 4 & 6 \\
            \hat y_i
                & 2{,}2 & 3{,}1 & 4{,}0 & 4{,}9 & 5{,}8 \\

                & -0{,}2 & -0{,}1 & 1{,}0 & -0{,}9 & 0{,}2

        \SSE_{\mathbf{X},\mathbf{Y}}(f)&=\sum_{i=1}^5 e_i^2=0{,}04+0{,}01+1+0{,}81+0{,}04=1{,}9.\\
        \MSE_{\mathbf{X},\mathbf{Y}}(f)&=\frac{1{,}9}{5}=0{,}38.

\begin{frame}{Příklad}{Koeficient determinace}
    \pause
    Nejlepší konstantní model je
    \[
        c(x)=\average{\mathbf{Y}}=4.
    \]
    \pause
    Jeho chyba je
    \[
    \begin{aligned}
        \MSE_{\mathbf X,\mathbf Y}(c)
        &=
        \frac15
        \sum_{i=1}^5
        (y_i-\average{\mathbf{Y}})^2=
        \frac15
        \left(
            (-2)^2+(-1)^2+1^2+0^2+2^2
        \right)=2.
    \end{aligned}
    \]
    \pause
    Pro regresní přímku jsme vypočítali $\MSE_{\mathbf X,\mathbf Y}(f)=0{,}38$.
    \pause
    Proto
    \[
        R^2
        =
        1-
        \frac{
            \MSE_{\mathbf X,\mathbf Y}(f)
        }{
            \MSE_{\mathbf X,\mathbf Y}(c)
        }
        =
        1-\frac{0{,}38}{2}
        =
        0{,}81.
    \]

    Regresní přímka odstranila $81\,\%$ střední kvadratické chyby nejlepšího konstantního modelu.
\end{frame}

\begin{frame}{Příklad}{Souvislost s~korelací}
    \pause
    Pro zadané soubory $\mathbf{X},\mathbf{Y}$ platí
    \[
        \sigma_\mathbf{X}
        =
        \sqrt{\frac15
        \sum_{i=1}^5
        (x_i-\average{\mathbf{X}})^2}
        =
        \sqrt{\frac{10}{5}}
        =
        \sqrt{2}
    \]
    a stejně $\sigma_\mathbf{Y}=\sqrt{\frac15\sum_{i=1}^5(y_i-\average{\mathbf{Y}})^2}=\sqrt{2}$.\\[1mm]
    \pause
    Kovariance je $\Cov(\mathbf X,\mathbf Y)=\frac15\cdot 9=1{,}8$. Proto
    \[
        \rho_{\mathbf X\mathbf Y}=
        \frac{
            \Cov(\mathbf X,\mathbf Y)
        }{
            \sigma_\mathbf{X}\sigma_\mathbf{Y}
        }=
        \frac{1{,}8}{\sqrt2\sqrt2}
        =
        0{,}9.
    \]
    \pause
    Pro jednoduchou lineární regresi tedy skutečně platí
    \[
        \boxed{
            R^2=\rho_{\mathbf X\mathbf Y}^2=0{,}9^2=0{,}81
        }.
    \]
\end{frame}

\subsection{Kdy dává lineární regrese smysl?}

\begin{frame}{Kdy dává lineární regrese smysl?}
    \pause
    Před použitím lineární regrese očekáváme zejména:
    \pause
    \begin{itemize}
        \item vztah mezi $\mathbf{X}$ a $\mathbf{Y}$ je přibližně lineární,\pause
        \item pozorování nejsou vzájemně závislá,\pause
        \item rezidua nevykazují systematický vzor,\pause
        \item velikost reziduí se s~rostoucími hodnotami v~$\mathbf{X}$ výrazně nemění.
    \end{itemize}
    \pause
    \begin{tinfoframe}
        Tyto podmínky zde chápeme intuitivně: bodový graf a~rezidua by neměly ukazovat zjevný problém.
    \end{tinfoframe}
\end{frame}

\section{Shrnutí}

\begin{frame}{Shrnutí}
    \pause
    \begin{itemize}
        \item \textbf{Lineární regrese} hledá přímku
        \[
            f(x)=ax+b,
        \]
        která co nejlépe odhaduje hodnoty souboru $\mathbf Y$
        z hodnot souboru $\mathbf X$.\pause

        \item Koeficienty určujeme \textbf{metodou nejmenších čtverců},
        tedy minimalizujeme
        \[
            \MSE_{\mathbf X,\mathbf Y}(f)
            =
            \frac1n\sum_{i=1}^n
            (y_i-f(x_i))^2.
        \]
        \pause
        \item Nejlepší regresní přímka prochází bodem $(\average{\mathbf{X}},\average{\mathbf{Y}})$ a~její koeficienty jsou
        \[
            a=
            \frac{\Cov(\mathbf X,\mathbf Y)}
                 {\Var(\mathbf X)}
            =
            \rho_{\mathbf{X}\mathbf{Y}}\frac{\sigma_Y}{\sigma_X},
            \qquad
            b=\average{\mathbf{Y}}-a\average{\mathbf{X}}.
        \]
    \end{itemize}

\end{frame}

\begin{frame}{Shrnutí}
    \pause
    \begin{itemize}
        \item \textbf{Koeficient determinace} porovnává regresní přímku $f$ s~nejlepším konstantním modelem $c(x)=\average{\mathbf{Y}}$:
        \[
            R^2
            =
            1-
            \frac{
                \MSE_{\mathbf X,\mathbf Y}(f)
            }{
                \MSE_{\mathbf X,\mathbf Y}
                (x\mapsto\average{\mathbf{Y}})
            }.
        \]
    \end{itemize}
    \pause
    \begin{timportantframe}
        Pro jednoduchou lineární regresi platí
        \[
            \boxed{R^2=\rho_{\mathbf{X}\mathbf{Y}}^2}.
        \]
        Čím je $R^2$ blíže jedné, tím větší část chyby konstantního modelu odstraníme použitím regresní přímky.
    \end{timportantframe}
    \pause
    \begin{talertframe}
        Lineární regrese zachycuje pouze lineární vztah, je citlivá na~odlehlé hodnoty a~sama o~sobě nedokazuje příčinnou souvislost.
    \end{talertframe}
    
\end{frame}
