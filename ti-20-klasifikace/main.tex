    \end{tnoticebox}

            \end{tnoticebox}

        \end{tabular}
    \end{center}

\begin{frame}{Výstupní statistický soubor}
    \pause
    \begin{itemize}
        \item Množinu všech možných tříd označíme
        \[
            \mathcal{Y}=\{c_1,c_2,\ldots,c_K\}.
        \]\pause
        \item Nechť
        \[
            \mathbf{Y}=(y_1,y_2,\ldots,y_n)
        \]
        je statistický soubor tříd jednotlivých objektů $\mathbf{x}_1,\dots,\mathbf{x}_n$.\pause
        \item Trénovací množinu představuje
        \[D=\set{(\mathbf{x}_1,y_1),
                (\mathbf{x}_2,y_2),
                \ldots,
                (\mathbf{x}_n,y_n)}\]
    \end{itemize}

    \vspace{1mm}
    \pause
    \begin{tinfoframe}[9.5cm]
        \begin{center}
            U~binární klasifikace často používáme $ \mathcal{Y}=\set{0,1}$.
        \end{center}
    \end{tinfoframe}

\end{frame}

\begin{frame}{Klasifikátor}
    \pause
    \begin{defblock}[Klasifikátor]
        Nechť \(\mathcal{Y}\) je množina tříd.
        \markdarkred{Klasifikátor} je funkce
        \[
            \mapping{h}{\R^p}{\mathcal{Y}},
        \]
        která každému vektoru znaků $\mathbf{x}\in\R^{p}$ přiřadí jednu třídu $c\in\mathcal{Y}$.
    \end{defblock}

    \vspace{2mm}
    \pause
    \begin{center}
        \input{images/klasifikator.tex}
    \end{center}

    \vspace{1mm}
    \pause
    \begin{tinfoframe}[\textwidth]
        V~praxi klasifikátor neznáme předem. Vytváříme ho na základě
        trénovací množiny \(D\).
    \end{tinfoframe}

\end{frame}

\begin{frame}{Cíl klasifikační úlohy}
    \pause
    \begin{tnoticebox}{Hlavní cíl}
        Cílem klasifikace není pouze správně zařadit objekty
        z~trénovací množiny, ale především správně klasifikovat nové,

    \end{tnoticebox}

        příznaky a~třídou, nikoli si pouze "zapamatovat" trénovací data.
    \end{timportantframe}

\end{frame}

\subsection{Druhy klasifikačních úloh}

\begin{frame}{Binární a~vícetřídní klasifikace}
    \pause
    \begin{itemize}
        \item Řekneme, že klasifikační úloha je \textbf{binární}, jestliže množina tříd má tvar
        \[
            \mathcal{Y}=\set{0,1}.
        \]
        \pause
        \item Řekneme, že klasifikační úloha je \textbf{vícetřídní} (anglicky \markdarkred{multi class classification}), jestliže množina tříd má tvar
        \[
            \mathcal{Y}=\{c_1,c_2,\ldots,c_K\},
            \qquad K\geqslant 3.
        \]
    \end{itemize}

    \vspace{1mm}
    \pause
    \begin{tinfoframe}
        V~obou případech každému objektu přiřazujeme právě jednu třídu $y_i\in\mathcal{Y}$.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Víceznačková klasifikace}
    \pause
    Řekneme, že klasifikační úloha je \textbf{víceznačková} (anglicky \markdarkred{multi-label classification}), jestliže jeden objekt může současně patřit do více tříd.
    \pause
    \vspace{2mm}

    Místo jedné třídy \(y_i\in\mathcal{Y}\) pak můžeme výsledek pro
    \(i\)-tý objekt zapsat například jako vektor
    \[
        \mathbf{y}_i=(y_{i1},y_{i2},\ldots,y_{iK})\in\{0,1\}^K,
    \]
    kde
    \[
        y_{ij}=1
        \quad\Longleftrightarrow\quad
        i\text{-tý objekt má }j\text{-tou značku}.
    \]
    \pause
    \vspace{1mm}

    Například fotografie může mít současně značky \emph{člověk}, \emph{auto} a~\emph{budova}.
    \pause
    \begin{timportantframe}
        V~této prezentaci se budeme dále věnovat především binární
        a~vícetřídní klasifikaci.
    \end{timportantframe}

\end{frame}

\subsection{Skóre a rozhodování klasifikátoru}

\begin{frame}{Skóre modelu}
    \pause
    \begin{defblock}{Skóre modelu}
        Nechť jsou dány vstupní statistické soubory
        \[
            \mathbf{X}_1,\mathbf{X}_2,\ldots,\mathbf{X}_p
        \]
        a~výstupní statistický soubor $\mathbf{Y}=(y_1,y_2,\ldots,y_n)$. \markdarkred{Skórovací funkce} je funkce
        \[
            \mapping{s}{\R^p}{\R}
        \]
        Hodnotu $s(\mathbf{x})$ nazýváme \markdarkred{skóre} prvku $\mathbf{x}\in\R^{p}$.
    \end{defblock}
    \pause
    Pro $i$-tý objekt položíme
    \[
        s_i=s(\mathbf{x}_i),
        \qquad
        \mathbf{x}_i=(x_{i1},x_{i2},\ldots,x_{ip}).
    \]
    \pause
    Skóre všech objektů tvoří statistický soubor
    \[
        \mathbf{S}=(s_1,s_2,\ldots,s_n).
    \]

\end{frame}

\begin{frame}{Od skóre k~předpovězené třídě}{Binární klasifikace}
    \pause
    Pro množinu tříd $\mathcal{Y}=\set{0,1}$ může model každému objektu \(\mathbf{x}_i\) přiřadit jediné skóre
    \[
        s_i=s(\mathbf{x}_i).
    \]
    \pause
    Pomocí rozhodovacího prahu \(t\) pak položíme
    \[
        \hat y_i=
        \begin{cases}
            1, & s_i\geqslant t,\\
            0, & s_i<t.
        \end{cases}
    \]

\end{frame}

\begin{frame}{Od skóre k~předpovězené třídě}{Vícetřídní klasifikace}
    \pause
    Pro množinu tříd $\mathcal{Y}=\set{c_1,c_2,\ldots,c_K}$ model vypočítá pro každý objekt skóre všech tříd:
    \[
        s_1(\mathbf{x}_i),\,
        s_2(\mathbf{x}_i),\ldots,
        s_K(\mathbf{x}_i).
    \]
    \pause
    Objekt přiřadíme třídě s~nejvyšším skóre:
    \[
        \begin{aligned}
            \hat k_i
            &=
            \argmax_{1\leqslant k\leqslant K}
            s_k(\mathbf{x}_i),\\
            \hat y_i&=c_{\hat k_i}.
        \end{aligned}
    \]
    \pause
    Predikované hodnoty všech objektů tvoří statistický soubor
    \[
        \hat{\mathbf{Y}}
        =
        (\hat y_1,\hat y_2,\ldots,\hat y_n).
    \]
    \pause
    \begin{tinfoframe}[12cm]
        \begin{center}
            Tedy pro soubory $\mathbf{X}_1,\dots,\mathbf{X}_p,\mathbf{Y}$ jsme získali soubor odhadů $\hat{\mathbf{Y}}$.
        \end{center}
    \end{tinfoframe}
\end{frame}

\begin{frame}{Rozhodovací hranice}
    \pause
    \begin{defblock}[Rozhodovací hranice]
        Nechť \(s\colon\R^p\to\R\) je skórovací funkce
        a~\(t\in\R\) rozhodovací práh.

        \markdarkred{Rozhodovací hranicí} (anglicky \markdarkred{decision boundary}) rozumíme množinu
        \[
            B_t
            =
            \set{\mathbf{x}\in\R^p : s(\mathbf{x})=t}.
        \]
    \end{defblock}
    \pause
    \begin{tinfoframe}[\textwidth]
        Rozhodovací hranice odděluje oblasti, v~nichž model přiřazuje
        objektům různé třídy.
    \end{tinfoframe}

\end{frame}

\end{frame}

\subsection{Proč nestačí lineární regrese?}

\begin{frame}{Připomenutí: lineární regrese}
    \pause
    \begin{itemize}
        \item Nechť jsou dány statistické soubory
        \[
            \mathbf{X}=(x_1,x_2,\ldots,x_n),
            \quad
            \mathbf{Y}=(y_1,y_2,\ldots,y_n).
        \]\pause
        Lineární model má tvar
        \[
            f(x)=ax+b.
        \]\pause
        \item Pro $i$-tý objekt dostaneme odhad
        \[
            \hat y_i=f(x_i)=ax_i+b.
        \]
    \end{itemize}
    \pause
    \begin{tinfoframe}
        \textbf{Problém:} lineární model může vracet libovolné reálné číslo, zatímco u~klasifikace chceme rozhodnout mezi třídami.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Lineární regrese pro třídy \(0\) a~\(1\)}

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.48\textwidth}
            \small
            \pause
            \begin{itemize}
                \item U~binární klasifikace máme
                \[
                    \mathbf{Y}=(y_1,\ldots,y_n),
                    \qquad
                    y_i\in\set{0,1}.
                \]
                \pause
                \item Lineární model můžeme formálně použít: $\hat y_i=f(x_i)$.
                \pause
                \item Abychom získali třídu, mohli bychom zvolit práh:
                \[
                    \hat c_i=
                    \begin{cases}
                        1, & \hat y_i\geqslant 0{,}5,\\
                        0, & \hat y_i<0{,}5.
                    \end{cases}
                \]
            \end{itemize}
        \end{column}
        \pause
        \begin{column}{0.50\textwidth}
            \begin{center}
                \input{images/linearni-regrese-pro-tridy-0-a-1.tex}
            \end{center}
        \end{column}

    \end{columns}
    \pause
    \begin{timportantframe}
        Takový postup sice dá klasifikátor, ale výstupy lineárního modelu
        nejsou omezeny na interval \(\langle0,1\rangle\).
    \end{timportantframe}

\end{frame}

\begin{frame}{Jaký výstup potřebujeme?}
    \pause
    \begin{tnoticebox}{Požadavek na model}
        Chceme zachovat lineární skóre $s(x)=ax+b$, ale převést jej na číslo z~intervalu \((0,1)\).
    \end{tnoticebox}

    \vspace{3mm}

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.48\textwidth}
            \pause

            \begin{itemize}
                \item záporná skóre převádí blízko k~\(0\),
                \item kladná skóre převádí blízko k~\(1\),
                \item hodnota \(z=0\) odpovídá středu:
                \[
                    g(0)=0{,}5.
                \]
            \end{itemize}

\end{frame}

\subsection{Logistický model}

\begin{frame}{Logistická funkce}
    \pause
    \begin{defblock}[Logistická funkce]
        \markdarkred{Logistická funkce} (též \markdarkred{sigmoida}) je funkce
        \[
            \sigma(z)=\frac{1}{1+e^{-z}},
            \qquad z\in\mathbb{R}.
        \]
    \end{defblock}

        \pause
        \begin{column}{0.44\textwidth}
            \small
            Platí:
            \[
                \sigma(z)\in(0,1)\quad\text{a}\quad \sigma(0)=0{,}5.
            \]

            \begin{tinfoframe}
                Velmi záporné hodnoty převádí blízko k~\(0\),
                velmi kladné hodnoty blízko k~\(1\).
            \end{tinfoframe}
        \end{column}
        \pause
        \begin{column}{0.54\textwidth}
            \begin{center}
                \input{images/logisticka-funkce.tex}
            \end{center}
        \end{column}
    \end{columns}

\end{frame}

\subsection{Od lineární k logistické regresi}

\begin{frame}{Od lineární k logistické regresi}
    \pause
    Nechť jsou dány statistické soubory
    \[
        \mathbf{X}=(x_1,x_2,\ldots,x_n),
        \qquad
        \mathbf{Y}=(y_1,y_2,\ldots,y_n),
        \qquad
        y_i\in\{0,1\}.
    \]
    \pause
    \begin{tnoticebox}{Myšlenka logistické regrese}
        Nejprve vypočítáme lineární skóre $z_i=ax_i+b$. Poté jej převedeme logistickou funkcí na $\hat p_i=\sigma(z_i)$.
    \end{tnoticebox}

    \vspace{1mm}
    \pause
    Tím dostaneme statistický soubor výstupů
    \[
        \hat{\mathbf{P}}
        =
        (\hat p_1,\hat p_2,\ldots,\hat p_n),
        \qquad
        \hat p_i\in(0,1).
    \]
    \pause
    \begin{tinfoframe}
        Hodnotu \(\hat p_i\) budeme chápat jako míru podpory modelu
        pro zařazení objektu do třídy~\(1\).
    \end{tinfoframe}

\end{frame}

\begin{frame}{Model logistické regrese}
    \pause
    \begin{itemize}
        \item Logistická regrese přiřadí hodnotě \(x\) číslo
        \[
            \hat p(x)
            =
            \sigma(ax+b)
            =
            \frac{1}{1+e^{-(ax+b)}}.
        \]\pause
        \item Pro $i$-tý prvek souboru $\mathbf{X}$ tedy platí
        \[
            \hat p_i
            =
            \hat p(x_i)
            =
            \frac{1}{1+e^{-(ax_i+b)}}.
        \]
    \end{itemize}
    \pause
    \begin{timportantframe}[\textwidth]
        Parametry \(a\) a~\(b\) určují tvar modelu. Při trénování hledáme
        takové hodnoty, aby model dobře odpovídal trénovacím datům (viz lineární regrese).
    \end{timportantframe}

\end{frame}

\subsection{Rozhodovací pravidlo}

\begin{frame}{Od výstupu modelu k~předpovězené třídě}

    Nechť logistický model přiřadí hodnotě \(x_i\) výstup
    \[
        \hat p_i
        =
        \hat p(x_i)
        =
        \frac{1}{1+e^{-(ax_i+b)}}.
    \]

    \begin{tnoticebox}{Rozhodovací pravidlo}
        Zvolme rozhodovací práh $t\in(0,1)$. Předpovězenou třídu \(i\)-tého objektu určíme vztahem
        \[
            h^{(t)}(x)
            =
            \hat y_i^{(t)}
            =
            \begin{cases}
                1, & \hat p(x)\geqslant t,\\[1mm]
                0, & \hat p(x)<t.
            \end{cases}
        \]
    \end{tnoticebox}

    \vspace{1mm}

    Předpovězené třídy všech obektů z~$\mathbf{X}$ tvoří statistický soubor
    \[
        \hat{\mathbf{Y}}^{(t)}
        =
        (\hat y_1^{(t)},\hat y_2^{(t)},\ldots,\hat y_n^{(t)}).
    \]

\end{frame}

\begin{frame}{Kdy model předpoví třídu \(1\)?}
    \pause
    \begin{propblock}
        Nechť $\hat p(x)=\sigma(ax+b)$ pro $a,b\in\R$ a~\(t\in(0,1)\). Potom platí
        \[
            \hat p(x)\geqslant t
            \quad\Longleftrightarrow\quad
            ax+b\geqslant\ln\left(\frac{t}{1-t}\right).
        \]
    \end{propblock}

    \pause

    \textbf{Důkaz:} Logistická funkce je rostoucí, a~proto
    \[
        \hat p(x)\geqslant t
        \quad\Longleftrightarrow\quad
        ax+b\geqslant\sigma^{-1}(t)=\ln\left(\frac{t}{1-t}\right).
    \]
    Z~toho dostáváme požadovanou ekvivalenci.
    \pause
    \begin{timportantframe}
        Speciálně pro \(t=0{,}5\) je $\ln\left(\frac{0{,}5}{1-0{,}5}\right)=\ln{1}=0$, takže o~třídě rozhoduje pouze znaménko výrazu \(ax+b\).
    \end{timportantframe}

\end{frame}

\begin{frame}{Rozhodovací hranice logistické regrese}
    \pause
    Ze vztahu
    \[
        ax_t+b
        =
        \ln\left(\frac{t}{1-t}\right)
    \]
    dostáváme
    \[
        x_t
        =
        \frac{
            \ln\left(\frac{t}{1-t}\right)-b
        }{a}.
    \]
    \pause
    \begin{center}

\subsection{Křížová entropie}

\begin{frame}{Proč potřebujeme ztrátovou funkci?}
    \pause
    Nechť logistický model pro statistické soubory
    \[
        \mathbf{X}=(x_1,\ldots,x_n),
        \qquad
        \mathbf{Y}=(y_1,\ldots,y_n),
        \qquad y_i\in\{0,1\},
    \]
    vrací hodnoty
    \[
        \hat{\mathbf{P}}
        =
        (\hat p_1,\ldots,\hat p_n).
    \]
    \pause
    \begin{tnoticebox}{Myšlenka}
        Při trénování nechceme hodnotit pouze výslednou třídu
        \(h^{(t)}(x_i)\), ale také to, \textbf{jak jistý} si model byl.
    \end{tnoticebox}

    \vspace{1mm}
    \pause
    \begin{center}
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{c|c|c}
            skutečnost & výstup modelu pro $t=0{,}5$ & komentář \\
            \hline
            \(y_i=1\) & \(\hat p_i=0{,}51\) & \markdarkgreen{správně}, ale \markdarkred{nejistě} \\
            \(y_i=1\) & \(\hat p_i=0{,}99\) & \markdarkgreen{správně} a~\markdarkgreen{jistě} \\
            \(y_i=1\) & \(\hat p_i=0{,}01\) & \markdarkred{chybně} a~\markdarkgreen{velmi jistě}
        \end{tabular}
    \end{center}

\end{frame}

\begin{frame}{Křížová entropie pro jeden objekt}
    \pause
    \begin{defblock}[Ztráta jednoho objektu]
        Pro skutečnou třídu \(y_i\in\{0,1\}\) a~výstup modelu
        \(\hat p_i\in(0,1)\) definujeme ztrátu
        \[
            \ell(y_i,\hat p_i)
            =
            -
            \left(
                y_i\ln \hat p_i
                +
                (1-y_i)\ln(1-\hat p_i)
            \right).
        \]
        Tato ztrátová funkce se nazývá \markdarkred{binární křížová entropie} (anglicky \markdarkred{binary cross-entropy}).
    \end{defblock}
    
    \vspace{1mm}
    \pause
    Po dosazení dostaneme jednodušší tvar:
    \[
        \ell(1,\hat p_i)=-\ln \hat p_i,
        \qquad
        \ell(0,\hat p_i)=-\ln(1-\hat p_i).
    \]
    \pause
    \begin{tinfoframe}
        Správná a~jistá odpověď má malou ztrátu.
        Sebejistá chybná odpověď má naopak ztrátu velmi vysokou.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Křížová entropie pro celý soubor}
    \pause
    \begin{defblock}[Průměrná křížová entropie]
        Nechť
        \[
            \mathbf{Y}=(y_1,\ldots,y_n),
            \qquad
            \hat{\mathbf{P}}=(\hat p_1,\ldots,\hat p_n).
        \]
        \markdarkred{Průměrnou křížovou entropii} definujeme vztahem
        \[
            L(\mathbf{Y},\hat{\mathbf{P}})=\frac1n\sum_{i=1}^{n}\ell(y_i,\hat p_i)
            =
            -\frac1n
            \sum_{i=1}^{n}
            \left(
                y_i\ln \hat p_i
                +
                (1-y_i)\ln(1-\hat p_i)
            \right).
        \]
    \end{defblock}

    \vspace{1mm}
    \pause
    \begin{tnoticebox}{Trénování logistické regrese}
        Při trénování hledáme parametry \(a\) a~\(b\), pro které je
        hodnota $L(\mathbf{Y},\hat{\mathbf{P}})$ co nejmenší.
    \end{tnoticebox}

\end{frame}

\begin{frame}{Chování křížové entropie}

    \begin{columns}[T,onlytextwidth]
        \pause
        \begin{column}{0.48\textwidth}
            \small
            Pro objekt se skutečnou třídou \(y=1\) je ztráta
            \[
                \ell(1,\hat p)=-\ln \hat p.
            \]

            \vspace{1mm}

            \begin{center}
                \renewcommand{\arraystretch}{1.25}
                \begin{tabular}{c|c}
                    \(\hat p\) & \(\ell(1,\hat p)\) \\
                    \hline
                    \(0{,}9\) & \(0{,}11\) \\
                    \(0{,}5\) & \(0{,}69\) \\
                    \(0{,}1\) & \(2{,}30\) \\
                    \(0{,}01\) & \(4{,}61\)
                \end{tabular}
            \end{center}
        \end{column}
        \pause
        \begin{column}{0.50\textwidth}
            \begin{center}
                \input{images/chovani-krizove-entropie.tex}
            \end{center}
        \end{column}

    \end{columns}
    \pause
    \begin{timportantframe}[\textwidth]
        Čím menší hodnotu \(\hat p\) model přiřadí objektu,
        který skutečně patří do třídy~\(1\), tím větší ztrátu dostane.
    \end{timportantframe}

\end{frame}

\subsection{Trénování logistické regrese}

\begin{frame}{Trénování logistického modelu}
    \pause
    \begin{itemize}
        \item Nechť jsou dány statistické soubory
        \[
            \mathbf{X}=(x_1,x_2,\ldots,x_n),
            \qquad
            \mathbf{Y}=(y_1,y_2,\ldots,y_n),
            \qquad
            y_i\in\{0,1\}.
        \]\pause
        \item Pro zvolené parametry \(a,b\in\mathbb{R}\) model vrací hodnoty
        \[
            \hat p_i(a,b)
            =
            \frac{1}{1+e^{-(ax_i+b)}}.
        \]\pause
        \item Parametry modelu volíme tak, aby byla průměrná křížová entropie
        co nejmenší:
        \[
            (a^\ast,b^\ast)
            =
            \argmin_{(a,b)\in\mathbb{R}^2}
            L(a,b),
        \]
        kde
        \[
            L(a,b)
            =
            -\frac1n\sum_{i=1}^{n}
            \left(
                y_i\ln\hat p_i(a,b)
                +(1-y_i)\ln\left(1-\hat p_i(a,b)\right)
            \right).
        \]
    \end{itemize}

\end{frame}

\begin{frame}{Jak hledáme vhodné parametry?}
    \pause
    Hodnoty \(a^\ast\) a~\(b^\ast\) obvykle nelze získat jednoduchým
    dosazením do vzorce. Hledáme je proto pomocí numerické optimalizace.

    \vspace{2mm}
    \pause
    \begin{center}
        \input{images/jak-hledame-vhodne-parametry.tex}
    \end{center}

    \vspace{2mm}
    \pause
    \begin{tnoticebox}{Postup učení}
        Tento postup opakujeme, dokud se hodnota ztrátové funkce
        dostatečně nezmenší nebo se parametry nepřestanou výrazně měnit. K~tomu se používá např. tzn. \textbf{gradientní sestup}.
    \end{tnoticebox}

\end{frame}

\begin{frame}{Myšlenka gradientního sestupu}

    \begin{columns}[T,onlytextwidth]
        \pause
        \begin{column}{0.47\textwidth}
            \small
            Při trénování chceme minimalizovat ztrátovou funkci
            \(L(a,b)\).

            \begin{tnoticebox}{Základní princip}
                Začneme s~nějakými hodnotami parametrů a~postupně je
                upravujeme směrem, ve~kterém se ztráta nejrychleji zmenšuje.
            \end{tnoticebox}
        \end{column}
        \pause
        \begin{column}{0.51\textwidth}
            \begin{center}
                \input{images/myslenka-gradientniho-sestupu.tex}
            \end{center}
        \end{column}

    \end{columns}
    \pause
    V~jednorozměrném případě:
    \[
        \theta_{k+1}
        =
        \theta_k-\eta L'(\theta_k),
    \]
    kde \(\eta>0\) je velikost kroku.\\[1mm]
    \pause
    Pro logistickou regresi: $\displaystyle (a,b)\gets (a,b) - \eta\left(\frac{\partial L}{\partial a},\frac{\partial L}{\partial b}\right)$.

\end{frame}

\subsection{Konkrétní příklad}

\begin{frame}{Příklad}{Zadání klasifikační úlohy}
    \pause
    Sledujeme, kolik hodin se student připravoval na test:
    \[
        \mathbf{X}=(1,2,3,4,5,6,7,8),\quad \mathbf{Y}=(0,0,0,1,0,1,1,1),
    \]
    kde \(y_i=1\) znamená, že student test úspěšně zvládl.
    \pause
    \begin{tnoticebox}[8cm]{Naučený model}
        Trénováním jsme získali logistický model
        \[
            \hat p(x)
            =
            \frac{1}{1+e^{-(1{,}28x-5{,}77)}}.
        \]
    \end{tnoticebox}
    \pause
    \begin{tinfoframe}
        Nyní chceme klasifikovat nového studenta, který se připravoval
        \(x=5\) hodin.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Příklad}{Výpočet lineárního skóre}
    \pause
    Pro nového studenta s~hodnotou \(x=5\) nejprve vypočítáme
    lineární skóre:
    \[
        z(x)=1{,}28x-5{,}77.
    \]

    \vspace{2mm}
    \pause
    Po dosazení:
    \[
        z(5)
        =
        1{,}28\cdot 5-5{,}77
        =
        6{,}40-5{,}77
        =
        0{,}63.
    \]
    \pause
    \begin{timportantframe}
        Kladné skóre znamená, že model se přiklání ke třídě~\(1\),
        ale samo o~sobě ještě není výslednou třídou.
    \end{timportantframe}

\end{frame}

\begin{frame}{Výpočet výstupu logistické funkce}
    \pause
    Lineární skóre převedeme logistickou funkcí:
    \[
        \hat p(5)
        =
        \frac{1}{1+e^{-0{,}63}}
        \approx
        0{,}65.
    \]
    \pause
    \begin{center}
        \input{images/vypocet-vystupu-logisticke-funkce.tex}
    \end{center}

\end{frame}

\begin{frame}{Příklad}{Přiřazení výsledné třídy}
    \pause
    Pro zvolený rozhodovací práh \(t\in(0,1)\) použijeme klasifikátor
    \[
        h^{(t)}(x)
        =
        \begin{cases}
            1, & \hat p(x)\geq t,\\[1mm]
            0, & \hat p(x)<t.
        \end{cases}
    \]

    \vspace{2mm}
    \pause
    Protože
    \[
        \hat p(5)\doteq 0{,}65,
    \]
    dostáváme například:
    \[
        h^{(0{,}5)}(5)=1,
        \qquad
        h^{(0{,}7)}(5)=0.
    \]
    \pause
    \begin{timportantframe}
        Stejný logistický model může při různých rozhodovacích prazích
        vést k~různým klasifikacím.
    \end{timportantframe}

\end{frame}

\subsection{Správné a chybné klasifikace}

\begin{frame}{Skutečná a~předpovězená třída}
    \pause
    Nechť jsou dány soubory
    \[
        \mathbf{X}=(x_1,\dots,x_n),
        \quad
        \mathbf{Y}=(y_1,\ldots,y_n),
        \quad
        \hat{\mathbf{Y}}^{(t)}
        =
        (\hat y_1^{(t)},\ldots,\hat y_n^{(t)}).
    \]
    \pause
    \begin{tnoticebox}{Co porovnáváme?}
        U~každého objektu porovnáváme skutečnou třídu \(y_i\)
        s~předpovězenou třídou $\hat y_i^{(t)}=h^{(t)}(x_i)$.
    \end{tnoticebox}

    \vspace{1mm}
    \pause
    \begin{center}
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{c|c|c|c}
            objekt & \(x_i\) & skutečnost \(y_i\) & předpověď \(\hat y_i^{(t)}\) \\
            \hline
            \(1\) & \(x_1\) & \(1\) & \(1\) \\
            \(2\) & \(x_2\) & \(0\) & \(1\) \\
            \(3\) & \(x_3\) & \(0\) & \(0\) \\
            \(4\) & \(x_4\) & \(1\) & \(0\)
        \end{tabular}
    \end{center}

\end{frame}

\begin{frame}{Čtyři možné výsledky}
    \pause
    U~binární klasifikace rozlišujeme čtyři případy podle toho,
    zda je skutečná a~předpovězená třída rovna \(0\) nebo \(1\).

    \vspace{2mm}
    \pause
    \begin{center}
        \renewcommand{\arraystretch}{1.35}
        \begin{tabular}{c|c|c}
            skutečnost & předpověď & označení \\
            \hline
            \(y_i=1\) & \(\hat y_i^{(t)}=1\)
                & {\small správně pozitivní výsledek (\markdarkred{true positive}, TP)} \\
            \(y_i=0\) & \(\hat y_i^{(t)}=0\)
                & {\small správně negativní výsledek (\markdarkred{true negative}, TN)} \\
            \(y_i=0\) & \(\hat y_i^{(t)}=1\)
                & {\small falešně pozitivní výsledek (\markdarkred{false positive}, FP)} \\
            \(y_i=1\) & \(\hat y_i^{(t)}=0\)
                & {\small falešně negativní výsledek (\markdarkred{false negative}, FN)}
        \end{tabular}
    \end{center}

\end{frame}

\begin{frame}{Počty jednotlivých výsledků}
    \pause
    \begin{defblock}
        Pro statistické soubory \(\mathbf{X}=(x_1,\dots,x_n)\), \(\mathbf{Y}=(y_1,\dots,y_n)\) a~klasifikátor \(h^{(t)}\) definujeme:
        \[
            \begin{aligned}
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
                &=
                \left|
                    \set{i:y_i=1,\ h^{(t)}(x_i)=1}
                \right|,\\
                \TN_{\mathbf{X},\mathbf{Y}}^{(t)}
                &=
                \left|
                    \set{i:y_i=0,\ h^{(t)}(x_i)=0}
                \right|,\\
                \FP_{\mathbf{X},\mathbf{Y}}^{(t)}
                &=
                \left|
                    \set{i:y_i=0,\ h^{(t)}(x_i)=1}
                \right|,\\
                \FN_{\mathbf{X},\mathbf{Y}}^{(t)}
                &=
                \left|
                    \set{i:y_i=1,\ h^{(t)}(x_i)=0}
                \right|.
            \end{aligned}
        \]
    \end{defblock}
    \pause
    \begin{timportantframe}[\textwidth]
        Tyto čtyři hodnoty závisí na statistických souborech
        \(\mathbf{X},\mathbf{Y}\) i~na zvoleném rozhodovacím prahu \(t\).
    \end{timportantframe}

\end{frame}

\begin{frame}{Čtyři možné výsledky binární klasifikace}

    \begin{center}
        \input{images/ctyri-mozne-vysledky-binarni-klasifikace.tex}
    \end{center}

    \vspace{2mm}

    \begin{tinfoframe}
        Dobré výsledky odpovídají zeleným polím
        \(\TP_{\mathbf{X},\mathbf{Y}}^{(t)}\) a~\(\TN_{\mathbf{X},\mathbf{Y}}^{(t)}\),
        chybné výsledky červeným polím
        \(\FP_{\mathbf{X},\mathbf{Y}}^{(t)}\) a~\(\FN_{\mathbf{X},\mathbf{Y}}^{(t)}\).
    \end{tinfoframe}

\end{frame}

\subsection{Správnost klasifikace}

\begin{frame}{Správnost klasifikace}
    \pause
    \begin{defblock}[Správnost]
        \markdarkred{Správností} klasifikátoru \(h^{(t)}\) (anglicky \markdarkred{accuracy}) na statistických souborech
        \(\mathbf{X},\mathbf{Y}\) rozumíme číslo
        \[
            \Accuracy_{\mathbf{X},\mathbf{Y}}^{(t)}
            =
            \frac{
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \TN_{\mathbf{X},\mathbf{Y}}^{(t)}
            }{
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \TN_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \FP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \FN_{\mathbf{X},\mathbf{Y}}^{(t)}
            }.
        \]
    \end{defblock}
    \pause
    Protože jmenovatel je celkový počet objektů \(n\), můžeme psát také $\Accuracy_{\mathbf{X},\mathbf{Y}}^{(t)}=\frac{\left|\set{i:h^{(t)}(x_i)=y_i}\right|}{n}$.
    \pause
    \begin{tinfoframe}
        Správnost vyjadřuje podíl objektů, které klasifikátor zařadil správně.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Příklad}{Výpočet správnosti}
    \pause
    Předpokládejme, že pro daný práh \(t\) vyšlo
    \[
        \begin{aligned}
            \TP_{\mathbf{X},\mathbf{Y}}^{(t)}&=36,
            \qquad
            \TN_{\mathbf{X},\mathbf{Y}}^{(t)}=44,\\
            \FP_{\mathbf{X},\mathbf{Y}}^{(t)}&=8,
            \qquad
            \FN_{\mathbf{X},\mathbf{Y}}^{(t)}=12.
        \end{aligned}
    \]
    \pause
    \vspace{1mm}

    Celkový počet objektů je
    \[
        n=36+44+8+12=100.
    \]
    \pause
    Proto
    \[
        \Accuracy_{\mathbf{X},\mathbf{Y}}^{(t)}
        =
        \frac{36+44}{100}
        =
        0{,}8.
    \]
    \pause
    \begin{timportantframe}[9cm]
        \begin{center}
            Klasifikátor tedy správně zařadil \(80\,\%\) objektů.
        \end{center}
    \end{timportantframe}

\end{frame}

\begin{frame}{Kdy může být správnost zavádějící?}
    \pause
    \begin{exblock}[Nevyvážené třídy]
        Představme si test na vzácnou nemoc:
        \[
            990 \text{ osob je zdravých},
            \qquad
            10 \text{ osob je nemocných}.
        \]
        Klasifikátor vždy odpoví "zdravý".
    \end{exblock}

    \vspace{1mm}
    \pause
    Dostaneme
    \[
        \TN_{\mathbf{X},\mathbf{Y}}^{(t)}=990,
        \quad
        \FN_{\mathbf{X},\mathbf{Y}}^{(t)}=10,
        \quad
        \TP_{\mathbf{X},\mathbf{Y}}^{(t)}=0,
        \quad
        \FP_{\mathbf{X},\mathbf{Y}}^{(t)}=0.
    \]
    \pause
    \[
        \Accuracy_{\mathbf{X},\mathbf{Y}}^{(t)}
        =
        \frac{990}{1000}
        =
        0{,}99.
    \]
    \pause
    \begin{timportantframe}
        Správnost je \(99\,\%\), ale klasifikátor \textbf{nenašel ani jednoho nemocného člověka}.
    \end{timportantframe}

\end{frame}

\subsection{Přesnost}

\begin{frame}{Přesnost}
    \pause
    \begin{defblock}[Přesnost]
        \markdarkred{Přesností} klasifikátoru \(h^{(t)}\) (anglicky \markdarkred{precision}) na statistických souborech
        \(\mathbf{X}\), \(\mathbf{Y}\) rozumíme číslo
        \[
            \Precision_{\mathbf{X},\mathbf{Y}}^{(t)}
            =
            \frac{
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
            }{
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \FP_{\mathbf{X},\mathbf{Y}}^{(t)}
            },
        \]
        pokud je jmenovatel nenulový.
    \end{defblock}
    \pause
    \begin{tinfoframe}
        Přesnost odpovídá na otázku:

        \[
            \text{Kolik objektů označených jako třída~\(1\)
            skutečně patří do třídy~\(1\)?}
        \]
    \end{tinfoframe}

\end{frame}

\begin{frame}{Příklad}{Výpočet přesnosti}
    \pause
    Předpokládejme, že pro daný práh \(t\) vyšlo
    \[
        \TP_{\mathbf{X},\mathbf{Y}}^{(t)}=36,
        \qquad
        \FP_{\mathbf{X},\mathbf{Y}}^{(t)}=8.
    \]

    \vspace{2mm}
    \pause
    Klasifikátor označil jako pozitivní celkem $36+8=44$ objektů.
    \pause
    \[
        \Precision_{\mathbf{X},\mathbf{Y}}^{(t)}
        =
        \frac{36}{36+8}
        =
        \frac{36}{44}
        \doteq
        0{,}82.
    \]
    \pause
    \begin{timportantframe}
        Přibližně \(82\,\%\) objektů označených jako třída~\(1\)
        bylo skutečně ze třídy~\(1\).
    \end{timportantframe}

\end{frame}

\begin{frame}{Kdy je přesnost důležitá?}
    \pause
    \begin{exblock}[Automatické blokování zpráv]
        Představme si klasifikátor, který označuje e-maily jako spam.
        Pokud označí běžný e-mail jako spam, může uživateli skrýt
        důležitou zprávu.
    \end{exblock}

    \vspace{1mm}
    \pause
    V~takové situaci chceme, aby mezi zprávami označenými jako spam
    bylo co nejméně běžných zpráv, tedy aby bylo malé $\FP_{\mathbf{X},\mathbf{Y}}^{(t)}$.
    \pause
    \begin{timportantframe}[\textwidth]
        Vysoká přesnost znamená, že pozitivním předpovědím modelu
        můžeme poměrně dobře věřit.
    \end{timportantframe}
    \pause
    \begin{tinfoframe}[\textwidth]
        Samotná přesnost ale nestačí. Model může být velmi přesný,
        a~přitom najít jen malou část všech skutečně pozitivních objektů.
    \end{tinfoframe}

\end{frame}

\subsection{Pokrytí}

\begin{frame}{Pokrytí}
    \pause
    \begin{defblock}[Pokrytí]
        \markdarkred{Pokrytím} klasifikátoru \(h^{(t)}\) (anglicky \markdarkred{recall}) na statistických souborech
        \(\mathbf{X}\), \(\mathbf{Y}\) rozumíme číslo
        \[
            \Recall_{\mathbf{X},\mathbf{Y}}^{(t)}
            =
            \frac{
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
            }{
                \TP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \FN_{\mathbf{X},\mathbf{Y}}^{(t)}
            },
        \]
        pokud je jmenovatel nenulový.
    \end{defblock}
    \pause
    \begin{tinfoframe}
        Pokrytí odpovídá na otázku:

        \[
            \text{Kolik skutečně pozitivních objektů klasifikátor našel?}
        \]
    \end{tinfoframe}

\end{frame}

\begin{frame}{Výpočet pokrytí na příkladu}
    \pause
    Předpokládejme, že pro daný práh \(t\) vyšlo
    \[
        \TP_{\mathbf{X},\mathbf{Y}}^{(t)}=36,
        \qquad
        \FN_{\mathbf{X},\mathbf{Y}}^{(t)}=12.
    \]

    \vspace{2mm}
    \pause
    Skutečně pozitivních objektů bylo celkem $36+12=48$.
    \pause
    \[
        \Recall_{\mathbf{X},\mathbf{Y}}^{(t)}
        =
        \frac{36}{36+12}
        =
        \frac{36}{48}
        =
        0{,}75.
    \]
    \pause
    \begin{timportantframe}[11.5cm]
        \begin{center}
            Klasifikátor našel \(75\,\%\) všech skutečně pozitivních objektů.
        \end{center}
    \end{timportantframe}

\end{frame}

\begin{frame}{Přesnost versus pokrytí}
    \pause
    Přesnost a~pokrytí sledují různé druhy chyb.

    \vspace{1mm}
    \pause
    \begin{center}
        \renewcommand{\arraystretch}{1.35}
        \begin{tabular}{c|c|c}
            metrika & co trestá & otázka \\
            \hline
            přesnost &
            \(\FP_{\mathbf{X},\mathbf{Y}}^{(t)}\) &
            Kolik pozitivních předpovědí bylo správně? \\
            pokrytí &
            \(\FN_{\mathbf{X},\mathbf{Y}}^{(t)}\) &
            Kolik skutečně pozitivních objektů jsme našli?
        \end{tabular}
    \end{center}

    \vspace{1mm}
    \pause
    \begin{timportantframe}[\textwidth]
        Zvýšení rozhodovacího prahu \(t\) obvykle zvyšuje opatrnost modelu:
        pozitivních předpovědí ubývá, což \textbf{může zvýšit přesnost},
        ale \textbf{snížit pokrytí}.
    \end{timportantframe}

\end{frame}

\subsection{F-míra}

\begin{frame}{F-míra}
    \pause
    \begin{defblock}[F-míra]
        Nechť je dán klasifikátor $h^{(t)}$ a statistické soubory $\mathbf{X},\mathbf{Y}$ takové, že jsou definovány hodnoty
        \[
            \Precision_{\mathbf{X},\mathbf{Y}}^{(t)}\neq 0
            \qquad\text{a}\qquad
            \Recall_{\mathbf{X},\mathbf{Y}}^{(t)}\neq 0.
        \]
        Pak hodnotu
        \[
            \Fscore_{1,\mathbf{X},\mathbf{Y}}^{(t)}
            =
            \frac{
                2
            }{
                (\Precision_{\mathbf{X},\mathbf{Y}}^{(t)})^{-1}
                +
                (\Recall_{\mathbf{X},\mathbf{Y}}^{(t)})^{-1}
            }
        \]
        nazýváme \markdarkred{\(F_1\)-míra} (anglicky \markdarkred{$F_1$-score}).
    \end{defblock}
    \pause
    \begin{tinfoframe}[\textwidth]
        \(F_1\)-míra kombinuje přesnost a~pokrytí do jednoho čísla.
        Vysoká hodnota vyžaduje, aby byly vysoké obě metriky.
    \end{tinfoframe}

\end{frame}

\begin{frame}{F-míra pomocí \(\TP\), \(\FP\) a~\(\FN\)}
    \pause
    \begin{propblock}
        Pokud jsou jmenovatele nenulové, pak platí
        \[
            \Fscore_{1,\mathbf{X},\mathbf{Y}}^{(t)}
            =
            \frac{
                2\TP_{\mathbf{X},\mathbf{Y}}^{(t)}
            }{
                2\TP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \FP_{\mathbf{X},\mathbf{Y}}^{(t)}
                +
                \FN_{\mathbf{X},\mathbf{Y}}^{(t)}
            }.
        \]
    \end{propblock}

    \pause
    \textbf{Důkaz}: Plyne z definice hodnot $\Precision_{\mathbf{X},\mathbf{Y}}^{(t)}$ a $\Recall_{\mathbf{X},\mathbf{Y}}^{(t)}$.

    \vspace{2mm}
    \pause
    \begin{timportantframe}[\textwidth]
        \(F_1\)-míra nezávisí přímo na počtu správně negativních výsledků
        \(\TN_{\mathbf{X},\mathbf{Y}}^{(t)}\).
        Proto bývá užitečnější než správnost u~nevyvážených tříd.
    \end{timportantframe}

\end{frame}

\begin{frame}{Příklad}{Výpočet $F_1$-míry}
    \pause
    Předpokládejme, že
    \[
        \TP_{\mathbf{X},\mathbf{Y}}^{(t)}=36,
        \qquad
        \FP_{\mathbf{X},\mathbf{Y}}^{(t)}=8,
        \qquad
        \FN_{\mathbf{X},\mathbf{Y}}^{(t)}=12.
    \]
    \pause
    Potom
    \[
        \Precision_{\mathbf{X},\mathbf{Y}}^{(t)}
        =
        \frac{36}{36+8}
        \doteq 0{,}82,
        \qquad
        \Recall_{\mathbf{X},\mathbf{Y}}^{(t)}
        =
        \frac{36}{36+12}
        =
        0{,}75.
    \]
    \pause
    \[
        \Fscore_{1,\mathbf{X},\mathbf{Y}}^{(t)}
        =
        2\cdot
        \frac{0{,}82\cdot 0{,}75}{0{,}82+0{,}75}
        \doteq
        0{,}78.
    \]

    \begin{timportantframe}[\textwidth]
        Výsledná \(F_1\)-míra je přibližně \(0{,}78\), tedy kompromis
        mezi přesností a~pokrytím.
    \end{timportantframe}

\end{frame}

\subsection{Porovnání metrik}

\begin{frame}{Shrnutí metrik}

    \begin{center}

        \begin{tabular}{c|c|c}

\end{frame}

\begin{frame}{Příklad}{Společný výpočet metrik}

\end{frame}

\subsection{Vyhodnocování na neznámých datech}

\begin{frame}{Proč nestačí trénovací data?}
    \pause
    \begin{talertbox}[\textwidth]{Problém}
        Pokud model hodnotíme na stejných datech, na kterých jsme ho učili,
        můžeme získat příliš optimistický výsledek.
    \end{talertbox}

    \vspace{2mm}
    \pause
    Model se totiž může naučit nejen obecný vztah mezi \(\mathbf{X}\)
    a~\(\mathbf{Y}\), ale také náhodné zvláštnosti trénovacích dat.

    \vspace{2mm}
    \pause
    \begin{timportantframe}[\textwidth]
        Důležité není pouze to, jak model funguje na známých datech,
        ale především to, jak bude fungovat na nových objektech.
    \end{timportantframe}

\end{frame}

\begin{frame}{Rozdělení dat}
    \pause
    \begin{center}
        \input{images/rozdeleni-dat.tex}
    \end{center}

    \vspace{2mm}
    \pause
    Data často rozdělíme na tři části:
    \begin{itemize}
        \item \markdarkred{trénovací data} slouží k~učení parametrů modelu,\pause
        \item \markdarkred{validační data} slouží k~volbě nastavení modelu,\pause
        \item \markdarkred{testovací data} slouží k~nezávislému závěrečnému hodnocení.
    \end{itemize}

\end{frame}

\begin{frame}{Validační data}
    \pause
    \begin{tnoticebox}{K~čemu slouží validace?}
        Validační data používáme k~volbě parametrů, jejichž hodnota není výsledkem trénování modelu.
    \end{tnoticebox}

    \vspace{1mm}
    \pause
    Například můžeme pomocí validačních dat volit:
    \begin{itemize}
        \item rozhodovací práh \(t\),\pause
        \item hodnotu \(k\) u~metody \(k\)-NN (viz později),\pause
        \item maximální hloubku rozhodovacího stromu (viz později),\pause
        \item vhodnou hodnoticí metriku pro danou úlohu.
    \end{itemize}
    \pause
    \begin{timportantframe}
        Na validačních datech model ladíme. Proto je nepoužíváme jako
        poslední nezávislý odhad kvality modelu.
    \end{timportantframe}
\end{frame}

\begin{frame}{Testovací data}
    \pause
    \begin{tnoticebox}{Závěrečné hodnocení}
        Testovací data použijeme až ve chvíli, kdy je model i~jeho nastavení
        \textbf{pevně zvoleno}.
    \end{tnoticebox}

    \vspace{1mm}
    \pause
    Na testovacích datech spočítáme například
    \[
        \begin{aligned}
            &\Accuracy_{\mathbf{X}_{\mathrm{test}},
            \mathbf{Y}_{\mathrm{test}}}^{(t)},
            \qquad
            \Precision_{\mathbf{X}_{\mathrm{test}},
            \mathbf{Y}_{\mathrm{test}}}^{(t)},\\
            &\Recall_{\mathbf{X}_{\mathrm{test}},
            \mathbf{Y}_{\mathrm{test}}}^{(t)},
            \qquad
            \Fscore_{1,\mathbf{X}_{\mathrm{test}},
            \mathbf{Y}_{\mathrm{test}}}^{(t)}.
        \end{aligned}
    \]
    \pause
    \begin{timportantframe}[\textwidth]
        Výsledek na testovacích datech je nejlepší odhad toho,
        jak bude model fungovat na nových datech.
    \end{timportantframe}

\end{frame}

\section{Algoritmus $k$-NN}

\titleframe{Algoritmus $k$-NN}

\subsection{Základní myšlenka}

\begin{frame}{Myšlenka metody \(k\)-NN}
    \pause
    \begin{tnoticebox}[8.5cm]{Základní intuice}
        \begin{center}
            "Podobné" objekty mají často stejnou třídu.
        \end{center}
    \end{tnoticebox}
    \pause
    Metoda \textbf{\(k\) nejbližších sousedů} (\markdarkred{\(k\)-nearest neighbors}, \(k\)-NN) klasifikuje nový objekt
    podle tříd nejbližších objektů z~trénovacích dat.
    \pause
    \[
        \mathbf{X}=(x_1,\ldots,x_n),
        \qquad
        \mathbf{Y}=(y_1,\ldots,y_n).
    \]
    \pause
    \begin{tinfoframe}[\textwidth]
        Na rozdíl od logistické regrese zde nehledáme vzorec
        \(\hat p(x)\). Při klasifikaci přímo porovnáváme nový objekt
        s~trénovacími objekty.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Nejbližší soused}
    \pause
    \begin{center}
        \input{images/nejblizsi-soused.tex}
    \end{center}
    \pause
    \begin{timportantframe}[11.0cm]
        \begin{center}
            Při \(k=1\) rozhoduje pouze jeden nejbližší trénovací objekt.
        \end{center}
    \end{timportantframe}

\end{frame}

\subsection{Algoritmus $k$-NN}

\begin{frame}{Množina \(k\) nejbližších sousedů}
    \pause
    \begin{defblock}[Vzdálenost objektů (vektorů)]
        Nechť jsou dány vektory $\mathbf{x},\mathbf{y}\in\R^{p}$. Pak jejich vzdálenost budeme značit $d(\mathbf{x},\mathbf{y})$.
    \end{defblock}

    \vspace{1mm}
    \pause
    \begin{defblock}[\(k\) nejbližších sousedů]
        Nechť jsou dány vstupní statistické soubory $\mathbf{X}_1,\dots,\mathbf{X}_p$. Pro libovolný vektor \(\mathbf{x}\in\mathbb{R}^p\) seřaďme indexy
        objektů $\mathbf{x}_i=(x_{i1},\dots,x_{ip})$ tak, aby
        \[
            d(\mathbf{x},\mathbf{x}_{\pi(1)})
            \leqslant
            d(\mathbf{x},\mathbf{x}_{\pi(2)})
            \leqslant
            \cdots
            \leqslant
            d(\mathbf{x},\mathbf{x}_{\pi(n)}).
        \]
        Potom definujeme $N_k(\mathbf{x})=\set{\pi(1),\pi(2),\ldots,\pi(k)}$.
    \end{defblock}

\end{frame}

\begin{frame}{Klasifikátor \(k\)-NN}
    \pause
    \begin{itemize}
        \item Nechť jsou dány vstupní statistické soubory $\mathbf{X}_1,\mathbf{X}_2,\ldots,\mathbf{X}_p$ a~výstupní statistický soubor
        \[
            \mathbf{Y}=(y_1,y_2,\ldots,y_n),
            \qquad
            y_i\in\mathcal{Y}.
        \]\pause
        \item Klasifikátor \(k\)-NN přiřadí novému objektu
        \(\mathbf{x}\in\mathbb{R}^p\) třídu, která se mezi jeho
        \(k\) nejbližšími sousedy vyskytuje nejčastěji:
        \[
            h^{(k)}(\mathbf{x})
            =
            \argmax_{c\in\mathcal{Y}}
            \left|
                \set{i\in N_k(\mathbf{x}):y_i=c}
            \right|.
        \]\pause
        \item Pokud nastane shoda hlasů, je nutné použít předem zvolené pravidlo. U~binární klasifikace se proto často volí liché \(k\).
    \end{itemize}

\end{frame}

    \begin{center}
        \input{images/algoritmus-k-nn-pro-x-1-x-2-y.tex}
    \end{center}

\end{frame}

\subsection{Volba parametru $k$}

\begin{frame}{Vliv hodnoty \(k\)}
    \pause    
    \small
    Stejný nový objekt \(\mathbf{x}\) může být při různých hodnotách
    \(k\) zařazen do různých tříd.

    \vspace{1mm}
    \pause
    \begin{center}
        \input{images/vliv-hodnoty-k.tex}
    \end{center}
    \pause
    \begin{tinfoframe}
        Pro \(k=1\) rozhoduje jediný nejbližší soused, zatímco pro \(k=5\)
        rozhoduje většinové hlasování pěti nejbližších sousedů.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Jak zvolit hodnotu \(k\)?}
    \pause
    \begin{tnoticebox}[\textwidth]{Volba pomocí validačních dat}
        Hodnotu \(k\) obvykle nevolíme podle trénovacích dat, ale pomocí
        validační množiny.
    \end{tnoticebox}

    \vspace{1mm}
    \pause
    \begin{itemize}
        \item Zvolíme množinu kandidátů $\mathcal{K}\subseteq \set{1,2,\ldots,n}$.\pause
        \item Pro každé \(k\in\mathcal{K}\) vytvoříme klasifikátor \(h^{(k)}\) a~spočítáme zvolenou metriku (správnost, přesnost, pokrytí, F-míra) na validačních datech.\pause
        \item Pokud tuto metriku označíme \(M_{\mathrm{val}}(h^{(k)})\), pak můžeme psát
        \[
            k^\ast
            \in
            \argmax_{k\in\mathcal{K}}\;
            M_{\mathrm{val}}(h^{(k)}).
        \]
    \end{itemize}
    
    \vspace{1mm}
    \pause
    \begin{tinfoframe}[\textwidth]
        U~binární klasifikace se často volí liché hodnoty \(k\), protože
        tím snižujeme riziko nerozhodného hlasování.
    \end{tinfoframe}

\end{frame}

\subsection{Vzdálenost mezi objekty}

\begin{frame}{Eukleidovská vzdálenost}
    \pause
    \begin{defblock}[Eukleidovská vzdálenost]
        Nechť \(\mathbf{u},\mathbf{v}\in\mathbb{R}^p\), kde
        \[
            \mathbf{u}=(u_1,\ldots,u_p),
            \qquad
            \mathbf{v}=(v_1,\ldots,v_p).
        \]
        \textbf{Eukleidovskou vzdáleností} \(\mathbf{u}\) a~\(\mathbf{v}\) rozumíme číslo
        \[
            d_E(\mathbf{u},\mathbf{v})
            =
            \sqrt{
                \sum_{i=1}^{p}(u_i-v_i)^2
            }.
        \]
    \end{defblock}
    
    \begin{tinfoframe}[\textwidth]
        V~rovině jde o~běžnou geometrickou vzdálenost dvou bodů.
        Metoda \(k\)-NN pak hledá objekty, které jsou novému objektu
        podle této vzdálenosti nejblíže.
    \end{tinfoframe}

\end{frame}

\begin{frame}{Další možnosti volby vzdálenosti}
    \pause
    Nechť $\mathbf{u}=(u_1,\ldots,u_p),\mathbf{v}=(v_1,\ldots,v_p)$ jsou dva objekty z~\(\mathbb{R}^p\).
    \pause
    \begin{itemize}
        \item \textbf{Minkowského vzdálenost:} Pro \(q\geq 1\) definujeme
        \[
            d_q(\mathbf{u},\mathbf{v})
            =
            \left(
                \sum_{j=1}^{p}|u_j-v_j|^q
            \right)^{1/q}.
        \]
        \pause
        \item Pro \(q=1\) získáváme \textbf{manhattanskou vzdálenost}
        \[
            d_1(\mathbf{u},\mathbf{v})
            =
            \sum_{j=1}^{p}|u_j-v_j|.
        \]
        \pause
        \item Pro \(q=2\) získáváme eukleidovskou vzdálenost \(d_E\).
        \pause
        \item Další možností je \textbf{maximová vzdálenost}
        \[
            d_{\infty}(\mathbf{u},\mathbf{v})
            =
            \max_{1\leq j\leq p}|u_j-v_j|.
        \]
    \end{itemize}
\end{frame}

\begin{frame}{Problém různých měřítek}
    \pause
    Uvažujme dva znaky:
    \[
        \mathbf{X}_1=\text{věk v~letech},
        \qquad
        \mathbf{X}_2=\text{roční příjem v~Kč}.
    \]
    \pause
    Pro nový objekt
    \[
        \mathbf{x}=(30,500\,000)
    \]
    a~dva trénovací objekty
    \[
        \mathbf{x}_A=(45,500\,000),
        \qquad
        \mathbf{x}_B=(30,510\,000)
    \]
    dostaneme
    \[
        d_E(\mathbf{x},\mathbf{x}_A)=15,
        \qquad
        d_E(\mathbf{x},\mathbf{x}_B)=10\,000.
    \]
    \pause
    \begin{timportantframe}[\textwidth]
        Číselně větší měřítko příjmu způsobí, že rozdíl \(10\,000\)~Kč
        zcela převáží rozdíl \(15\)~let věku. To lze vyřešit např. \markdarkred{min-max normalizací}.
    \end{timportantframe}

\end{frame}

\begin{frame}{Připomenutí: min-max normalizace dat}
    \pause
    Nechť
    \[
        \mathbf{X}_j=(x_{1j},\ldots,x_{nj})
    \]
    je \(j\)-tý vstupní statistický soubor. Označme
    \[
        m_j=\min_i x_{ij},
        \qquad
        M_j=\max_i x_{ij}.
    \]
    Pokud \(M_j>m_j\), definujeme normalizovanou hodnotu
    \[
        \tilde{x}_{ij}
        =
        \frac{x_{ij}-m_j}{M_j-m_j}.
    \]
    \pause
    \begin{tinfoframe}
        Po normalizaci leží hodnoty daného znaku v~intervalu \(\langle0,1\rangle\).
        Jednotlivé znaky pak mají srovnatelný vliv na vzdálenost.
    \end{tinfoframe}

\end{frame}

\subsection{Vlastnosti $k$-NN}

\begin{frame}{Výhody metody \(k\)-NN}
    \pause
    {   
        \setbeamertemplate{itemize item}{%
            \raisebox{-0.15\height}{%
                \includegraphics[
                    width=0.6cm
                ]{images/face-smile-emoji-icon.pdf}%
            }%
        }

        \begin{itemize}
            \item jednoduchá a~názorná myšlenka,\pause
            \item téměř žádná fáze trénování,\pause
            \item použitelná pro binární i~vícetřídní klasifikaci,\pause
            \item může vytvářet nelineární rozhodovací hranice,\pause
            \item nepředpokládá konkrétní tvar vztahu mezi znaky a~třídou.
        \end{itemize}
    }
    \pause
    \begin{tnoticeframe}[\textwidth]
        Metoda je velmi intuitivní: nový objekt klasifikujeme podle toho,
        jaké třídy mají jemu podobné objekty.
    \end{tnoticeframe}

\end{frame}

\begin{frame}{Nevýhody metody \(k\)-NN}
    \pause
     {
        \setbeamertemplate{itemize item}{%
            \raisebox{-0.15\height}{%
                \includegraphics[
                    width=0.6cm
                ]{images/face-frown-emoji-icon.pdf}%
            }%
        }

        \begin{itemize}
            \item klasifikace nového objektu může být pomalá,\pause
            \item je nutné uchovávat trénovací data,\pause
            \item výsledek závisí na~volbě \(k\) a~na~zvolené vzdálenosti,\pause
            \item metoda je citlivá na~různá měřítka znaků,\pause
            \item při velkém počtu znaků se vzdálenosti hůře interpretují.
        \end{itemize}
    }
    \pause
    \begin{talertframe}[\textwidth]
        Metoda \(k\)-NN přesouvá většinu práce do okamžiku,
        kdy chceme klasifikovat nový objekt.
    \end{talertframe}

\end{frame}

\begin{frame}{Kdy je metoda \(k\)-NN vhodná?}
    \pause
    Metoda \(k\)-NN je vhodná zejména tehdy, když:
    \begin{itemize}
        \item podobnost objektů dobře odpovídá jejich třídám,\pause
        \item počet trénovacích objektů není příliš velký,\pause
        \item znaky lze rozumně normalizovat,\pause
        \item chceme jednoduchý a~snadno vysvětlitelný klasifikátor.
    \end{itemize}
    \pause
    \begin{timportantframe}
        Klíčová otázka zní: \textit{Má zvolená vzdálenost skutečně vystihovat
        podobnost objektů z~hlediska dané klasifikační úlohy?}
    \end{timportantframe}

\end{frame}
