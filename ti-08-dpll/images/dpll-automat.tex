% Průběh algoritmu DPLL jako vývojový diagram v mřížce 3 x 3 (tok „hadem“:
% první řádek zleva doprava, druhý zprava doleva, výsledky ve třetím).
% Makra \flow... jsou v assets/macros.tex.
%
% Sloupce určuje druhý řádek (rekurzivní krok a dvě podmínky), uzly prvního
% a třetího řádku se zarovnají pod ně. Mezera mezi sloupci se dopočítá
% z \linewidth a odhadu šířek sloupců, takže diagram vyplní šířku snímku
% v libovolném poměru stran a písmo zůstane stejně velké.
\begingroup
\def\dpllwA{34mm}% odhad šířky sloupce A (rekurzivní krok)
\def\dpllwB{46mm}% odhad šířky sloupce B (podmínka s prázdnou klauzulí)
\def\dpllwC{34mm}% odhad šířky sloupce C (eliminace, podmínka s prázdnou formulí)
% Mezera mezi sloupci: zbytek šířky řádku po odečtení sloupců a přesahu
% popisků vpravo, nejméně však 8 mm.
\pgfmathsetlengthmacro{\dpllgap}{max(8mm, (\linewidth - \dpllwA - \dpllwB - \dpllwC - 6mm) / 2)}%
\begin{tikzpicture}[
    font=\normalsize,
    lbl/.style={font=\small},
]

    % Výšky řádků
    \coordinate (row1) at (0, 0);
    \coordinate (row3) at (0, -51mm);

    % Druhý řádek: rekurzivní krok a podmínky
    \flowstep[minimum height=17mm]{branch}{(0, -27mm)}
        {Rekurzivní krok:\\zvol $\ell$, opakuj\\pro $\varphi\land\ell$ a $\varphi\land\neg \ell$}
    \flowdecision[right=\dpllgap of branch]{testclause}{}
        {Obsahuje $\varphi$\\prázdnou\\klauzuli?}
    \flowdecision[right=\dpllgap of testclause]{testempty}{}
        {Je $\varphi$\\prázdná?}

    % První řádek
    \flowio{input}{(row1 -| branch)}{$\varphi$ v CNF}
    \flowstep[minimum height=12mm]{unit}{(row1 -| testclause)}
        {Jednotková\\propagace}
    \flowstep[minimum height=12mm]{pure}{(row1 -| testempty)}
        {Eliminace\\čistých literálů}

    % Třetí řádek: výsledky
    \flowio{unsat}{(row3 -| testclause)}{$\varphi$ je\\nesplnitelná}
    \flowio{sat}{(row3 -| testempty)}{$\varphi$ je\\splnitelná}

    % Jednotková propagace a eliminace čistých literálů se opakují,
    % dokud je co dělat.
    \draw[flowarrow] ($(unit.north)+(-4mm,0)$)
        .. controls +(100:10mm) and +(80:10mm) ..
        node[flowlabel, lbl, above] {existuje\\jednotková klauzule}
        ($(unit.north)+(4mm,0)$);
    \draw[flowarrow] ($(pure.north)+(-4mm,0)$)
        .. controls +(100:10mm) and +(80:10mm) ..
        node[flowlabel, lbl, above] {existuje\\čistý literál}
        ($(pure.north)+(4mm,0)$);

    % Hlavní tok
    \flowarrow{input}{unit}
    \flowarrowlabel{unit}{lbl, below}{žádná\\jednotková\\klauzule}{pure}
    \flowarrowlabel{pure}{lbl, right}{žádný\\čistý literál}{testempty}

    % Test prázdné formule
    \flowbranch{testempty}{lbl, right}{ano}{sat}
    \flowbranch{testempty}{lbl, above}{ne}{testclause}

    % Test prázdné klauzule
    \flowbranch{testclause}{lbl, right}{ano}{unsat}
    \flowbranch{testclause}{lbl, above}{ne}{branch}

    % Rekurzivní volání na nové instanci
    \flowarrowlabel{branch}{lbl, right}{rekurzivně\\na nové instanci}{input}

\end{tikzpicture}%
\endgroup
