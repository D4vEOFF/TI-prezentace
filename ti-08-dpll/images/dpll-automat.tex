% Průběh algoritmu DPLL jako vývojový diagram v mřížce 3 x 3 (tok „hadem“:
% první řádek zleva doprava, druhý zprava doleva, výsledky ve třetím).
% Šířky sloupců jsou pevné, mezery mezi nimi se dopočítají z \linewidth,
% takže diagram vyplní šířku snímku v libovolném poměru stran a písmo
% zůstane stejně velké.
\begingroup
\def\dpllwA{32mm}% šířka uzlů ve sloupci A
\def\dpllwB{30mm}% šířka uzlů ve sloupci B
\def\dpllwC{32mm}% šířka uzlů ve sloupci C
% Mezera mezi sloupci: zbytek šířky řádku po odečtení uzlů a přesahu
% popisků vpravo, nejméně však 11 mm.
\pgfmathsetlengthmacro{\dpllgap}{max(11mm, (\linewidth - \dpllwA - \dpllwB - \dpllwC - 8mm) / 2)}%
\pgfmathsetlengthmacro{\dpllxB}{\dpllwA / 2 + \dpllgap + \dpllwB / 2}%
\pgfmathsetlengthmacro{\dpllxC}{\dpllxB + \dpllwB / 2 + \dpllgap + \dpllwC / 2}%
\begin{tikzpicture}[
    font=\normalsize,
    lbl/.style={font=\small, align=center, inner sep=2pt},
    nodeA/.style={task, minimum width=\dpllwA, minimum height=12mm},
    nodeB/.style={task, minimum width=\dpllwB, minimum height=12mm},
    nodeC/.style={task, minimum width=\dpllwC, minimum height=12mm},
]

    % Uzly
    \node[nodeA] (input) at (0, 0) {$\varphi$ v CNF};
    \node[nodeB] (unit) at (\dpllxB, 0) {Jednotková\\propagace};
    \node[nodeC] (pure) at (\dpllxC, 0) {Eliminace\\čistých literálů};

    \node[nodeC] (testempty) at (\dpllxC, -26mm) {Je $\varphi$\\prázdná?};
    \node[nodeB] (testclause) at (\dpllxB, -26mm) {Obsahuje $\varphi$\\prázdnou klauzuli?};
    \node[nodeA, minimum height=17mm] (branch) at (0, -26mm)
        {Rekurzivní krok:\\zvol $\ell$, opakuj\\pro $\varphi\land\ell$ a $\varphi\land\neg \ell$};

    \node[nodeC] (sat) at (\dpllxC, -48mm) {$\varphi$ je\\splnitelná};
    \node[nodeB] (unsat) at (\dpllxB, -48mm) {$\varphi$ je\\nesplnitelná};

    % Jednotková propagace a eliminace čistých literálů se opakují,
    % dokud je co dělat.
    \draw[diredge] ($(unit.north)+(-4mm,0)$)
        .. controls +(100:10mm) and +(80:10mm) ..
        node[lbl, above] {existuje\\jednotková klauzule} ($(unit.north)+(4mm,0)$);
    \draw[diredge] ($(pure.north)+(-4mm,0)$)
        .. controls +(100:10mm) and +(80:10mm) ..
        node[lbl, above] {existuje\\čistý literál} ($(pure.north)+(4mm,0)$);

    % Hlavní tok
    \draw[diredge] (input) -- (unit);
    \draw[diredge] (unit) -- (pure);
    \node[lbl, anchor=north] at ($(unit.south east)!0.5!(pure.south west)$)
        {žádná\\jednotková\\klauzule};

    \draw[diredge] (pure) -- node[lbl, right] {žádný\\čistý literál} (testempty);

    % Test prázdné formule
    \draw[diredge] (testempty) -- node[lbl, right] {ano} (sat);
    \draw[diredge] (testempty) -- node[lbl, above] {ne} (testclause);

    % Test prázdné klauzule
    \draw[diredge] (testclause) -- node[lbl, right] {ano} (unsat);
    \draw[diredge] (testclause) -- node[lbl, above] {ne} (branch);

    % Rekurzivní volání na nové instanci
    \draw[diredge] (branch) -- node[lbl, right] {rekurzivně\\na nové instanci} (input);

\end{tikzpicture}%
\endgroup
