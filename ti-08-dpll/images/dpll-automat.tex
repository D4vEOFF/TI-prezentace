    % Uzly
    \node[task, minimum width=22mm] (input) {$\varphi$ v CNF};

    \node[task, minimum width=22mm, below=of pure] (testempty) {Je $\varphi$ prázdná?};

    \node[task, minimum width=18mm, right=18mm of testempty] (sat) {$\varphi$ je splnitelná};

    \node[task, minimum width=28mm, below=of testempty] (testclause) {Obsahuje $\varphi$\\prázdnou klauzuli?};

    \node[task, minimum width=18mm, right=18mm of testclause] (unsat) {$\varphi$ je nesplnitelná};

    \node[task, minimum width=30mm, below=of testclause] (branch) {Rekurzivní krok:\\

    \draw[diredge] (unit) -- node[right, xshift=1mm, align=center] {\scriptsize žádná jednotková klauzule} (pure);

    \draw[diredge] (pure) -- node[right, xshift=1mm, align=center] {\scriptsize žádný čistý literál} (testempty);

    \draw[diredge] (testempty) -- node[right] {\scriptsize ne} (testclause);

    % Test prázdné klauzule
    \draw[diredge] (testclause) -- node[above] {\scriptsize ano} (unsat);
    \draw[diredge] (testclause) -- node[right] {\scriptsize ne} (branch);

    % Rekurzivní návrat
    \draw[diredge] (branch.west) to[out=180,in=-120]
        node[pos=0.5, left, align=right] {\scriptsize rekurzivně na nové instanci}
