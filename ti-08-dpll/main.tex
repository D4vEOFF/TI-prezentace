    \begin{frame}{Konjunktivní normální forma}
        \visible<2->{\begin{defblock}[Literál a klauzule]
            Mějme logické proměnné $x_1,x_2,\dots,x_n$. Pak definujeme následující:
            \begin{itemize}
                \item \markdarkred{Literál} je jakákoliv \textbf{proměnná $x_i$, nebo její negace $\neg x_i$}
                \item \markdarkred{Klauzule} sestavá z jednoho nebo více literálů oddělených buď spojkou $\land$, nebo $\lor$.
            \end{itemize}
        \end{defblock}}
        \visible<3->{\begin{defblock}[Konjunktivní normální forma]
            Mějme logickou formuli $\varphi$ s proměnnými $x_1,x_2,\dots,x_n$. Pak řekneme, že $\varphi$ je v \markdarkred{konjunktivní normální formě} (CNF), pokud sestává z klauzulí oddělených spojkou $\land$ a literály v každé klauzuli jsou odděleny spojkou $\lor$.
        \end{defblock}}
    \end{frame}
