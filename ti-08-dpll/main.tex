    \section{Připomenutí}

    \begin{frame}{Konjunktivní normální forma}
        \visible<2->{\begin{defblock}[Literál a klauzule]
            Mějme logické proměnné $x_1,x_2,\dots,x_n$. Pak definujeme následující:
            \begin{itemize}
                \item \markdarkred{Literál} je jakákoliv \textbf{proměnná $x_i$, nebo její negace $\neg x_i$}
                \item \markdarkred{Klauzule} sestavá z jednoho nebo více literálů oddělených buď spojkou $\land$, nebo $\lor$.
            \end{itemize}
        \end{defblock}}
        \visible<3->{\begin{defblock}[Konjunktivní normální forma]
            Mějme logickou formuli $\varphi$ s proměnnými $x_1,x_2,\dots,x_n$. Pak řekneme, že $\varphi$ je v \markdarkred{konjunktivní normální formě} (CNF), pokud sestává z klauzulí oddělených spojkou $\land$ a literály v každé klauzuli jsou odděleny spojkou $\lor$.
        \end{defblock}}
    \end{frame}

    \begin{frame}{Problémy SAT a UNSAT}
        \visible<2->{\decproblem{Splnitelnost formule v CNF ($\SAT$)}{
            Logická formule $\varphi$ v CNF.
        }{Je formule $\varphi$ splnitelná?}}
        \visible<3->{\decproblem{Nesplnitelnost formule v CNF ($\UNSAT$)}{
            Logická formule $\varphi$ v CNF.
        }{Je formule $\varphi$ nesplnitelná?}}
    \end{frame}

    \subsection{Úvod}

    \begin{frame}{Úvod}
        \begin{itemize}
            \visible<2->{\item Algoritmus řešící $\SAT$ (tzv. \textbf{SAT solver})}
            \visible<3->{\item Martin \markdarkred{D}avis, Hilary \markdarkred{P}utnam, George \markdarkred{L}ogemann, Donald W. \markdarkred{L}oveland (1961)}
            \visible<4->{\item Postup založený na rekurzivním "zkracování" logické formule v CNF.}
        \end{itemize}
    \end{frame}

    \subsection{Jednotková klauzule, polarita a čisté literály}

    \begin{frame}{Jednotková klauzule, polarita a čisté literály}
        \visible<2->{\begin{defblock}[Polarita literálu]
            Je-li literál buď proměnná $x$, nebo její negace $\neg x$, pak říkáme, že
            \begin{itemize}
                \item $x$ má \markdarkred{kladnou polaritu} a
                \item $\neg x$ má \markdarkred{zápornou polaritu}.
            \end{itemize}
        \end{defblock}}
        \visible<3->{\begin{defblock}[Jednotková klauzule]
            Klauzule se nazývá \markdarkred{jednotková}, pokud obsahuje \textbf{právě jeden literál}.
        \end{defblock}}
        \visible<4->{\begin{defblock}[Čistý literál]
            Literál $\ell$ v logické formuli $\varphi$ nazveme \markdarkred{čistý}, pokud se v ní vyskytuje \textbf{pouze v jedné polaritě}.
        \end{defblock}}
    \end{frame}

    \subsection{Princip algoritmu DPLL}

    \begin{frame}{Princip algoritmu DPLL}
        \begin{itemize}
            \visible<2->{\item Na vstupu máme logickou formuli $\varphi$ v CNF s proměnnými $x_1,\dots,x_n$.}
            \visible<3->{\item Chceme zjistit, zdali je $\varphi$ splnitelná.}
            \visible<4->{\item Algoritmus \textbf{opakovaně} aplikuje na $\varphi$ postupně dvojici procedur:}
            \begin{itemize}
                \visible<5->{\item \markdarkred{jednotková propagace} (unit propagation) a}
                \visible<6->{\item \markdarkred{eliminace čistých literálů} (pure literal elimination).}
            \end{itemize}
            \visible<7->{\item Ze vzniklé formule $\varphi^{\prime}$ zkusíme určit, zdali \textbf{je či není splnitelná}.}
            \visible<8->{\item Pokud to nepůjde, formuli "vhodně" upravíme a znovu aplikujeme stejný postup.}
        \end{itemize}
        \visible<9->{\begin{timportantframe}
            \begin{center}
                Má-li být formule $\varphi$ splnitelná, pak musíme být schopni splnit \textbf{každou její klauzuli}. 
            \end{center}
        \end{timportantframe}}
    \end{frame}

    \subsection{Jednotková propagace}

    \begin{frame}{Jednotková propagace}
        \visible<2->{Nachází-li se ve formuli jednotková klauzule, tj.
        \[\ldots\land(\ldots)\land(\ell)\land(\ldots)\land\ldots,\]
        pak nutně musí být $\ell=1$.}
        \begin{itemize}
            \visible<3->{\item Na vstupu máme formuli $\varphi$ a nějaký její literál $\ell$.}
            \visible<4->{\item Pokud je $\ell=x_i$, pak $x_i=1$. Naopak pokud $\ell=\neg x_i$, pak $x_i=0$.}
            \visible<5->{\item V takovém případě můžeme odstranit všechny ostatní klauzule obsahující $\ell$.}
            \visible<6->{\begin{tnoticeframe}
                \begin{center}
                    Klauzule ve tvaru $(\ldots\lor \ell \lor \ldots)$ jsou splněny a dále není třeba je uvažovat (můžeme je odstranit).
                \end{center}
            \end{tnoticeframe}}
            \visible<7->{\item Naopak pokud nějaká klauzule obsahuje $\ell$ \textbf{v opačné polaritě} (tj. $\neg\ell$), pak jeho hodnota je $0$.}
            \visible<8->{\begin{tnoticeframe}
                \begin{center}
                    Takový literál můžeme z formule odstranit. Hodnota klauzule je závislá pouze na zbylých literálech.
                \end{center}
            \end{tnoticeframe}}
        \end{itemize}
    \end{frame}

    \subsection{Eliminace čistých literálů}

    \begin{frame}{Eliminace čistých literálů}
        \visible<2->{Pokud se vyskytuje nějaký literál $\ell$ ve formuli $\varphi$ \textbf{pouze v jedné polaritě}, tj.
        \[\ldots(\ldots\lor\ell\lor\ldots)\land\ldots\land(\ldots\land\ell\land\ldots)\land\ldots\]
        Formule $\varphi$ tedy neobsahuje $\neg\ell$.}
        \begin{itemize}
            \visible<3->{\item Na vstupu máme formuli $\varphi$ a její literál $\ell$.}
            \visible<4->{\item V takovém případě existuje jednoznačné ohodnocení literálu $\ell$ takové, že všechny klauzule jej obsahující jsou splněny.}
            \visible<5->{\item Resp. pokud $\ell=x_i$, pak nastavíme $x_i=1$. Jinak pokud $\ell=\neg x_i$, pak nastavíme $x_i=0$.}
        \end{itemize}
        \visible<6->{\begin{tnoticeframe}
            \begin{center}
                Takové obsahující $\ell$ můžeme z $\varphi$ odstranit, protože už jsou splněny.
            \end{center}
        \end{tnoticeframe}}
    \end{frame}

    \begin{frame}{Postup algoritmu}
        \visible<2->{Na vstupu dostaneme formuli $\varphi$ v CNF.}
        \begin{enumerate}
            \visible<3->{\item Opakovaně aplikujeme \textbf{jednotkovou eliminaci} na formuli $\varphi$, dokud v ní existují jednotkové klauzule.}
            \visible<4->{\item Opakovaně aplikujeme \textbf{eliminaci čistých literálů} na formuli $\varphi$, dokud v ní existují čisté literály.}
            \visible<5->{\item Pokud je výsledná formule $\varphi^{\prime}$ prázdná (tzn. splnili jsme všechny klauzule), pak \textbf{prohlásíme $\varphi$ za splnitelnou} a skončíme.}
            \visible<6->{\item Jinak pokud výsledná formule $\varphi^{\prime}$ obsahuje prázdnou klauzuli, pak \textbf{prohlásíme $\varphi$ za nesplnitelnou} a skončíme.}
            \visible<7->{\item Vybereme náhodně literál $\ell$ ve formuli $\varphi^{\prime}$ a zavoláme rekurzivně DPLL na formule $\varphi^{\prime}\land\ell$ a $\varphi^{\prime}\land\neg\ell$.}
            \visible<8->{\item Je-li $\varphi^{\prime}\land\ell$ nebo $\varphi^{\prime}\land\neg\ell$ splnitelná, je i $\varphi$ splnitelná.}
        \end{enumerate}
    \end{frame}

    \begin{frame}{Pseudokód algoritmu}
        \begin{center}
            \begin{minipage}{0.9\textwidth}
                \begin{algorithm}[H]
                    \caption{DPLL}
                    \KwIn{Formule $\varphi$ v~CNF}
                    \While{\textup{existuje jednotková klauzule $(\ell)$ ve formuli $\varphi$}}{
                        $\varphi\gets\textsc{UnitPropagation}(\ell,~\varphi)$
                    }
                    \While{\textup{existuje literál $\ell$ v~jediné polaritě ve $\varphi$}}{
                        $\varphi\gets\textsc{PureLiteralElimination}(\ell,\varphi)$
                    }
                    \tcp{Všechny klauzule jsou pravdivé}
                    \lIf{\textup{je $\varphi$ prázdná}}{\Return{$1$}}

                    \tcp{Všechny literály v~klauzuli jsou nepravdivé}
                    \lIf{\textup{obsahuje $\varphi$ prázdnou klauzuli}}{\Return{$0$}}
                    \tcp{Zkusíme ohodnocení pro $\ell=0$ a~$\ell=1$}
                    Vyber náhodně literál $\ell$\\

                    \Return{$\textup{DPLL}(\varphi\land\ell)\lor\textup{DPLL}(\varphi\land\neg\ell)$}
                \end{algorithm}
            \end{minipage}
        \end{center}
    \end{frame}

    \subsection{Časová složitost}

    \begin{frame}{Časová složitost DPLL}
        \begin{itemize}
            \visible<2->{\item Nejhorší případ nastane, pokud při žádném volání $\operatorname{DPLL}$ neexituje v příslušné formuli \textbf{žádná jednotková klauzule} vyjma té, kterou jsme vytvořili při sestavení formule $\varphi\land\ell$, resp. $\varphi\land\neg\ell$, ani \textbf{čistý literál}.}
            \visible<3->{\item Máme-li formuli ve tvaru $\varphi\land\ell$, resp. $\varphi\land\neg\ell$, pak literál $\ell$ eliminujeme v dalším volání a všechny ostatní zůstanou zachované.}
            \visible<4->{\begin{timportantframe}
                \begin{center}
                    Formule se tedy zkrátí pouze o \textbf{jeden literál}.
                \end{center}
            \end{timportantframe}}
            \visible<5->{\item Označme si $n$ počet literálů vstupní formule $\varphi$.}
            \visible<6->{\item Dále si označme:}
            \begin{itemize}
                \visible<7->{\item $T(n)$ = počet kroků algoritmu nad formulí o $n$ literálech,}
                \visible<8->{\item $r(n)$ = režie jednoho volání.}
            \end{itemize}
        \end{itemize}
    \end{frame}

    \begin{frame}{Časová složitost DPLL}
        \begin{itemize}
            \visible<2->{\item V jednom volání DPLL strávíme $r(n)$ kroků eliminací \textbf{jednotkových klauzulí} a \textbf{čistých literálů}.}
            \visible<3->{\item Na konci voláme dvakrát DPLL na vstup velikosti $n-1$, čímž strávíme $2\cdot T(n-1)$ kroků.}
            \visible<4->{\item Můžeme předpokládat, že režíí strávíme polynomiální čas, tzn. $r(n)=\bigO(n^{k})$, kde $k\in\R^{+}$.}
            \visible<5->{\item Tím dostáváme následující rovnici:
            \[T(n)=2\cdot T(n-1)+r(n)=2\cdot T(n-1)+\bigO(n^{k}).\]}
            \visible<6->{\begin{tinfoframe}
                \begin{center}
                    Tzv. \textbf{rekurentní rovnice}, kdy hodnota zkoumané funkce zavisí na "sobě samotné".
                \end{center}
            \end{tinfoframe}}
            \visible<7->{\item Lze ukázat, že $T(n)=\bigO(2^{n})$. Algoritmus DPLL je tedy v nejhorším případě exponenciální.}
        \end{itemize}
    \end{frame}

    \begin{frame}{Řešitelnost problému $\SAT$}
        \begin{itemize}
            \visible<2->{\item Problém $\SAT$ dodnes neumíme řešit v polynomiálním čase.}
            \visible<3->{\item Naštěstí u "velké" části logických formulí však umíme rozhodnout rychle, zdali jsou splnitelné.}
        \end{itemize}
        \visible<4->{\begin{talertframe}
            \begin{center}
                Dodnes však není známý algoritmus pro $\SAT$, který by běžel nad libovolnou formulí \textbf{v polynomiálním čase}.
            \end{center}
        \end{talertframe}}
    \end{frame}
