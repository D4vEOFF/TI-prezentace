% Shrnutí převodů mezi problémy (styly problem a reduction z assets/macros.tex).
\begin{tikzpicture}[font=\small]

    % uzly
    \node[problem] (sat)      at (0,3.2) {\SAT};
    \node[problem] (threesat) at (4.6,3.2) {\THREESAT};
    \node[problem] (clique)   at (0,0) {\CLIQUE};
    \node[problem] (indset)   at (4.6,0) {\INDSET};

    % SAT <-> 3-SAT
    \draw[reduction] (sat) to[bend left=18] node[above] {$\bigO(n)$} (threesat);
    \draw[reduction] (threesat) to[bend left=18] node[below] {$\bigO(1)$} (sat);

    % 3-SAT -> NzMna
    \draw[reduction] (threesat) -- node[right] {$\bigO(n^2)$} (indset);

    % Klika <-> NzMna
    \draw[reduction] (clique) to[bend left=18] node[above] {$\bigO(n^2)$} (indset);
    \draw[reduction] (indset) to[bend left=18] node[below] {$\bigO(n^2)$} (clique);

    % NzMna -> SAT
    \draw[reduction] (indset) -- node[pos=0.72, below left] {$\bigO(n^3)$} (sat);

\end{tikzpicture}
