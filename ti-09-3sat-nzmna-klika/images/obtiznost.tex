% Složení převodu f a algoritmu pro B jako vývojový diagram (makra \flow...)
% nad převodem problémů A -> B (styly problem a reduction); vše
% z assets/macros.tex.
\begin{tikzpicture}[font=\small]
    % Konkretni vypocet
    \flowio{x}{(0,-0.8)}{vstup $x$}
    \flowstep{red}{(2.6,-0.8)}{převod $f$}
    \flowio{fx}{(5.9,-0.8)}{instance pro $B$}
    \flowstep{alg}{(9.3,-0.8)}{algoritmus pro $B$}

    \flowarrow{x}{red}
    \flowarrowlabel{red}{above}{$f(x)$}{fx}
    \flowarrow{fx}{alg}

    % Abstraktni rovina
    \node[problem] (A) at (0,1.0) {$A$};
    \node[problem] (B) at (5.9,1.0) {$B$};

    \draw[reduction] (A) to[bend left=18] node[above] {$f$} (B);

    % Propojeni abstraktni a konkretni roviny
    \draw[thick, densely dashed] (A) -- (x);
    \draw[thick, densely dashed] (B) -- (fx);

    % Popisky slozitosti (na společné lince pod krokem)
    \coordinate (cplx) at (0,-1.3);
    \node[anchor=north] at (cplx -| x) {$|x|=a$};
    \node[anchor=north] at (cplx -| red) {čas $\bigO(a^\ell)$};
    \node[anchor=north] at (cplx -| fx) {$|f(x)|=b=\bigO(a^\ell)$};
    \node[anchor=north] at (cplx -| alg) {čas $\bigO(b^k)$};

    % Zaverecny odhad
    \node[align=center] at (4.65,-2.75)
        {$\bigO(b^k)=\bigO((a^\ell)^k)=\bigO(a^{k\ell})$\\[1mm]
         Celkem: $\bigO(a^\ell+a^{k\ell})$};
\end{tikzpicture}
