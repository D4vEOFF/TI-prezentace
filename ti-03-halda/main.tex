    \section{Binární stromy}

    \titleframe{Binární stromy}

    \begin{frame}{Binární strom}
        \visible<2->{\begin{defblock}[Binární strom]
            Strom $T$ nazveme \markdarkred{binární}, pokud je \textbf{zakořeněný} a každý vrchol (rodič) má maximálně \textbf{dva syny}, u nichž rozlišujeme, který je levý a který pravý.
        \end{defblock}}
        \visible<3->{\begin{figure}
            \centering
            \input{images/binarni-strom.tex}
        \end{figure}}
    \end{frame}

    \begin{frame}{Hladiny v binárním stromě}
        \begin{defblock}[Hladina]
            Nechť $T=(V,E)$ je binární strom, $v_0$ jeho kořen a číslo $k\in\N_0$. Pak $k$-tou hladinou stromu $T$ rozumíme množinu vrcholů
            \[L_k=\set{v\in V: d(v,v_0)=k}.\]
            Tedy hladina $L_k$ je množina všech vrcholů, které mají stejnou vzdálenost od kořene.
        \end{defblock}
    \end{frame}

    \subsection{Minimová binární halda}

    \begin{frame}{Příklad na začátek \dots}
        \begin{itemize}
            \visible<2->{\item Máme zadaný soubor čísel, v němž chceme rychle najít \textbf{minimum}, resp. \textbf{maximum}.}
            \visible<3->{\item Prvky lze uložit do seznamu/pole.}
            \visible<4->{\begin{center}
                \begin{minipage}{0.8\textwidth}
                    \begin{algorithm}[H]
                        \caption{Minimum}
                        \KwIn{Seznam prvků $x_1,x_2,\dots,x_n$}
                        $\textup{min}\gets x_1$\;
                        \For{$i=1,2,\dots,n$}{
                            \lIf{$\textup{min}<x_i$}{$\textup{min}\gets x_i$}
                        }
                        \Return{$\textup{min}$}
                    \end{algorithm}
                \end{minipage}
            \end{center}}
            \visible<5->{\item Časová složitost bude zjevně lineární, tj. $\bigO(n)$.}
            \visible<6->{\item Rychlejšího algoritmu lze dosáhnout pomocí tzv. \markdarkred{binární haldy}.}
        \end{itemize}
    \end{frame}

    \begin{frame}{Minimová binární halda}
        \begin{itemize}
            \visible<2->{\item Budeme pracovat s binárním stromem $T=(V,E)$.}
            \visible<3->{\item Každému vrcholu $v$ přiřadíme tzv. \markdarkred{klíč}, který bude v našem případě představovat libovolné celé číslo. Tuto hodnotu budeme značit $\key(v)$.}
            \visible<4->{\begin{tinfoframe}[0.8\textwidth]
                \begin{center}
                    Formálně se jedná o zobrazení $\mapping{\key}{V}{\Z}$.
                \end{center}
            \end{tinfoframe}}
        \end{itemize}
        \visible<5->{\begin{defblock}[Minimová binární halda]
            \markdarkred{Minimová binární halda} je datová struktura tvaru binárního stromu, v jehož každém vrcholu je uložen jeden prvek. Přitom platí:
            \begin{enumerate}
                \item \textbf{Tvar haldy:} Všechny hladiny kromě poslední jsou plně obsazené. Poslední hladina je zaplněna zleva.
                \item \textbf{Haldové uspořádání:} Je-li $v$ vrchol a $s$ jeho syn, pak platí $\key(v)\leqslant \key(s)$.
            \end{enumerate}
        \end{defblock}}
    \end{frame}

        \end{enumerate}

    \subsection{Tvrzení a počtu hladin}

    \begin{frame}{Počet hladin haldy}
        \visible<2->{\begin{propblock}
            Binární halda s $n$ prvky má $\lceil\log_2(n+1)\rceil$ hladin.
        \end{propblock}}
        \begin{itemize}
            \visible<3->{\item Předpokládejme, že uvažovaná halda má všechny hladiny plně obsazené. Jejich počet si označme $h$ a počet vrcholů $n$.}
            \visible<4->{\item Pak počet vrcholů lze spočítat jako
            \[n=2^{0}+2^{1}+2^{2}+\dots+2^{h-1}=2^{h}-1.\]}
            \visible<5->{\item Úpravou získáme
            \[n=2^{h}-1\iff n+1=2^{h}\iff h=\log_2(n+1).\]}
            \visible<6->{\item Pokud poslední hladina není zcela zaplněná, bude pak nebude $h$ celé číslo. To lze "opravit" přidáním zaokrouhlení nahoru:
            \[h=\lceil\log_2(n+1)\rceil.\]}
        \end{itemize}
    \end{frame}

    \subsection{Operace Insert}

    \begin{frame}{Operace $\textsc{Insert}$ -- vkládání vrcholu}
        \begin{itemize}
            \visible<2->{\item Prázdnou nebo jednoprvkovou haldu lze vytvořit snadno.}
            \visible<3->{\item Uvažujme, že do obecné (minimové) haldy chceme přidat nový prvek.}
            \visible<4->{\item Z podmínky na \textbf{tvar haldy} plyne, že nový vrchol buď}
            \begin{itemize}
                \visible<5->{\item přidáme na konec poslední hladiny, nebo}
                \visible<6->{\item založíme novou hladinu a vrchol přidáme úplně vlevo, pokud je poslední hladina plná.}
            \end{itemize}
        \end{itemize}
        \visible<7->{\begin{talertframe}[0.8\textwidth]
            \begin{center}
                \textbf{Problém:} Tímto krokem může dojít k porušení haldového uspořádání, neboť pro nový vrchol $v$ a jeho rodiče $r$ může nastat, že $\key(v)<\key(r)$.
            \end{center}
        \end{talertframe}}
    \end{frame}

    \subsection{Procedura BubbleUp}

    \begin{frame}{Procedura $\textsc{BubbleUp}$}
        \begin{itemize}
            \visible<2->{\item Po vložení nového vrcholu musíme haldu upravit tak, aby platilo haldové uspořádání, není-li splněno.}
            \visible<3->{\item Jako první prohodíme nový vrchol $v$ s jeho rodičem $r$, pokud $\key(v)<\key(r)$.}
            \visible<4->{\item Prohozením vrcholů jsme opravili uvedenou nerovnost, ale mohli jsme způsobit chybu o hladinu výš.}
            \visible<5->{\item V takovém případě postup opakujeme pro vrchol $v$ a jeho rodičem $r^{\prime}$ atd.}
            \visible<6->{\item Takový procedura se někdy nazývá $\textsc{BubbleUp}$, neboť vrchol postupně "probublá" nahoru (potenciálně až do kořene, kde se procedura zastaví).}
        \end{itemize}
    \end{frame}

                    \caption{Operace $\textsc{Insert}$}
                    \KwIn{Nový vrchol $v$}
                    $n\gets n+1$\;
                    \If{\textup{\textup{poslední hladina je plná}}}{
                        \textup{vlož $v$ do nové hladiny co nejvíce vlevo}
                    }
                    \Else{\textup{vlož $v$ do poslední hladiny zleva}}
                    \textup{Zavolej $\textsc{BubbleUp}(v)$.}
                \end{algorithm}
            \end{minipage}
        \end{center}}
    \end{frame}

    \subsection{Operace ExtractMin}

    \begin{frame}{Operace $\textsc{ExtractMin}$ -- odstranění minima}
        \begin{itemize}
            \visible<2->{\item Již víme, že přečtení minima zvládneme v konstantním čase (stačí se podívat na klíč v kořeni haldy a vrátit jej).}
            \visible<3->{\item Pokud budeme ale chtít odstranit minimum, bude situace již o trochu složitější.}
        \end{itemize}
        \visible<4->{\begin{tnoticeframe}
            Musíme vyřešit dvě otázky:
            \begin{itemize}
                \visible<5->{\item Co bude náš nový kořen?}
                \visible<6->{\item Jak opravit haldu, aby platilo haldové uspořádání?}
            \end{itemize}
        \end{tnoticeframe}}
    \end{frame}

    \begin{frame}{Operace $\textsc{ExtractMin}$}
        \begin{itemize}
            \item Abychom zachovali tvar haldy, přemístíme \textbf{nejpravější vrchol z poslední hladiny} a umístíme jej na pozici kořene.
            \item Tím však může dojít k porušení haldového uspořádání.
            \item Idea opravy bude podobná jako v případě operace $\textsc{Insert}$, ale nový kořen bude tentokrát "probublávat" směrem dolů.
            \begin{itemize}
                \item Pokud klíč v novém kořeni $v$ je větší než než klíč jedom z jeho synů, prohodíme je.
                \item Pokud je hodnota klíče v novém kořeni větším než klíče v obou jeho synech, prohodíme jej s tím, jehož klíč je menší.
                \item Pokud stále není splněno haldové uspořádání, tak postup opakujeme.
            \end{itemize}
        \end{itemize}
    \end{frame}

    \subsection{Rozbor časové složitosti}

    \begin{frame}{Rozbor časové složitosti}
        \begin{itemize}
            \visible<2->{\item Rozebereme časovou složitost operací $\textsc{Insert}$ a $\textsc{ExtractMin}$.}
            \visible<3->{\item Operace $\textsc{Min}$ má zjevně složitost konstantní.}
            \visible<4->{\item Největší roli budou hrát procedury $\textsc{BubbleUp}$ a $\textsc{BubbleDown}$. Zbylé operace stihneme v konstantním čase.}
            \visible<5->{\begin{timportantframe}[0.8\textwidth]
                \begin{center}
                    Každý vrchol "probublá" v nejhorším případě \textbf{všemi hladinami}.
                \end{center}
            \end{timportantframe}}
            \visible<6->{\item Podle dokázaného tvrzení víme, že binární halda má celkově $\lceil\log_2(n+1)\rceil$ hladin.}
            \visible<7->{\item Pak celková časová složitost činí
            \begin{align*}
                \bigO(\log_2(n+1))&=\bigO(\log_2(n+n))=\bigO(\log_2{2}+\log_2{n})=\bigO(\log_2{n}) \\
                &=\bigO\left(\frac{1}{\log{2}}\cdot\log{n}\right)=\bigO(\log{n})
            \end{align*}
            }
        \end{itemize}
    \end{frame}

    \section{Další operace na haldě}

    \titleframe{Další operace}

    \begin{frame}{Změna klíče ve vrcholu}
        \begin{itemize}
            \visible<2->{\item Ještě se nabízí otázka, jak zařídit udržování haldy v případě, že budeme chtít upravovat klíče ve vrcholech.}
            \visible<3->{\item K tomu se nám opět budou operace $\textsc{BubbleUp}$ a $\textsc{BubbleDown}$.}
            \visible<4->{\item Opět uvažujme, že pracujeme s minimovou binární haldou (pro maximovou bude postup analogický).}
            \visible<5->{\item Budeme chtít přídat následující operace:}
            \begin{itemize}
                \visible<6->{\item $\textsc{Increase}(v,x)$ = zvýší hodnotu klíče $\key(v)$ o $x>0$.}
                \visible<7->{\item $\textsc{Decrease}(v,x)$ = sníží hodnotu klíče $\key(v)$ o $x>0$.}
            \end{itemize}
        \end{itemize}
    \end{frame}

    \subsection{Operace Increase a Decrease}

    \begin{frame}{Operace $\textsc{Increase}$ a $\textsc{Decrease}$}
        \begin{itemize}
            \visible<2->{\item Když budeme chtít zvýšit hodnotu klíče ve vrcholu $v$ o hodnotu $x$, opět se musíme zamyslet, jak se může strom "pokazit".}
            \visible<3->{\item Pokud zvýšíme hodnotu klíče vrcholu $v$, pak v rodiči $r$ hodnota klíče stále splňuje požadovanou nerovnost:
            \[\key(r)<\key(v)<\key(v)+x.\]}
            \visible<4->{\item Ovšem mohlo se pokazit haldové uspořádání v synech upraveného vrcholu $v$.}
            \visible<5->{\begin{talertframe}
                \begin{center}
                    To však již umíme vyřešit procedurou $\textsc{BubbleDown}$! Stačí nám tedy zavolat $\textsc{BubbleDown}(v)$ a jsme hotovi.
                \end{center}
            \end{talertframe}}
            \visible<6->{\item Podobně budeme postupovat v případě operace $\textsc{Decrease}\implies$ zavoláme $\textsc{BubbleUp}(v)$.}
        \end{itemize}
    \end{frame}

    \begin{frame}{Složitost operací $\textsc{Increase}$ a $\textsc{Decrease}$}
        \visible<2->{Je zjevné, že nejdéle potrvájí procedury $\textsc{BubbleUp}$ a $\textsc{BubbleDown}$, o nicž víme, že trvají $\bigO(\log{n})$.}
        \visible<3->{\begin{timportantframe}[0.8\textwidth]
            \begin{center}
                Tzn. i tyto operace mají \textbf{logaritmickou časovou složitost}.
            \end{center}
        \end{timportantframe}}
    \end{frame}
