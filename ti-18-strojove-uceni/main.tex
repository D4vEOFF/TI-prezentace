\titleframe{Strojové učení}

\subsection{Vymezení pojmu "strojové učení"}

\begin{frame}{Co se rozumí strojovým učením?}

    \small

    \pause

    \begin{itemize}
        \item Ve statistice jsme pracovali se statistickými soubory zachycující jeden, nebo více statistických znaků:
    \end{itemize}

    \[
        \mathbf{X}
        =
        \begin{array}{c@{\;}c}
            \begin{array}{cccc}
                \mathbf{X}_1
                &
                \mathbf{X}_2
                &
                \cdots
                &
                \mathbf{X}_p
            \end{array}
            &
            \\[0.2cm]
            \left(
            \begin{array}{cccc}
                x_{11} & x_{12} & \cdots & x_{1p}\\
                x_{21} & x_{22} & \cdots & x_{2p}\\
                \vdots & \vdots & \ddots & \vdots\\
                x_{n1} & x_{n2} & \cdots & x_{np}
            \end{array}
            \right)
            &
            \begin{array}{c}
                \mathbf{x}_1\\
                \mathbf{x}_2\\
                \vdots\\
                \mathbf{x}_n
            \end{array}
        \end{array}.
    \]

    \pause

    \begin{itemize}
        \item Typicky jsme však pracovali se soubory popisující \markdarkred{jeden statistický znak}, tj. $\mathbf{X}=(x_1,\dots,x_n)$.
    \end{itemize}

    \pause

    \begin{center}
        \input{images/co-se-rozumi-strojovym-ucenim.tex}
    \end{center}

    \pause

    \begin{center}
        \textbf{Strojové učení:} Část chování systému se nastavuje
        podle dat. 
    \end{center}

\end{frame}

\begin{frame}{Co se rozumí strojovým učením?}

    \begin{tinfoframe}[\textwidth]
        Obecně strojovým učením rozumíme skupinu statistických algoritmů učící se \markdarkred{ze~známých dat} a~zobecňující výsledky \markdarkred{na data neznámá}.
    \end{tinfoframe}

\end{frame}

\subsection{Trénovací a testovací data}

\begin{frame}{Trénovací a testovací data}

    \small

    \pause

    \begin{itemize}
        \item Mějme statistické soubory
        \[
            \mathbf{X}_1,\mathbf{X}_2,\dots,\mathbf{X}_k,
        \]
        kde hodnota $x_{ij}$ popisuje $j$-tý znak u $i$-tého objektu.
    \end{itemize}

    \pause

    \[
        \mathbf{x}_i=(x_{i1},x_{i2},\dots,x_{ik})
    \]

    \pause

    \begin{center}
        \input{images/trenovaci-a-testovaci-data.tex}
    \end{center}

    \pause

    \begin{itemize}
        \item \textbf{Trénovací data} používáme k nastavení modelu.
        \item \textbf{Testovací data} model při trénování nevidí;
        používáme je až k ověření, zda model funguje i na nových datech.
    \end{itemize}

    \pause

    \begin{center}
        \textbf{Dobrý model se nemá jen naučit trénovací data nazpaměť.}
    \end{center}

\end{frame}

\subsection{Učení a trénování modelu}

\begin{frame}{Učení a trénování modelu}

    \pause

    \begin{itemize}
        \item \textbf{Učení} je obecný proces, při němž model získává
        z dat nebo zkušenosti informaci a přizpůsobuje jí své chování.
    \end{itemize}

    \pause

    \begin{itemize}
        \item \textbf{Trénování} je konkrétní fáze učení, během níž
        algoritmus upravuje model podle \textbf{trénovacích dat}
        a podle toho, jaké chyby model dělá.
    \end{itemize}

    \pause

    \vspace{0.2cm}

    \begin{center}
        \input{images/uceni-a-trenovani-modelu.tex}
    \end{center}

    \pause

    \vspace{0.15cm}

    \begin{itemize}
        \item Data $\mathbf{X}_1,\dots,\mathbf{X}_k$ zde chápeme obecně:
        mohou to být různé statistické znaky, pozorování nebo jiné
        informace, z~nichž se model učí.
    \end{itemize}

\end{frame}

\begin{frame}{Učení a trénování modelu}

    \pause
    \begin{itemize}
        \item Při trénování model vytváří výstupy, ty se vyhodnocují
        a~podle vzniklé chyby se upravují parametry
        $\boldsymbol{\theta}=(\theta_1,\dots,\theta_m)$.\pause
        \item Pomocí výsledného $f_{\boldsymbol{\theta}}$ modelu spočítá
        \[\hat{y}_{ij} = f_{\boldsymbol{\theta}}(x_{ij}),\]
        přičemž $\mathbf{X}_i=(x_{i1},\dots,x_{in})$ a $\hat{\mathbf{Y}}_i=(\hat{y}_{i1},\dots,\hat{y}_{in})$.
    \end{itemize}

    \pause

    \begin{tinfoframe}[11.5cm]
        \begin{center}
            \textbf{Po skončení trénování} používáme model pro nová data.
        \end{center}
    \end{tinfoframe}

    \pause

    \begin{tnoticeframe}[11.5cm]
        Pro jednoduchost se zaměříme na modely, které mají navstupu pouze statistických souborů (trénovacích dat) $\mathbf{X}$, popř. dvojici souborů $\mathbf{X},\mathbf{Y}$.
    \end{tnoticeframe}

\end{frame}

\begin{frame}{Základní členění strojového učení}

    \pause

    \begin{center}
        \input{images/zakladni-cleneni-strojoveho-uceni-1.tex}
    \end{center}

    \vspace{0.1cm}

    \pause

    \begin{itemize}
        \item \textbf{Učení s učitelem:}
        Máme spárované statistické soubory
        \[
            (x_1,y_1),\ldots,(x_n,y_n),
        \]
        tedy vstupy $\mathbf{X}$ a známé výstupy $\mathbf{Y}$.
        Cílem je odhadovat $y$ pro nové vstupy $x$.\pause
        \vspace{0.1cm}
        \item \textbf{Učení bez učitele:}
        Máme pouze statistický soubor
        \[
            x_1,\ldots,x_n,
        \]
        tedy data $\mathbf{X}$ bez známého výstupu $\mathbf{Y}$.
        Cílem je hledat strukturu, podobnosti nebo skupiny v~datech.
    \end{itemize}

\end{frame}

\begin{frame}{Základní členění strojového učení}

    \pause

    \begin{center}
        \input{images/zakladni-cleneni-strojoveho-uceni-2.tex}

\section{Učení s učitelem}

\titleframe{Učení s učitelem}

\subsection{Vymezení}

\begin{frame}{Učení s učitelem}

    \pause

    \begin{itemize}
        \item Máme dva statistické soubory
        \[
            \mathbf{X}=(x_1,x_2,\dots,x_n),
            \qquad
            \mathbf{Y}=(y_1,y_2,\dots,y_n).
        \]\pause
        \item Každé vstupní hodnotě $x_j$ odpovídá známý výstup $y_j$:
        \[
            (x_1,y_1),(x_2,y_2),\dots,(x_n,y_n).
        \]\pause
        \item Při trénování používáme dvojice $(x_j,y_j)$.
    \end{itemize}

    \pause

    \begin{tnoticeframe}[\textwidth]
        \begin{center}
            \textbf{"Učitel"} znamená, že při trénování známe
            požadovaný výstup pro každý trénovací vstup.
        \end{center}
    \end{tnoticeframe}

\end{frame}

\begin{frame}{Jak probíhá učení s učitelem?}
    
    \pause

    \begin{center}
        \input{images/jak-probiha-uceni-s-ucitelem.tex}
    \end{center}

    \pause

    \begin{itemize}
        \item Model se z trénovacích dat učí vztah mezi
        vstupem $\mathbf{X}$ a~výstupem~$\mathbf{Y}$.\pause
        \item Po skončení trénování odhaduje výstup také
        pro nové vstupní hodnoty.
    \end{itemize}

\end{frame}

\subsection{Regrese a klasifikace}

\begin{frame}{Regrese a klasifikace}

    \small

    \begin{center}
        \input{images/regrese-a-klasifikace.tex}
    \end{center}

    \pause

    \begin{itemize}
        \item \textbf{Regrese:} chceme odhadnout číselnou hodnotu.
        \[
            \text{počet odpracovaných hodin}
            \longmapsto
            \text{výše mzdy}
        \]\pause
        \item \textbf{Klasifikace:} chceme určit kategorii.
        \[
            \text{počet bodů}
            \longmapsto
            \text{prospěl / neprospěl}
        \]
    \end{itemize}

\end{frame}

\subsection{Regrese}

\begin{frame}{Regrese}

    \pause

    \begin{itemize}
        \item Při regresi je výstupní statistický soubor $\mathbf{Y}$ číselný, tj. $y_j\in\mathbb{R}$.\pause
        \item Model se snaží popsat závislost výstupu $\mathbf{Y}$ na~vstupu $\mathbf{X}$.\pause
        \[
            \hat y=\hat f(x).
        \]\pause
        \item Výsledkem je číselný odhad pro novou vstupní hodnotu $x$.
    \end{itemize}

    \pause

    \begin{center}
        \textbf{Regrese nemusí být lineární.}\\
        Lineární regrese je pouze jedním z možných regresních modelů.
    \end{center}

\end{frame}

\subsection{Lineární regrese}

\begin{frame}{Jednoduchá lineární regrese}

    \small

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.48\textwidth}

            \pause

            \begin{itemize}
                \item Máme vstupní statistický soubor $\mathbf{X}$ a číselný výstupní soubor $\mathbf{Y}$,
                \[
                    y_j=\beta_0+\beta_1x_j+\varepsilon_j.
                \]\pause
                \item Z trénovacích dvojic $(x_j,y_j)$
                odhadneme koeficienty $\beta_0$ a $\beta_1$.
                \[
                    \hat y=\hat\beta_0+\hat\beta_1x.
                \]
            \end{itemize}       

        \end{column}

        \begin{column}{0.48\textwidth}
            \centering

            \input{images/jednoducha-linearni-regrese.tex}

        \end{column}
    \end{columns}

\end{frame}

\subsection{Klasifikace}

\begin{frame}{Klasifikace}

    \small

    \pause

    \begin{itemize}
        \item Při klasifikaci výstupní statistický soubor
        $\mathbf{Y}$ obsahuje místo číselných hodnot označení tříd:
        \[
            y_j\in\set{A,B,C,\dots}.
        \]\pause
        \item Model pro nový vstup $x$ určí odhadovanou třídu:
        \[
            \hat y=\hat f(x).
        \]\pause
        \item \textbf{Binární klasifikace:} máme dvě možné třídy.
        \pause
        \item \textbf{Vícetřídní klasifikace:} máme více možných tříd.
    \end{itemize}

\end{frame}

\begin{frame}{Rozhodovací hranice}

    \pause

    \begin{itemize}
        \item Uvažujme objekty popsané jediným vstupním znakem $\mathbf{X}$.\pause
        \item Klasifikační model rozdělí možné hodnoty $\mathbf{X}$
        na oblasti odpovídající různým třídám.
    \end{itemize}

    \pause

    \begin{center}
        \input{images/rozhodovaci-hranice-1.tex}
    \end{center}

    \pause

    \begin{center}
        Například:
        \[
            \hat y=
            \begin{cases}
                A, & x<c,\\
                B, & x\geq c.
            \end{cases}
        \]
    \end{center}

\end{frame}

\begin{frame}{Rozhodovací hranice}

    \small

    \pause

    \begin{itemize}
        \item Mějme dva vstupní statistické soubory
        \[
            \mathbf{X}_1=(x_{11},\dots,x_{1n}),
            \qquad
            \mathbf{X}_2=(x_{21},\dots,x_{2n})
        \]
        a~výstupní soubor
        \[
            \mathbf{Y}=(y_1,\dots,y_n),
            \qquad
            y_j\in\{A,B\}.
        \]\pause
        \item $j$-tý objekt je popsán bodem $(x_{1j},x_{2j})$ a~přísluší mu třída $y_j$.
    \end{itemize}

    \pause

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.60\textwidth}
            \centering

\subsection{Vyhodnocení modelu}

\begin{frame}{Vyhodnocení modelu u učení s učitelem}

    \pause

    \begin{itemize}
        \item Pro každý index $j$ model vytvoří odhad
        \[
            \hat y_j=\hat f(x_j).
        \]\pause
        \item Odhad $\hat y_j$ porovnáme se známou skutečnou hodnotou $y_j$.\pause
        \item U regrese lze sledovat chybu
        \[
            e_j=y_j-\hat y_j.
        \]\pause
        \item U klasifikace sledujeme, zda model určil správnou třídu:
        \[
            \hat y_j=y_j
            \qquad\text{nebo}\qquad
            \hat y_j\neq y_j.
        \]
    \end{itemize}

    \pause

    \begin{timportantframe}[\textwidth]
        \begin{center}
            Testovací data ukazují, jak model funguje na dosud neviděných vstupních hodnotách.
        \end{center}
    \end{timportantframe}

\end{frame}

\section{Učení bez učitele}

\titleframe{Učení bez učitele}

\subsection{Vymezení}

\begin{frame}{Učení bez učitele}

    \pause

    \begin{itemize}
        \item Máme pouze vstupní statistický soubor
        \[
            \mathbf{X}=(x_1,x_2,\dots,x_n).
        \]\pause
        \item Výstupní soubor
        \[
            \mathbf{Y}=(y_1,y_2,\dots,y_n)
        \]
        není znám.
    \end{itemize}

    \pause

    \begin{center}
        \[
            \text{učení s učitelem: } \boxed{\mathbf{X},\mathbf{Y}}
            \qquad\qquad
            \text{učení bez učitele: } \boxed{\mathbf{X}}
        \]
    \end{center}

    \pause

    \begin{timportantframe}
        \begin{center}
            \textbf{"Bez učitele"} znamená, že nemáme předem dané "správné odpovědi".
        \end{center}
    \end{timportantframe}

\end{frame}

\begin{frame}{Co je cílem učení bez učitele?}

    \pause

    \begin{itemize}
        \item Cílem není odhadovat známý výstup $\mathbf{Y}$.\pause
        \item Cílem je hledat určitou \textbf{strukturu} uvnitř samotných $\mathbf{X}$.
    \end{itemize}

    \pause

    \begin{itemize}
        \item Typicky hledáme například:
        \begin{itemize}
            \item skupiny podobných objektů,
            \item odlehlé hodnoty,
            \item skryté vztahy mezi objekty,
            \item jednodušší reprezentaci dat.
        \end{itemize}
    \end{itemize}
\end{frame}

            \mathbf{X}_2=(x_{21},\dots,x_{2n}).

\end{frame}

\subsection{Algoritmus k-means}

\begin{frame}{Algoritmus $k$-means}

    \pause

    \begin{itemize}
        \item Chceme rozdělit body do $k$ shluků.\pause
        \item Každý shluk reprezentuje jeho \textbf{střed} (anglicky \markdarkred{centroid}).
    \end{itemize}

    \pause

    \begin{center}
        \input{images/algoritmus-k-means-1.tex}
    \end{center}

    \pause

    \begin{tinfoframe}[\textwidth]
        \begin{center}
            Algoritmus končí ve~chvíli, kdy se přiřazení bodů nebo poloha středů už "výrazně" nemění.
        \end{center}
    \end{tinfoframe}

\end{frame}

\begin{frame}{Algoritmus k-means}

    \pause

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.52\textwidth}
            \centering

            \input{images/algoritmus-k-means-2.tex}

        \end{column}

        \begin{column}{0.44\textwidth}

            \pause

            \begin{itemize}
                \item Body jsou přiřazeny k nejbližšímu středu.\pause
                \item Nový střed je průměr bodů v daném shluku.\pause
                \item Pro každý shluk tak vznikne jeho reprezentant.
            \end{itemize}

        \end{column}

    \end{columns}

    \pause

    \begin{center}
        Shluk je tedy reprezentován jeho středem.
    \end{center}

\end{frame}

\begin{frame}{Pseudokód $k$-means}
    \scriptsize
    \begin{center}
        \begin{minipage}{0.9\textwidth}
            \begin{algorithm}[H]
                \caption{$k$-means}
                \KwIn{Objekty $\mathbf{x}_1,\dots,\mathbf{x}_n\in\mathbb{R}^d$
                a~počet shluků $k$}
                \KwOut{Shluky $C_1,\dots,C_k$ a~jejich středy
                $\boldsymbol{\mu}_1,\dots,\boldsymbol{\mu}_k$}

                Zvol počáteční středy
                $\boldsymbol{\mu}_1,\dots,\boldsymbol{\mu}_k$\\

                \lForEach{\textup{objekt $\mathbf{x}_i$}}{
                    $c(i)\gets 0$
                }

                $\operatorname{cont}\gets 1$\\

                \While{$\operatorname{cont}= 1$}{
                    $\operatorname{cont}\gets 0$\\

                    \tcp{Přiřazení objektů k nejbližším středům}
                    \ForEach{$i=1,\dots,n$}{
                        $j\gets
                        \argmin_{1\leq \ell\leq k}
                        d(\mathbf{x}_i,\boldsymbol{\mu}_{\ell})$

                        \If{$c(i)\neq j$}{
                            $c(i)\gets j$\\
                            $\operatorname{cont}\gets 1$
                        }
                    }

                    \tcp{Přepočet středů shluků}
                    \For{$j\gets 1,\dots,k$}{
                        $C_j\gets
                        \set{\mathbf{x}_i: c(i)=j}$\\

                        \lIf{$C_j\neq\emptyset$}{
                            $\boldsymbol{\mu}_j
                            \gets
                            \frac{1}{|C_j|}
                            \sum_{\mathbf{x}_i\in C_j}\mathbf{x}_i$
                        }
                    }
                }
            \end{algorithm}
        \end{minipage}
    \end{center}
\end{frame}

\begin{frame}{Co víme o $k$-means?}
    \begin{propblock}[Bez důkazu]
        Algoritmus $k$-means se vždy zastaví.
    \end{propblock}

    \pause
    \textbf{Nevýhody:}
    \begin{itemize}
        \item Počet shluků $k$ zadáváme předem.\pause
        \item Výsledek může záviset na počáteční volbě středů.\pause
        \item Algoritmus nemusí najít nejlepší možné rozdělení.\pause
        \item Nehodí se pro všechny tvary shluků.
    \end{itemize}

\end{frame}

\begin{frame}{Další úlohy učení bez učitele}

    \pause

    \begin{itemize}
        \item \textbf{Redukce dimenze:} zjednodušení dat při zachování důležitých informací.\pause
        \item \textbf{Hledání odlehlých hodnot:} nalezení objektů,
        které se výrazně liší od ostatních.\pause
        \item \textbf{Asociace a pravidelnosti:} hledání často společně
        se vyskytujících jevů.\pause
        \item \textbf{Vizualizace dat:} převod složitých dat do podoby,
        ve které je možné je lépe pochopit.
    \end{itemize}

\end{frame}

\subsection{Vymezení}

\subsection{Zpětnovazební učení}

\begin{frame}{Zpětnovazební učení}

    \pause

    \begin{itemize}
        \item Mějme množinu možných stavů prostředí $\mathcal{S}$ a~pro každý stav $s\in\mathcal{S}$ množinu přípustných akcí $\mathcal{A}(s)$.\pause
        \item V čase $t$ se agent nachází ve stavu $s_t\in\mathcal{S}$ a~zvolí akci $a_t\in\mathcal{A}(s_t)$.\pause
        \item Prostředí následně agentovi poskytne
        \begin{itemize}
            \item číselnou odměnu $r_{t+1}\in\mathbb{R}$ a
            \item nový stav $s_{t+1}\in\mathcal{S}$.
        \end{itemize}
        \[
            \left(s_t,a_t,r_{t+1},s_{t+1}\right)
        \]
    \end{itemize}
    \pause

    \begin{tinfoframe}[11cm]
        \begin{center}
            Jedna taková čtveřice představuje jednu \markdarkred{zkušenost agenta}.
        \end{center}
    \end{tinfoframe}

\end{frame}

\subsection{Posloupnost zkušeností}

\begin{frame}{Posloupnost zkušeností}

    \pause

    \begin{itemize}
        \item Agent při interakci s~prostředím získává posloupnost
        \[
            s_0,a_0,r_1,s_1,a_1,r_2,s_2,a_2,r_3,\dots
        \]\pause
        \item Po každé akci tedy pozoruje její následek:
        \[
            (s_t,a_t)
            \longrightarrow
            (r_{t+1},s_{t+1}).
        \]\pause
        \item Odměna $r_{t+1}$ vyjadřuje, nakolik byl bezprostřední následek akce žádoucí.\pause
        \item Odměna však obvykle přímo neříká, která akce byla správná.
    \end{itemize}

    \pause

    \begin{tinfoframe}[9.5cm]
        \begin{center}
            Agent se učí z~následků vlastních rozhodnutí.
        \end{center}
    \end{tinfoframe}

\end{frame}

\begin{frame}{Agent a~prostředí}

    \begin{center}

    \end{center}

    \pause

    \begin{enumerate}
        \item Agent pozoruje současný stav prostředí $s_t$.\pause
        \item Na jeho základě zvolí akci $a_t$.\pause
        \item Prostředí přejde do~nového stavu $s_{t+1}$ a~poskytne agentovi odměnu $r_{t+1}$.
    \end{enumerate}

\subsection{Cíl zpětnovazebního učení}

\begin{frame}{Cíl zpětnovazebního učení}

    \pause

    \begin{itemize}
        \item Agent nechce maximalizovat pouze okamžitou odměnu,
        ale celkový dlouhodobý zisk $\sum_{i=0}^{\infty}r_i$.\pause
        \item Často proto používáme diskontovaný součet odměn
        \[
            G(t)
            =
            \sum_{i=0}^{\infty}\gamma^{i}r_{t+i+1},
        \]
        kde $0\leqslant\gamma<1$.
    \end{itemize}

    \pause

    \begin{itemize}
        \item Parametr $\gamma$ určuje, jak velkou váhu mají budoucí odměny.
    \end{itemize}

    \pause

    \begin{align*}
        \gamma \approx 0
        \quad&\Rightarrow\quad
        \text{důraz na okamžitou odměnu}\\
        \gamma \approx 1
        \quad&\Rightarrow\quad
        \text{důraz na dlouhodobý výsledek}
    \end{align*}

    \pause

    \begin{center}
        Cílem je naučit se volit akce tak, aby \markdarkred{očekávaný dlouhodobý zisk byl co největší}.
    \end{center}

\end{frame}

\subsection{Politika agenta}

\begin{frame}{Politika agenta}

    \small

    \pause

    \begin{itemize}
        \item \textbf{Politika} určuje, jakou akci agent zvolí v~daném stavu $\pi(s)=a$.
    \end{itemize}

    \pause

    \begin{center}
        \input{images/politika-agenta.tex}
    \end{center}

    \pause

    \begin{itemize}
        \item Při použití politiky $\pi$ získá agent posloupnost odměn
        \[
            r_1^\pi,r_2^\pi,r_3^\pi,\dots
        \]
        a celkovou diskontovanou odměnu
        \[
            G^\pi=G^{\pi}(0)
            =
            r_1^\pi
            +
            \gamma r_2^\pi
            +
            \gamma^2r_3^\pi
            +
            \cdots.
        \]\pause
        \item Cílem je nalézt politiku s nejvyšší celkovou odměnou:
        \[
            \pi^*=\argmax_{\pi} G^\pi.
        \]
    \end{itemize}

    \pause

    \begin{center}
        \textbf{Agent se tedy neučí jednu akci,
        ale celou strategii chování.}
    \end{center}

\end{frame}

\subsection{Q-learning}

\begin{frame}{Q-learning}{základní myšlenka}

    \pause

    \begin{itemize}
        \item Agent si pro každou dvojici
        \[
            (s,a)
        \]
        pamatuje hodnotu
        \[
            Q(s,a).
        \]
    \end{itemize}

    \pause

    \begin{center}
        \textbf{$Q(s,a)$ říká, jak výhodné se zatím jeví provést akci $a$ ve stavu $s$.}
    \end{center}

    \pause

    \begin{itemize}
        \item Po každé zkušenosti
        \[
            (s_t,a_t,r_{t+1},s_{t+1})
        \]
        agent hodnotu $Q(s_t,a_t)$ upraví.\pause
        \item Čím větší je $Q(s,a)$, tím lákavější je akci $a$ ve stavu $s$ zvolit.
    \end{itemize}

\end{frame}

\begin{frame}{Aktualizace hodnot $Q(s,a)$}

    \pause

    \begin{itemize}
        \item Po provedení akce $a_t$ ve stavu $s_t$ agent získá odměnu
        $r_{t+1}$ a nový stav $s_{t+1}$.
        \[
            Q(s_t,a_t)
            \gets
            Q(s_t,a_t)
            +
            \alpha
            \left[
                r_{t+1}
                +
                \gamma
                \max_a Q(s_{t+1},a)
                -
                Q(s_t,a_t)
            \right].
        \]
        \item $\alpha$ určuje rychlost učení.
        \item $\gamma$ určuje váhu budoucích odměn.
        \item $\max_a Q(s_{t+1},a)$ odhaduje nejlepší možné pokračování.
    \end{itemize}

    \pause

    \begin{tinfoframe}[\textwidth]
        \begin{center}
            Agent tedy postupně opravuje své odhady podle toho, jaké možnosti prozkoumal.
        \end{center}
    \end{tinfoframe}

\end{frame}

\begin{frame}{Výsledná politika $\pi$}

    \pause

    \begin{itemize}
        \item Po dostatečném množství zkušeností má agent
        pro každý stav a každou přípustnou akci hodnotu $Q(s,a)$.
    \end{itemize}

    \pause

    \begin{center}
        \begin{tabular}{c|c}
            akce $a$ & $Q(s,a)$\\
            \hline
            nahoru  & $1{,}4$\\
            doprava & $4{,}2$\\
            dolů    & $-0{,}5$\\
            vlevo   & $0{,}8$
        \end{tabular}
    \end{center}

    \pause

    \begin{itemize}
        \item Agent ve stavu $s$ zvolí akci s nejvyšší hodnotou:
        \[
            \pi(s)=\argmax_{a}Q(s,a)=\text{doprava}.
        \]\pause
        \item Naučená politika je tedy tvořena nejlepší akcí pro každý možný stav.
    \end{itemize}

\end{frame}
