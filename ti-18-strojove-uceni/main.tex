\subsection{Vymezení pojmu "strojové učení"}

\begin{frame}{Co se rozumí strojovým učením?}

    \small



    \visible<2->{\begin{itemize}
        \item Ve statistice jsme pracovali se statistickými soubory zachycujícími jeden, nebo více statistických znaků:
    \end{itemize}

    \[
        \mathbf{X}
        =
        \begin{array}{c@{\;}c}
            \begin{array}{cccc}
                \mathbf{X}_1
                &
                \mathbf{X}_2
                &
                \cdots
                &
                \mathbf{X}_p
            \end{array}
            &
            \\[0.2cm]
            \left(
            \begin{array}{cccc}
                x_{11} & x_{12} & \cdots & x_{1p}\\
                x_{21} & x_{22} & \cdots & x_{2p}\\
                \vdots & \vdots & \ddots & \vdots\\
                x_{n1} & x_{n2} & \cdots & x_{np}
            \end{array}
            \right)
            &
            \begin{array}{c}
                \mathbf{x}_1\\
                \mathbf{x}_2\\
                \vdots\\
                \mathbf{x}_n
            \end{array}
        \end{array}.
    \]}



    \visible<3->{\begin{itemize}
        \item Typicky jsme však pracovali se soubory popisujícími \markdarkred{jeden statistický znak}, tj. $\mathbf{X}=(x_1,\dots,x_n)$.
    \end{itemize}}



    \visible<4->{\begin{center}
        \obrazek[0.4\textheight]{images/co-se-rozumi-strojovym-ucenim.tex}
    \end{center}}



    \visible<5->{\begin{center}
        \textbf{Strojové učení:} Část chování systému se nastavuje
        podle dat. 
    \end{center}}

\end{frame}

\begin{frame}{Co se rozumí strojovým učením?}

    \begin{tinfoframe}
        Obecně strojovým učením rozumíme skupinu statistických algoritmů, které se učí \markdarkred{ze~známých dat} a~zobecňují výsledky \markdarkred{na data neznámá}.
    \end{tinfoframe}

\end{frame}

\subsection{Trénovací a~testovací data}

\begin{frame}{Trénovací a~testovací data}

    \footnotesize



    \visible<2->{\begin{itemize}
        \item Mějme statistické soubory
        \[
            \mathbf{X}_1,\mathbf{X}_2,\dots,\mathbf{X}_k,
        \]
        kde hodnota $x_{ij}$ popisuje $j$-tý znak u~$i$-tého objektu.
    \end{itemize}}



    \visible<3->{\[
        \mathbf{x}_i=(x_{i1},x_{i2},\dots,x_{ik})
    \]}



    \visible<4->{\begin{center}
        \obrazek[0.28\textheight]{images/trenovaci-a-testovaci-data.tex}
    \end{center}}



    \visible<5->{\begin{itemize}
        \item \textbf{Trénovací data} používáme k~nastavení modelu.
        \item \textbf{Testovací data} model při trénování nevidí;
        používáme je až k~ověření, zda model funguje i~na nových datech.
    \end{itemize}}



    \visible<6->{\begin{center}
        \textbf{Dobrý model se nemá jen naučit trénovací data nazpaměť.}
    \end{center}}

\end{frame}

\subsection{Učení a~trénování modelu}

\begin{frame}{Učení a~trénování modelu}



    \visible<2->{\begin{itemize}
        \item \textbf{Učení} je obecný proces, při němž model získává
        z~dat nebo zkušenosti informaci a~přizpůsobuje jí své chování.
    \end{itemize}}



    \visible<3->{\begin{itemize}
        \item \textbf{Trénování} je konkrétní fáze učení, během níž
        algoritmus upravuje model podle \textbf{trénovacích dat}
        a~podle toho, jaké chyby model dělá.
    \end{itemize}}



    \visible<4->{\vspace{0.2cm}

    \begin{center}
        \obrazek[0.4\textheight]{images/uceni-a-trenovani-modelu.tex}
    \end{center}}



    \visible<5->{\vspace{0.15cm}

    \begin{itemize}
        \item Data $\mathbf{X}_1,\dots,\mathbf{X}_k$ zde chápeme obecně:
        mohou to být různé statistické znaky, pozorování nebo jiné
        informace, z~nichž se model učí.
    \end{itemize}}

\end{frame}

\begin{frame}{Učení a~trénování modelu}


    \visible<2->{\begin{itemize}
        \item Při trénování model vytváří výstupy, ty se vyhodnocují
        a~podle vzniklé chyby se upravují parametry
        $\boldsymbol{\theta}=(\theta_1,\dots,\theta_m)$.
        \visible<3->{\item Pomocí výsledného modelu $f_{\boldsymbol{\theta}}$ spočítá
        \[\hat{y}_{ij} = f_{\boldsymbol{\theta}}(x_{ij}),\]
        přičemž $\mathbf{X}_i=(x_{i1},\dots,x_{in})$ a~$\hat{\mathbf{Y}}_i=(\hat{y}_{i1},\dots,\hat{y}_{in})$.}
    \end{itemize}}



    \visible<4->{\begin{tinfoframe}
        \begin{center}
            \textbf{Po skončení trénování} používáme model pro nová data.
        \end{center}
    \end{tinfoframe}}



    \visible<5->{\begin{tnoticeframe}
        Pro jednoduchost se zaměříme na modely, které mají na vstupu pouze statistický soubor (trénovací data) $\mathbf{X}$, popř. dvojici souborů $\mathbf{X},\mathbf{Y}$.
    \end{tnoticeframe}}

\end{frame}

\begin{frame}{Základní členění strojového učení}



    \visible<2->{\begin{center}
        \obrazek[0.4\textheight]{images/zakladni-cleneni-strojoveho-uceni-1.tex}
    \end{center}

    \vspace{0.1cm}}



    \visible<3->{\begin{itemize}
        \item \textbf{Učení s~učitelem:}
        Máme spárované statistické soubory
        \[
            (x_1,y_1),\ldots,(x_n,y_n),
        \]
        tedy vstupy $\mathbf{X}$ a~známé výstupy $\mathbf{Y}$.
        Cílem je odhadovat $y$ pro nové vstupy $x$.
        \visible<4->{\vspace{0.1cm}
        \item \textbf{Učení bez učitele:}
        Máme pouze statistický soubor
        \[
            x_1,\ldots,x_n,
        \]
        tedy data $\mathbf{X}$ bez známého výstupu $\mathbf{Y}$.
        Cílem je hledat strukturu, podobnosti nebo skupiny v~datech.}
    \end{itemize}}

\end{frame}

\subsection{Základní členění ML}

\begin{frame}{Základní členění strojového učení}



    \visible<2->{\begin{center}
        \obrazek[0.4\textheight]{images/zakladni-cleneni-strojoveho-uceni-2.tex}
    \end{center}

    \vspace{0.1cm}}

        \]
    \end{itemize}}
    
\end{frame}

\subsection{Vymezení}

\begin{frame}{Učení s~učitelem}



    \visible<2->{\begin{itemize}
        \item Máme dva statistické soubory
        \[
            \mathbf{X}=(x_1,x_2,\dots,x_n),
            \qquad
            \mathbf{Y}=(y_1,y_2,\dots,y_n).
        \]
        \visible<3->{\item Každé vstupní hodnotě $x_j$ odpovídá známý výstup $y_j$:
        \[
            (x_1,y_1),(x_2,y_2),\dots,(x_n,y_n).
        \]}
        \visible<4->{\item Při trénování používáme dvojice $(x_j,y_j)$.}
    \end{itemize}}



    \visible<5->{\begin{tnoticeframe}
        \begin{center}
            \textbf{"Učitel"} znamená, že při trénování známe
            požadovaný výstup pro každý trénovací vstup.
        \end{center}
    \end{tnoticeframe}}

\end{frame}

\begin{frame}{Jak probíhá učení s~učitelem?}
    


    \visible<2->{\begin{center}
        \obrazek[0.4\textheight]{images/jak-probiha-uceni-s-ucitelem.tex}
    \end{center}}



    \visible<3->{\begin{itemize}
        \item Model se z~trénovacích dat učí vztah mezi
        vstupem $\mathbf{X}$ a~výstupem~$\mathbf{Y}$.
        \visible<4->{\item Po skončení trénování odhaduje výstup také
        pro nové vstupní hodnoty.}
    \end{itemize}}

\end{frame}

\subsection{Regrese a~klasifikace}

\begin{frame}{Regrese a~klasifikace}

    \small

    \begin{center}
        \obrazek[0.4\textheight]{images/regrese-a-klasifikace.tex}
    \end{center}



    \visible<2->{\begin{itemize}
        \item \textbf{Regrese:} chceme odhadnout číselnou hodnotu.
        \[
            \text{počet odpracovaných hodin}
            \longmapsto
            \text{výše mzdy}
        \]
        \visible<3->{\item \textbf{Klasifikace:} chceme určit kategorii.
        \[
            \text{počet bodů}
            \longmapsto
            \text{prospěl / neprospěl}
        \]}
    \end{itemize}}

\end{frame}

\subsection{Regrese}

\begin{frame}{Regrese}



    \visible<2->{\begin{itemize}
        \item Při regresi je výstupní statistický soubor $\mathbf{Y}$ číselný, tj. $y_j\in\mathbb{R}$.
        \visible<3->{\item Model se snaží popsat závislost výstupu $\mathbf{Y}$ na~vstupu $\mathbf{X}$.}
        \visible<4->{\[
            \hat y=\hat f(x).
        \]}
        \visible<5->{\item Výsledkem je číselný odhad pro novou vstupní hodnotu $x$.}
    \end{itemize}}



    \visible<6->{\begin{center}
        \textbf{Regrese nemusí být lineární.}\\
        Lineární regrese je pouze jedním z~možných regresních modelů.
    \end{center}}

\end{frame}

\subsection{Lineární regrese}

\begin{frame}{Jednoduchá lineární regrese}

    \small

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.48\textwidth}



            \visible<2->{\begin{itemize}
                \item Máme vstupní statistický soubor $\mathbf{X}$ a~číselný výstupní soubor $\mathbf{Y}$,
                \[
                    y_j=\beta_0+\beta_1x_j+\varepsilon_j.
                \]
                \visible<3->{\item Z~trénovacích dvojic $(x_j,y_j)$
                odhadneme koeficienty $\beta_0$ a~$\beta_1$.
                \[
                    \hat y=\hat\beta_0+\hat\beta_1x.
                \]}
            \end{itemize}}       

        \end{column}

        \begin{column}{0.48\textwidth}\visible<3->{
    \obrazek[0.4\textheight]{images/jednoducha-linearni-regrese.tex}

        }\end{column}
    \end{columns}

\end{frame}

\subsection{Klasifikace}

\begin{frame}{Klasifikace}

    \small



    \visible<2->{\begin{itemize}
        \item Při klasifikaci výstupní statistický soubor
        $\mathbf{Y}$ obsahuje místo číselných hodnot označení tříd:
        \[
            y_j\in\set{A,B,C,\dots}.
        \]
        \visible<3->{\item Model pro nový vstup $x$ určí odhadovanou třídu:
        \[
            \hat y=\hat f(x).
        \]}
        \visible<4->{\item \textbf{Binární klasifikace:} máme dvě možné třídy.}

        \visible<5->{\item \textbf{Vícetřídní klasifikace:} máme více možných tříd.}
    \end{itemize}}

\end{frame}

\begin{frame}{Rozhodovací hranice}



    \visible<2->{\begin{itemize}
        \item Uvažujme objekty popsané jediným vstupním znakem $\mathbf{X}$.
        \visible<3->{\item Klasifikační model rozdělí možné hodnoty $\mathbf{X}$
        na oblasti odpovídající různým třídám.}
    \end{itemize}}



    \visible<4->{\begin{center}
        \obrazek[0.4\textheight]{images/rozhodovaci-hranice-1.tex}
    \end{center}}



    \visible<5->{\begin{center}
        Například:
        \[
            \hat y=
            \begin{cases}
                A, & x<c,\\
                B, & x\geqslant c.
            \end{cases}
        \]
    \end{center}}

\end{frame}

\begin{frame}{Rozhodovací hranice}

    \small



    \visible<2->{\begin{itemize}
        \item Mějme dva vstupní statistické soubory
        \[
            \mathbf{X}_1=(x_{11},\dots,x_{1n}),
            \qquad
            \mathbf{X}_2=(x_{21},\dots,x_{2n})
        \]
        a~výstupní soubor
        \[
            \mathbf{Y}=(y_1,\dots,y_n),
            \qquad
            y_j\in\set{A,B}.
        \]
        \visible<3->{\item $j$-tý objekt je popsán bodem $(x_{1j},x_{2j})$ a~přísluší mu třída $y_j$.}
    \end{itemize}}



    \visible<4->{\begin{columns}[T,onlytextwidth]

        \begin{column}{0.60\textwidth}
    \obrazek[0.4\textheight]{images/rozhodovaci-hranice-2.tex}
        \end{column}



        \begin{column}{0.36\textwidth}\visible<5->{

            Model rozdělí rovinu možných hodnot na~dvě oblasti.
            \visible<6->{\[
                \hat y=
                \begin{cases}
                    A,
                    & x_1+x_2<5,\\[1mm]
                    B,
                    & x_1+x_2\geqslant 5.
                \end{cases}
            \]}

        }\end{column}

    \end{columns}}

\end{frame}

\subsection{Vyhodnocení modelu}

\begin{frame}{Vyhodnocení modelu u~učení s~učitelem}
    \small



    \visible<2->{\begin{itemize}
        \item Pro každý index $j$ model vytvoří odhad
        \[
            \hat y_j=\hat f(x_j).
        \]
        \visible<3->{\item Odhad $\hat y_j$ porovnáme se známou skutečnou hodnotou $y_j$.}
        \visible<4->{\item U~regrese lze sledovat chybu
        \[
            e_j=y_j-\hat y_j.
        \]}
        \visible<5->{\item U~klasifikace sledujeme, zda model určil správnou třídu:
        \[
            \hat y_j=y_j
            \qquad\text{nebo}\qquad
            \hat y_j\neq y_j.
        \]}
    \end{itemize}}



    \visible<6->{\begin{timportantframe}
        \begin{center}
            Testovací data ukazují, jak model funguje na dosud neviděných vstupních hodnotách.
        \end{center}
    \end{timportantframe}}

\end{frame}

\section{Učení bez učitele}

\begin{titleframeenv}{Učení bez učitele}

\subsection{Vymezení}

\begin{frame}{Učení bez učitele}



    \visible<2->{\begin{itemize}
        \item Máme pouze vstupní statistický soubor
        \[
            \mathbf{X}=(x_1,x_2,\dots,x_n).
        \]
        \visible<3->{\item Výstupní soubor
        \[
            \mathbf{Y}=(y_1,y_2,\dots,y_n)
        \]
        není znám.}
    \end{itemize}}



    \visible<4->{\begin{center}
        \[
            \text{učení s učitelem: } \boxed{\mathbf{X},\mathbf{Y}}
            \qquad\qquad
            \text{učení bez učitele: } \boxed{\mathbf{X}}
        \]
    \end{center}}



    \visible<5->{\begin{timportantframe}
        \begin{center}
            \textbf{"Bez učitele"} znamená, že nemáme předem dané "správné odpovědi".
        \end{center}
    \end{timportantframe}}

\end{frame}

\begin{frame}{Co je cílem učení bez učitele?}



    \visible<2->{\begin{itemize}
        \item Cílem není odhadovat známý výstup $\mathbf{Y}$.
        \visible<3->{\item Cílem je hledat určitou \textbf{strukturu} uvnitř samotných $\mathbf{X}$.}
    \end{itemize}}



    \visible<4->{\begin{itemize}
        \item Typicky hledáme například:
        \begin{itemize}
            \item skupiny podobných objektů,
            \item odlehlé hodnoty,
            \item skryté vztahy mezi objekty,
            \item jednodušší reprezentaci dat.
        \end{itemize}
    \end{itemize}}
\end{frame}

\begin{frame}{Příklad}{Hledání skupin v~datech}



    \visible<2->{\begin{itemize}
        \item Mějme dva vstupní statistické soubory
        \[
            \mathbf{X}_1=(x_{11},\dots,x_{1n}),
            \qquad
            \mathbf{X}_2=(x_{21},\dots,x_{2n}).
        \]
        \visible<3->{\item Každý objekt $j$ je popsán bodem $(x_{1j},x_{2j})$.}
    \end{itemize}}



    \visible<4->{\begin{center}
        \obrazek[0.28\textheight]{images/priklad-hledani-skupin-v-datech.tex}
    \end{center}}



    \visible<5->{\begin{tinfoframe}
        \begin{center}
            Algoritmus se snaží skupiny odhalit \textbf{bez znalosti tříd}
            nebo \textbf{známých výstupů}.
        \end{center}
    \end{tinfoframe}}

\end{frame}

\subsection{Shluková analýza}

\begin{frame}{Shluková analýza}



    \visible<2->{\begin{itemize}
        \item \textbf{Shlukování} rozděluje objekty do~shluků (anglicky \markdarkred{cluster}) podle jejich podobnosti.
        \visible<3->{\item Objekty ve stejném shluku mají být navzájem podobné.}
        \visible<4->{\item Objekty v~různých shlucích mají být naopak odlišné.}
    \end{itemize}}

    \visible<4->{\vspace{0.15cm}

    \begin{center}
        \obrazek[0.4\textheight]{images/shlukova-analyza.tex}
    \end{center}}



    \visible<5->{\begin{itemize}
        \item Při shlukování musíme rozhodnout například:
        \begin{itemize}
            \item jak budeme měřit podobnost,
            \item kolik shluků očekáváme,
            \item jaký tvar shluků považujeme za přirozený.
        \end{itemize}
    \end{itemize}}

\end{frame}

\begin{frame}{Podobnost a~vzdálenost}



    \visible<2->{\begin{itemize}
        \item U~bodů v~rovině lze podobnost měřit např. \markdarkred{eukleidovskou vzdáleností}.
        \[
            d(P,Q)
            =
            \sqrt{(p_1-q_1)^2+(p_2-q_2)^2}.
        \]
    \end{itemize}}



    \visible<3->{\begin{center}

    \end{center}}



    \visible<4->{\begin{timportantframe}
        \begin{center}
            Výsledek shlukování závisí na~tom, jak podobnost definujeme.
        \end{center}
    \end{timportantframe}}

\end{frame}

\subsection{Algoritmus k-means}

\begin{frame}{Algoritmus $k$-means}



    \visible<2->{\begin{itemize}
        \item Chceme rozdělit body do $k$ shluků.
        \visible<3->{\item Každý shluk reprezentuje jeho \textbf{střed} (anglicky \markdarkred{centroid}).}
    \end{itemize}}



    \visible<4->{\begin{center}
        \obrazek[0.4\textheight]{images/algoritmus-k-means-1.tex}
    \end{center}}



    \visible<5->{\begin{tinfoframe}
        \begin{center}
            Algoritmus končí ve~chvíli, kdy se přiřazení bodů nebo poloha středů už "výrazně" nemění.
        \end{center}
    \end{tinfoframe}}

\end{frame}

\begin{frame}{Algoritmus k-means}



    \visible<2->{\begin{columns}[T,onlytextwidth]

        \begin{column}{0.52\textwidth}
    \obrazek[0.4\textheight]{images/algoritmus-k-means-2.tex}

        \end{column}

        \begin{column}{0.44\textwidth}



            \visible<3->{\begin{itemize}
                \item Body jsou přiřazeny k~nejbližšímu středu.
                \visible<4->{\item Nový střed je průměr bodů v~daném shluku.}
                \visible<5->{\item Pro každý shluk tak vznikne jeho reprezentant.}
            \end{itemize}}

        \end{column}

    \end{columns}}



    \visible<6->{\begin{center}
        Shluk je tedy reprezentován jeho středem.
    \end{center}}

\end{frame}

\begin{frame}{Pseudokód $k$-means}
    \scriptsize
    \begin{center}
        \begin{adjustbox}{max width=\linewidth, max totalheight=0.8\textheight}
        \begin{minipage}{0.9\textwidth}
            \begin{algorithm}[H]
                \caption{$k$-means}
                \KwIn{Objekty $\mathbf{x}_1,\dots,\mathbf{x}_n\in\mathbb{R}^d$
                a~počet shluků $k$}
                \KwOut{Shluky $C_1,\dots,C_k$ a~jejich středy
                $\boldsymbol{\mu}_1,\dots,\boldsymbol{\mu}_k$}

                Zvol počáteční středy
                $\boldsymbol{\mu}_1,\dots,\boldsymbol{\mu}_k$\\

                \lForEach{\textup{objekt $\mathbf{x}_i$}}{
                    $c(i)\gets 0$
                }

                $\operatorname{cont}\gets 1$\\

                \While{$\operatorname{cont}= 1$}{
                    $\operatorname{cont}\gets 0$\\

                    \tcp{Přiřazení objektů k~nejbližším středům}
                    \ForEach{$i=1,\dots,n$}{
                        $j\gets
                        \argmin_{1\leqslant \ell\leqslant k}
                        d(\mathbf{x}_i,\boldsymbol{\mu}_{\ell})$

                        \If{$c(i)\neq j$}{
                            $c(i)\gets j$\\
                            $\operatorname{cont}\gets 1$
                        }
                    }

                    \tcp{Přepočet středů shluků}
                    \For{$j\gets 1,\dots,k$}{
                        $C_j\gets
                        \set{\mathbf{x}_i: c(i)=j}$\\

                        \lIf{$C_j\neq\emptyset$}{
                            $\boldsymbol{\mu}_j
                            \gets
                            \frac{1}{|C_j|}
                            \sum_{\mathbf{x}_i\in C_j}\mathbf{x}_i$
                        }
                    }
                }
            \end{algorithm}
        \end{minipage}
        \end{adjustbox}
    \end{center}
\end{frame}

\begin{frame}{Co víme o~$k$-means?}
    \begin{propblock}[Bez důkazu]
        Algoritmus $k$-means se vždy zastaví.
    \end{propblock}


    \visible<2->{\textbf{Nevýhody:}
    \begin{itemize}
        \item Počet shluků $k$ zadáváme předem.
        \visible<3->{\item Výsledek může záviset na počáteční volbě středů.}
        \visible<4->{\item Algoritmus nemusí najít nejlepší možné rozdělení.}
        \visible<5->{\item Nehodí se pro všechny tvary shluků.}
    \end{itemize}}

\end{frame}

\begin{frame}{Další úlohy učení bez učitele}



    \visible<2->{\begin{itemize}
        \item \textbf{Redukce dimenze:} zjednodušení dat při zachování důležitých informací.
        \visible<3->{\item \textbf{Hledání odlehlých hodnot:} nalezení objektů,
        které se výrazně liší od ostatních.}
        \visible<4->{\item \textbf{Asociace a~pravidelnosti:} hledání často společně
        se vyskytujících jevů.}
        \visible<5->{\item \textbf{Vizualizace dat:} převod složitých dat do podoby,
        ve které je možné je lépe pochopit.}
    \end{itemize}}

\end{frame}

\section{Zpětnovazební učení}

\begin{titleframeenv}{Zpětnovazební učení}

\end{titleframeenv}

\subsection{Zpětnovazební učení}

\begin{frame}{Zpětnovazební učení}



    \visible<2->{\begin{itemize}
        \item Mějme množinu možných stavů prostředí $\mathcal{S}$ a~pro každý stav $s\in\mathcal{S}$ množinu přípustných akcí $\mathcal{A}(s)$.
        \visible<3->{\item V~čase $t$ se agent nachází ve stavu $s_t\in\mathcal{S}$ a~zvolí akci $a_t\in\mathcal{A}(s_t)$.}
        \visible<4->{\item Prostředí následně agentovi poskytne
        \begin{itemize}
            \item číselnou odměnu $r_{t+1}\in\mathbb{R}$ a
            \item nový stav $s_{t+1}\in\mathcal{S}$.
        \end{itemize}
        \[
            \left(s_t,a_t,r_{t+1},s_{t+1}\right)
        \]}
    \end{itemize}}


    \visible<5->{\begin{tinfoframe}
        \begin{center}
            Jedna taková čtveřice představuje jednu \markdarkred{zkušenost agenta}.
        \end{center}
    \end{tinfoframe}}

\end{frame}

\subsection{Posloupnost zkušeností}

\begin{frame}{Posloupnost zkušeností}



    \visible<2->{\begin{itemize}
        \item Agent při interakci s~prostředím získává posloupnost
        \[
            s_0,a_0,r_1,s_1,a_1,r_2,s_2,a_2,r_3,\dots
        \]
        \visible<3->{\item Po každé akci tedy pozoruje její následek:
        \[
            (s_t,a_t)
            \longrightarrow
            (r_{t+1},s_{t+1}).
        \]}
        \visible<4->{\item Odměna $r_{t+1}$ vyjadřuje, nakolik byl bezprostřední následek akce žádoucí.}
        \visible<5->{\item Odměna však obvykle přímo neříká, která akce byla správná.}
    \end{itemize}}



    \visible<6->{\begin{tinfoframe}
        \begin{center}
            Agent se učí z~následků vlastních rozhodnutí.
        \end{center}
    \end{tinfoframe}}

\end{frame}

\subsection{Agent a~prostředí}

\begin{frame}{Příklad}{Agent v~"mřížkovém světě"}

    \begin{columns}[T,onlytextwidth]

        \begin{column}{0.40\textwidth}
    \obrazek[0.4\textheight]{images/priklad-agent-v-mrizkovem-svete.tex}

        \end{column}

        \begin{column}{0.55\textwidth}



            \visible<2->{\begin{itemize}
                \item \textbf{Stav} $s_t$: aktuální políčko agenta.
                \visible<3->{\item \textbf{Akce} $a_t$: nahoru, dolů, vlevo, vpravo.}
            \end{itemize}}



            \visible<4->{\begin{itemize}
                \item \textbf{Odměna $r_{t+1}$}:
                \[
                    \begin{array}{rcl}
                    \text{běžný krok} &:& -1,\\
                    \text{cíl} &:& +10,\\
                    \text{past} &:& -10.
                    \end{array}
                \]
            \end{itemize}}

        \end{column}

    \end{columns}



    \visible<5->{\begin{tnoticeframe}
        \begin{center}
            Agent se učí volit cestu, která vede k~co nejvyšší celkové odměně.
        \end{center}
    \end{tnoticeframe}}

\end{frame}

\begin{frame}{Agent a~prostředí}

    \begin{center}
        \obrazek[0.4\textheight]{images/agent-a-prostredi.tex}
    \end{center}



    \visible<2->{\begin{enumerate}
        \item Agent pozoruje současný stav prostředí $s_t$.
        \visible<3->{\item Na jeho základě zvolí akci $a_t$.}
        \visible<4->{\item Prostředí přejde do~nového stavu $s_{t+1}$ a~poskytne agentovi odměnu $r_{t+1}$.}
    \end{enumerate}}



    \visible<5->{\begin{center}
        Tento cyklus se opakuje v~čase
        \[
            t=0,1,2,\dots
        \]
    \end{center}}

\end{frame}

\subsection{Cíl zpětnovazebního učení}

\begin{frame}{Cíl zpětnovazebního učení}



    \visible<2->{\begin{itemize}
        \item Agent nechce maximalizovat pouze okamžitou odměnu,
        ale celkový dlouhodobý zisk $\sum_{i=0}^{\infty}r_i$.
        \visible<3->{\item Často proto používáme diskontovaný součet odměn
        \[
            G(t)
            =
            \sum_{i=0}^{\infty}\gamma^{i}r_{t+i+1},
        \]
        kde $0\leqslant\gamma<1$.}
    \end{itemize}}



    \visible<4->{\begin{itemize}
        \item Parametr $\gamma$ určuje, jak velkou váhu mají budoucí odměny.
    \end{itemize}}



    \visible<5->{\begin{align*}
        \gamma \approx 0
        \quad&\Rightarrow\quad
        \text{důraz na okamžitou odměnu}\\
        \gamma \approx 1
        \quad&\Rightarrow\quad
        \text{důraz na dlouhodobý výsledek}
    \end{align*}}



    \visible<6->{\begin{center}
        Cílem je naučit se volit akce tak, aby \markdarkred{očekávaný dlouhodobý zisk byl co největší}.
    \end{center}}

\end{frame}

\subsection{Politika agenta}

\begin{frame}{Politika agenta}

    \small



    \visible<2->{\begin{itemize}
        \item \textbf{Politika} určuje, jakou akci agent zvolí v~daném stavu $\pi(s)=a$.
    \end{itemize}}



    \visible<3->{\begin{center}
        \obrazek[0.4\textheight]{images/politika-agenta.tex}
    \end{center}}



    \visible<4->{\begin{itemize}
        \item Při použití politiky $\pi$ získá agent posloupnost odměn
        \[
            r_1^\pi,r_2^\pi,r_3^\pi,\dots
        \]
        a~celkovou diskontovanou odměnu
        \[
            G^\pi=G^{\pi}(0)
            =
            r_1^\pi
            +
            \gamma r_2^\pi
            +
            \gamma^2r_3^\pi
            +
            \cdots.
        \]
        \visible<5->{\item Cílem je nalézt politiku s~nejvyšší celkovou odměnou:
        \[
            \pi^*=\argmax_{\pi} G^\pi.
        \]}
    \end{itemize}}



    \visible<6->{\begin{center}
        \textbf{Agent se tedy neučí jednu akci,
        ale celou strategii chování.}
    \end{center}}

\end{frame}

\subsection{Q-learning}

\begin{frame}{Q-learning}{základní myšlenka}



    \visible<2->{\begin{itemize}
        \item Agent si pro každou dvojici
        \[
            (s,a)
        \]
        pamatuje hodnotu
        \[
            Q(s,a).
        \]
    \end{itemize}}



    \visible<3->{\begin{center}
        \textbf{$Q(s,a)$ říká, jak výhodné se zatím jeví provést akci $a$ ve stavu $s$.}
    \end{center}}



    \visible<4->{\begin{itemize}
        \item Po každé zkušenosti
        \[
            (s_t,a_t,r_{t+1},s_{t+1})
        \]
        agent hodnotu $Q(s_t,a_t)$ upraví.
        \visible<5->{\item Čím větší je $Q(s,a)$, tím lákavější je akci $a$ ve stavu $s$ zvolit.}
    \end{itemize}}

\end{frame}

\begin{frame}{Aktualizace hodnot $Q(s,a)$}



    \visible<2->{\begin{itemize}
        \item Po provedení akce $a_t$ ve stavu $s_t$ agent získá odměnu
        $r_{t+1}$ a~nový stav $s_{t+1}$.
        \[
            Q(s_t,a_t)
            \gets
            Q(s_t,a_t)
            +
            \alpha
            \left[
                r_{t+1}
                +
                \gamma
                \max_a Q(s_{t+1},a)
                -
                Q(s_t,a_t)
            \right].
        \]
        \item $\alpha$ určuje rychlost učení.
        \item $\gamma$ určuje váhu budoucích odměn.
        \item $\max_a Q(s_{t+1},a)$ odhaduje nejlepší možné pokračování.
    \end{itemize}}



    \visible<3->{\begin{tinfoframe}
        \begin{center}
            Agent tedy postupně opravuje své odhady podle toho, jaké možnosti prozkoumal.
        \end{center}
    \end{tinfoframe}}

\end{frame}

\begin{frame}{Výsledná politika $\pi$}



    \visible<2->{\begin{itemize}
        \item Po dostatečném množství zkušeností má agent
        pro každý stav a~každou přípustnou akci hodnotu $Q(s,a)$.
    \end{itemize}}



    \visible<3->{\begin{center}
        \begin{tabular}{c|c}
            akce $a$ & $Q(s,a)$\\
            \hline
            nahoru  & $1{,}4$\\
            doprava & $4{,}2$\\
            dolů    & $-0{,}5$\\
            vlevo   & $0{,}8$
        \end{tabular}
    \end{center}}



    \visible<4->{\begin{itemize}
        \item Agent ve stavu $s$ zvolí akci s~nejvyšší hodnotou:
        \[
            \pi(s)=\argmax_{a}Q(s,a)=\text{doprava}.
        \]
        \visible<5->{\item Naučená politika je tedy tvořena nejlepší akcí pro každý možný stav.}
    \end{itemize}}

\end{frame}
