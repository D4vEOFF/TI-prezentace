    \subsection{Motivace k použití heuristiky}

    \begin{frame}{Motivace k použití heuristiky}
        \begin{itemize}
            \visible<2->{\item Velmi často se stává, že hledáme nejkratší cestu pouze do jednoho vrcholu.}
            \visible<3->{\item Tzn. máme graf ohodnocený graf $G=(V,E,w)$, počáteční vrchol $v_0\in V$ a nově \markdarkred{cílový vrchol $g\in V$}.}
            \visible<4->{\item V \textbf{nezáporně ohodnoceném grafu} lze použít Dijkstrův algoritmus.}
        \end{itemize}
        \visible<5->{\begin{timportantframe}
            \begin{center}
                Dijkstrův algoritmus nevyužívá znalosti cílového vrcholu $\implies$ všechny vrcholy "preferuje" stejně.
            \end{center}
        \end{timportantframe}}
        \visible<6->{\begin{tnoticeframe}
            \begin{center}
                Máme-li o cílovém vrcholu nějakou dodatečnou informaci, lze ji využít k \textbf{odhadu} vzdálenosti z libovolného vrcholu $v$ do $g$.
            \end{center}
        \end{tnoticeframe}}
    \end{frame}

    \begin{frame}{Bludiště}
        Např. v bludišti, kde vrcholy představují jednotlivá pole známe navíc i jejich souřadnice $(x,y)$.
        \begin{figure}
            \centering
            \input{images/bludiste.tex}
        \end{figure}
    \end{frame}

    \begin{frame}{Eukleidovská vzdálenost}
        \begin{figure}
            \centering
            \input{images/bludiste-euklid.tex}
        \end{figure}
        \begin{itemize}
            \item Předpokládejme, že máme pole se souřadnicemi $(x,y)$ a cíl má souřadnice $(x_g,y_g)$.
            \item \markdarkred{Eukleidovskou vzdáleností} těchto dvou polí se rozumí hodnota
            \[\psi_e(x,y)=\sqrt{(x_g-x)^{2}+(y_g-y)^{2}}.\]
        \end{itemize}
    \end{frame}

    \begin{frame}{Manhattanská vzdálenost}
        \begin{figure}
            \centering
            \input{images/bludiste-manhattan.tex}
        \end{figure}
        \begin{itemize}
            \item Vzdálenost ale můžeme měřit i jinak -- třeba tzn. \markdarkred{manhattanskou vzdáleností}:
            \[\psi_m(x,y)=|x_g-x|+|y_g-y|.\]
        \end{itemize}
    \end{frame}

    \begin{frame}{Idea heuristiké funkce}
        \begin{itemize}
            \item Předpokládejme, že máme libovolný \textbf{nezáporně ohodnocený graf $G=(V,E,w)$} a libovolný vrchol $v\in V$ a cílový vrchol $g\in V$.
            \item Zavedeme si tzv. \markdarkred{heuristickou funkci} $\mapping{\psi}{V}{\R_0^{+}}$ (nebo zkráceně jen \markdarkred{heuristika}).
            \item $\psi(v)$ = odhad délky cesty z vrcholu $v$ do cíle $g$.
        \end{itemize}
        \begin{tnoticeframe}[0.8\textwidth]
            \begin{center}
                Pro cílový vrchol $g$ předpokládáme, že $\psi(g)=0$.
            \end{center}
        \end{tnoticeframe}
    \end{frame}

    \begin{frame}{Heuristická funkce}
        \begin{defblock}{Heuristická funkce}
            Nechť $G=(V,E,w)$ je libovolný ohodnocený graf. Heuristická funkce je zobrazení $\mapping{f}{V}{\R_0^{+}}$, která je \markdarkred{přípustná} a \markdarkred{konzistentní}.
        \end{defblock}
    \end{frame}
