    \section{Heuristická funkce}

    \titleframe{Heuristická funkce}

    \begin{frame}{Motivace k použití heuristiky}
        \begin{itemize}
            \item Velmi často se stává, že hledáme nejkratší cestu pouze do jednoho vrcholu.
            \item Tzn. máme graf ohodnocený graf $G=(V,E,w)$, počáteční vrchol $v_0\in V$ a nově \markdarkred{cílový vrchol $g\in V$}.
            \item V \textbf{nezáporně ohodnoceném grafu} lze použít Dijkstrův algoritmus.
        \end{itemize}
        \begin{timportantframe}
            \begin{center}
                Dijkstrův algoritmus nevyužívá znalosti cílového vrcholu $\implies$ všechny vrcholy "preferuje" stejně.
            \end{center}
        \end{timportantframe}
        \begin{tnoticeframe}
            \begin{center}
                Máme-li o cílovém vrcholu nějakou dodatečnou informaci, lze ji využít k \textbf{odhadu} vzdálenosti z libovolného vrcholu $v$ do $g$.
            \end{center}
        \end{tnoticeframe}
    \end{frame}
