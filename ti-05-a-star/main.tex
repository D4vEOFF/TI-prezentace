    \section{Heuristická funkce}

    \titleframe{Heuristická funkce}

    \subsection{Motivace k použití heuristiky}

    \begin{frame}{Motivace k použití heuristiky}
        \begin{itemize}
            \visible<2->{\item Velmi často se stává, že hledáme nejkratší cestu pouze do jednoho vrcholu.}
            \visible<3->{\item Tzn. máme graf ohodnocený graf $G=(V,E,w)$, počáteční vrchol $v_0\in V$ a nově \markdarkred{cílový vrchol $g\in V$}.}
            \visible<4->{\item V \textbf{nezáporně ohodnoceném grafu} lze použít Dijkstrův algoritmus.}
        \end{itemize}
        \visible<5->{\begin{timportantframe}
            \begin{center}
                Dijkstrův algoritmus nevyužívá znalosti cílového vrcholu $\implies$ všechny vrcholy "preferuje" stejně.
            \end{center}
        \end{timportantframe}}
        \visible<6->{\begin{tnoticeframe}
            \begin{center}
                Máme-li o cílovém vrcholu nějakou dodatečnou informaci, lze ji využít k \textbf{odhadu} vzdálenosti z libovolného vrcholu $v$ do $g$.
            \end{center}
        \end{tnoticeframe}}
    \end{frame}

    \subsection{Idea heuristiké funkce}

    \begin{frame}{Idea heuristiké funkce}
        \begin{itemize}
            \visible<2->{\item Předpokládejme, že máme libovolný \textbf{nezáporně ohodnocený graf $G=(V,E,w)$} a libovolný vrchol $v\in V$ a cílový vrchol $g\in V$.}
            \visible<3->{\item Zavedeme si tzv. \markdarkred{heuristickou funkci} $\mapping{\psi}{V}{\R_0^{+}}$ (nebo zkráceně jen \markdarkred{heuristika}).}
            \visible<4->{\item $\psi(v)$ = odhad délky cesty z vrcholu $v$ do cíle $g$.}
        \end{itemize}
        \visible<5->{\begin{tnoticeframe}[0.8\textwidth]
            \begin{center}
                Pro cílový vrchol $g$ předpokládáme, že $\psi(g)=0$.
            \end{center}
        \end{tnoticeframe}}
    \end{frame}

    \subsection{Požadavky na heuristiku}

    \begin{frame}{Co musí heuritika splňovat?}
        \begin{itemize}
            \visible<2->{\item \textbf{Přípustnost}}
            \begin{itemize}
                \visible<3->{\item Heuristika $\psi$ \markdarkred{nesmí nadhodnocovat} reálnou vzdálenost cesty do cílového vrcholu.}
                \visible<4->{\item Symbolicky to znamená, že je-li $v$ aktuální vrchol a $g$ cílový vrchol, pak
                \[\psi(s)\leqslant d(v,s).\]}
            \end{itemize}
            \visible<5->{\item \textbf{Konzistence}}
            \begin{itemize}
                \visible<6->{\item Mějme dvojici sousedních vrcholů $u$ a $v$ propojených hranou $uv$. Pak musí platit
                \[\psi(u)\leqslant w(u,v)+\psi(v).\]}
            \end{itemize}
        \end{itemize}
    \end{frame}

    \subsection{Definice heuristiky}

    \begin{frame}{Heuristická funkce}
        \visible<2->{\begin{defblock}
            Nechť $G=(V,E,w)$ je libovolný ohodnocený graf a $g\in V$ cílový vrchol. Heuristická funkce je zobrazení $\mapping{\psi}{V}{\R_0^{+}}$, která je \markdarkred{přípustná}, \markdarkred{konzistentní} a navíc $\psi(g)=0$.
        \end{defblock}}
        \visible<3->{\begin{talertframe}[0.8\textwidth]
            Speciálně, když položíme $\psi(v)=0$ pro každý vrchol $v\in V$, tak dostaneme původní Dijkstrův algoritmus.
        \end{talertframe}}
    \end{frame}

    \section{Algoritmus \Astar}

    \titleframe{Algoritmus \Astar}

    \subsection{Idea algoritmu}

    \begin{frame}{Idea algoritmu}
        \begin{itemize}
            \item Po vzoru Dijkstrova algoritmu budeme používat prioritní frontu pro následné vybírání vrcholů.
            \begin{timportantframe}[0.8\textwidth]
                \begin{center}
                    Vrcholy však nebudeme vybírat podle hodnoty $D$, ale podle tzv. \textbf{$f$-skóre}.
                \end{center}
            \end{timportantframe}
            \item Pro vrchol $v$ definujeme jeho $f$-skóre následovně:
            \[f(v)=D(v)+\psi(v).\]
        \end{itemize}
    \end{frame}

    \subsection{Postup algoritmu \Astar}

    \begin{frame}{Postup algoritmu \Astar}
        \begin{itemize}
            \visible<2->{\item Opět pracujeme se vzdálenostmi $D$.}
            \visible<3->{\item A dále si nově musíme uchovávat i $f$-skóre všech vrcholů.}
        \end{itemize}
        \begin{enumerate}
            \visible<4->{\item Inicializujeme hodnoty vzdáleností $D$ a $f$-skóre všech vrcholů na $\infty$ a stavy vrcholů označíme jako \textbf{nenalezené}.}
            \visible<5->{\item \textbf{Otevřeme} počáteční vrchol $v_0$ a spočítáme $D(v_0),f(v_0)$.}
            \visible<6->{\item Vybereme vrchol $v$ s nejmenším $f$-skóre (pokud nějaké existují). Pokud se jedná o cílový vrchol $g$, tak skončíme.}
            \visible<7->{\item Pro každého souseda $s$ vrcholu $v$ se podíváme, zda jsme do něj nenašli kratší cestu. Pokud ano, \textbf{otevřeme} vrchol $s$ a aktualizujeme hodnoty $D(s)$, $f(s)$.}
            \visible<8->{\item \textbf{Uzavíráme} vrchol $v$, vracíme se ke kroku 3 a postup opakujeme.}
        \end{enumerate}
    \end{frame}

    \subsection{Pseudokód algoritmu \Astar}

    \begin{frame}{Pseudokód algoritmu \Astar}
        \begin{center}
            \begin{minipage}{0.9\textwidth}
                \begin{algorithm}[H]
                    \caption{\Astar}
                    \KwIn{Nezáporně ohodnocený graf $G=(V,E)$ a počáteční a cílový vrchol $v_0,g\in V$}
                    % \tcp{Inicializace}
                    \ForEach{\textup{vrchol $v\in V$}}{
                        $stav(v)\gets\textit{nenalezený},\, D(v)\gets\infty,\,f(v)\gets\infty$
                    }
                    $stav(v_0)\gets\textit{otevřený},\, D(v_0)\gets 0,\,f(v_0)\gets\psi(v_0)$\\
                    \While{\textup{existují otevřené vrcholy}}{
                        $v\gets$ otevřený vrchol s~nejmenším $f$-skóre\\
                        \lIf{$v=g$}{\Return{$D(v_G)$}}
                        \ForEach{\textup{soused $s$ vrcholu $v$}}{
                            % \tcp{Našli jsme lepší cestu}
                            \If{$D(v)+w(v,w)<D(s)$}{
                                $D(s)\gets D(v)+w(v,s)$\\
                                $f(s)\gets D(v)+w(v,s)+\psi(s)$\\
                                $stav(s)\gets\textit{otevřený}$
                            }
                        }
                        $stav(v)\gets\textit{uzavřený}$
                    }
                \end{algorithm}
            \end{minipage}
        \end{center}
    \end{frame}

    \subsection{Správnost algoritmu \Astar}

    \begin{frame}{Správnost algoritmu \Astar}
        \visible<2->{\begin{propblock}
            Když algoritmus \Astar otevírá vrchol $v$, pak platí, že $D(v)=d(v,v_0)$, je-li heuristika $\psi$ konzistentní.
        \end{propblock}}
        \begin{itemize}
            \visible<3->{\item Mějme libovolný nezáporně ohodnocený graf $G=(V,E,w)$ a konzistentní heuristiku $\mapping{\psi}{V}{\R_0^{+}}$}
            \visible<4->{\item Předpokládejme pro \textbf{spor}, že naopak algoritmus vrchol $v$, pro který platí
            \[d(v_0,v)<D(v)\]
            $\implies$ $D(v)$ neodpovídá délce nejkratší cesty do $v$.}
            \visible<5->{\item Pak optimální cesta $v$ musí vést přes nějaký jeho sousední vrchol $v^{\prime}$, tj.
            \[D(v)>D(v^{\prime})+w(v,v^{\prime})=d(v_0,v).\]}
            \visible<6->{\item Podobně jako u Dijkstrova algoritmu ukážeme, že \Astar by měl správně upřednostnit jiný vrchol než $v$.}
        \end{itemize}
        % \begin{propblock}
        %     Nechť je dán nezáporně ohodnocený graf $G=(V,E,w)$, počáteční vrchol $v_0\in V$, cílový vrchol $g\in V$ a~libovolný vrchol $v\in V$. Pak na konci výpočtu \Astar platí $D(v)=d(v,v_0)$. Speciálně $D(g)=d(v_0,g)$.
        % \end{propblock}
    \end{frame}

    \begin{frame}{Správnost algoritmu \Astar}
        \begin{columns}
            \column{40mm}
            \visible<2->{\begin{figure}
                \centering
                \input{images/spravnost-astar.tex}
            \end{figure}}
            
            \column{\dimexpr\linewidth-40mm-5mm}
            \begin{itemize}
                \visible<3->{\item Protože heuristika $\psi$ je konzistentní, musí platit
                \[\psi(v^{\prime})\leqslant w(v,v^{\prime})+\psi(v).\]}
                \visible<4->{\item Pro $f(v^{\prime})$ tedy platí následující:
                \begin{align*}
                    f(v^{\prime})&=D(v^{\prime})+\psi(v^{\prime})\leqslant D(v^{\prime})+w(v,v^{\prime})+\psi(v) \\
                    &=D(v)+\psi(v)=f(v).
                \end{align*}}
                \visible<5->{\item Celkově $f(v^{\prime})<f(v)$.}
                \visible<6->{\item To však znamená, že algoritmus by upřednostnil buď $v^{\prime}$, a nebo jiný vrchol různý od $v$, což je \markdarkred{spor}.}
            \end{itemize}
        \end{columns}
    \end{frame}
