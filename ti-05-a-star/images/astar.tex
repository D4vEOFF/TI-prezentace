\tikzset{
    astarmazevertex/.style={
        rectangle,
        draw=black,
        thick,
        minimum size=8.5mm,
        inner sep=0pt,
        font=\scriptsize,
        align=center
    },
    astarmazewall/.style={
        rectangle,
        draw=black,
        thick,
        fill=black,
        minimum size=8.5mm,
        inner sep=0pt
    },
    astarmazeedge/.style={
        very thick,
        draw=astartree
    },
    astarmazeinfo/.style={
        draw=black,
        rounded corners=2mm,
        fill=white,
        inner sep=2mm,
        font=\small
    }
}

% obecná buňka
% #1 = x
% #2 = y
% #3 = výplň
% #4 = obsah
\newcommand{\astarcell}[4]{%
    \node[astarmazevertex, fill=#3] at (#1,#2) {#4};
}

% zeď
\newcommand{\astarwall}[2]{%
    \node[astarmazewall] at (#1,#2) {};
}

% zvýrazněná hrana aktuálního stromu
\newcommand{\astargridedge}[4]{%
    \draw[astarmazeedge] (#1,#2) -- (#3,#4);
}

% běžná označená buňka s g|f
% #1 = x, #2 = y, #3 = barva, #4 = text nahoře, #5 = g, #6 = f
\newcommand{\astarstatecell}[6]{%
    \astarcell{#1}{#2}{#3}{#4\\[-1.0mm]\tiny $#5\,|\,#6$}
}

% neoznačený vrchol jen s g|f
% #1 = x, #2 = y, #3 = barva, #4 = g, #5 = f
\newcommand{\astarplaincell}[5]{%
    \astarcell{#1}{#2}{#3}{\tiny $#4\,|\,#5$}
}

% jedna scéna animace
% #1 = titulek
% #2 = obsah open množiny
% #3 = zvýrazněné hrany
% #4 = barevně zvýrazněné buňky
\newcommand{\AStarMazeState}[4]{%
\begin{tikzpicture}[x=8.5mm,y=8.5mm]

    % -------------------------------------------------
    % základní mřížka 6 x 5
    % -------------------------------------------------
    \foreach \x in {0,...,5}{
        \foreach \y in {0,...,4}{
            \node[astarmazevertex, fill=astarunknown] at (\x,\y) {};
        }
    }

    % zdi
    \astarwall{0}{1}
    \astarwall{1}{1}
    \astarwall{3}{0}
    \astarwall{3}{1}
    \astarwall{1}{3}
    \astarwall{2}{3}
    \astarwall{3}{3}
    \astarwall{4}{3}

    % souřadnicové popisky
    \foreach \x in {0,...,5}{
        \node[font=\scriptsize] at (\x,-0.75) {$\x$};
    }
    \foreach \y in {0,...,4}{
        \node[font=\scriptsize] at (-0.75,\y) {$\y$};
    }

    % titulek
    \node[anchor=west, font=\bfseries] at (-0.8,5.1) {#1};

    % info box
    \node[
        astarmazeinfo,
        anchor=north west,
        text width=5.2cm,
        align=left
    ] at (6.2,4.7) {%
        \textbf{Start:} $s=(0,0)$\\
        \textbf{Cíl:} $g=(5,4)$\\[1mm]
        \textbf{Heuristika:}\\
        $\psi_m(x,y)=|x-5|+|y-4|$\\[1mm]
        \textbf{V buňce:} $D\,|\, f$\\[1mm]
        \textbf{Otevřené:}\\
        #2
    };

    % aktuální strom
    #3

    % zvýrazněné buňky
    #4

\end{tikzpicture}%
}
