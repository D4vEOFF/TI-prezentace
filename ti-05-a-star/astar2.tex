% vrchol s popiskem a dvojicí D|f
% #1 = jméno souřadnice
% #2 = barva výplně
% #3 = text vrcholu
% #4 = D(v)
% #5 = f(v)
\newcommand{\astargraphnode}[5]{%
    \node[astarvertex, fill=#2] at (#1) {$#3$\\[-0.8mm]\tiny $#4\,|\,#5$};
}
