\begin{frame}{Substituční metoda}
    \pause
    \begin{itemize}
        \item Mějme abecedu $\Sigma = \set{a_0,a_1,\ldots,a_{q-1}}$.\pause
        \item Zvolíme tajný posun \(0<k<q\).\pause
        \item Znak \(a_i\) zašifrujeme jako $a_i \mapsto a_{(i+k)\bmod q}$.\pause

        \[
            s(w)=s(a_1\dots a_n)=a_{(1+k)\bmod q}\dots a_{(n+k)\bmod q}.

    \end{itemize}

\subsection{Jednorázová šifra}

\begin{frame}{Jednorázová šifra}

    \begin{itemize}
        \item Anglicky \emph{one-time pad}.
        \item Alice i~Bob sdílí náhodný tajný klíč $k \in \set{0,1}^n$. Zprávu $m \in \set{0,1}^n$ Alice zašifruje pomocí operace XOR:
        \[
            c = m \oplus k.
        \]
        \item Bob zprávu \markdarkred{dešifruje stejným klíčem}:
        \[
            c \oplus k=(m \oplus k)\oplus k=m.
        \]
        \item Použita během 2. světové války, studené války, \dots
    \end{itemize}

\end{frame}

\begin{frame}{Bezpečnost jednorázové šifry}

    Jednorázová šifra splňuje:
    \[
        \Pr(M=m \mid C=c)=\Pr(M=m).
    \]
    Zachycený šifrový text tedy útočníkovi neposkytuje žádnou informaci
    o původní zprávě.

    \vspace{0.25cm}

    \begin{tnoticeframe}
        Pro každou zprávu \(m\) a každý šifrový text \(c\) existuje právě jeden klíč:
        \[
            k = m \oplus c.
        \]

        Protože \(k\) je vybrán rovnoměrně náhodně z \(2^n\) možností,
        \[
            \Pr(C=c \mid M=m)
            =
            \Pr(K=m\oplus c)
            =
            \frac{1}{2^n}.
        \]
    \end{tnoticeframe}

    \vspace{0.15cm}

    \begin{talertbox}{Důležité podmínky}
        Klíč musí být náhodný, stejně dlouhý jako zpráva a~použitý pouze jednou.
    \end{talertbox}

\end{frame}
