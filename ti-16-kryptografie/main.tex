\subsection{Perfektní schéma symetrického šifrování}

\begin{frame}{Symetrická kryptografie}

    \visible<2->{Alice i~Bob používají stejný tajný klíč $k\in\set{0,1}^{n}$.
    \[
        c = \mathbf{E}(m, k),
        \qquad
        m = \mathbf{D}(c, k).
    \]}

    \visible<3->{Tzn. klíč $k$ lze použít k~\markdarkred{šifrování i~dešifrování}.}

\end{frame}

\begin{frame}{Perfektní schéma}
    \obrazek[0.4\textheight]{images/perfektni-schema.tex}
\end{frame}

\subsection{Jednorázová šifra}

\begin{frame}{Jednorázová šifra}

    \visible<2->{\begin{itemize}
        \item Anglicky \emph{one-time pad}.
        \visible<3->{\item Alice i~Bob sdílí náhodný tajný klíč $k \in \set{0,1}^n$. Zprávu $m \in \set{0,1}^n$ Alice zašifruje pomocí operace XOR:
        \[
            c = m \oplus k.
        \]}

        \visible<4->{\item Bob zprávu \markdarkred{dešifruje stejným klíčem}:
        \[
            c \oplus k=(m \oplus k)\oplus k=m.
        \]}

        \visible<5->{\item Použita během 2. světové války, studené války, \dots}
    \end{itemize}}

\end{frame}

\begin{frame}{Schéma jednorázové šifry}

    \obrazek[0.4\textheight]{images/schema-jednorazove-sifry.tex}

\end{frame}

\begin{frame}{Bezpečnost jednorázové šifry}

    \visible<2->{\begin{itemize}
        \item Jednorázová šifra splňuje:
        \[
            \Pr(M=m \mid C=c)=\Pr(M=m).
        \]
        Zachycený šifrový text tedy útočníkovi neposkytuje žádnou informaci
        o~původní zprávě.

        \visible<3->{\item Pro každou zprávu \(m\) a~každý šifrový text \(c\) existuje právě jeden klíč $k = m \oplus c$.}
        \visible<4->{\item Protože \(k\) je vybrán rovnoměrně náhodně z~\(2^n\) možností,
        \[
            \Pr(C=c \mid M=m)=\Pr(K=m\oplus c)=\frac{1}{2^n}.
        \]
        Výsledná pravděpodobnost je \markdarkred{nezávislá na původní zprávě $m$}.}
    \end{itemize}}

    \visible<5->{\begin{talertbox}{Důležité podmínky}
        Klíč $k$ musí být \textbf{náhodný}, \textbf{stejně dlouhý} jako zpráva $m$ a~\textbf{použitý pouze jednou}.
    \end{talertbox}}

\end{frame}

\begin{frame}{Proč se klíč nesmí použít vícekrát?}

    \visible<2->{\begin{itemize}
        \item Nechť Alice zašifruje dvě zprávy stejným klíčem \(k\):
        \[
            c_1 = m_1 \oplus k,
            \qquad
            c_2 = m_2 \oplus k.
        \]

        \visible<3->{\item Útočník může spočítat:
        \[
            c_1 \oplus c_2
            =
            (m_1 \oplus k)\oplus(m_2 \oplus k)
            =
            m_1 \oplus m_2.
        \]
        Klíč se tedy vyruší.}
    \end{itemize}}

    \visible<4->{\begin{timportantbox}{Důsledek}
        Útočník nezíská přímo \(m_1\) a~\(m_2\), ale získá informaci
        o~ jejich vzájemném vztahu. Pokud jednu zprávu zná nebo odhadne,
        může snadno získat i~druhou.
    \end{timportantbox}}

\end{frame}

\begin{frame}{Nevýhoda jednorázové šifry}

    \visible<2->{\begin{talertframe}
        Klíč $k$ musí být stejně dlouhý jako zpráva $m$. Bezpečně předat klíč délky \(n\) může být stejně obtížné
        jako bezpečně předat samotnou zprávu délky \(n\).
    \end{talertframe}}

    \visible<3->{To představuje problém, pokud klíč $k$ nemůžeme druhé straně předat bezpečně.}
\end{frame}

\subsection{Základní princip}

\begin{frame}{Algoritmus RSA}

    \visible<2->{Zkratka \textbf{RSA} pochází ze jmen autorů Ron \textbf{R}ivest, Adi \textbf{S}hamir a~Leonard \textbf{A}dleman.}

    \visible<3->{\begin{enumerate}
        \item Vygeneruj \markdarkred{dvě velká prvočísla} $p,q$ a~zvol \markdarkred{soukromý exponent} $s$ nesoudělný s~$(p-1)(q-1)$.
        \visible<4->{\item Spočítáme $n=p\cdot q$.}
        \visible<5->{\item Spočítáme číslo $v$ tak, aby
        \[v\cdot s\bmod (p-1)(q-1)=1.\]}
        \visible<6->{\item Položíme veřejný a~soukromý klíč rovny uspořádaným dvojicím
        \[k_V=(v,n)\quad\text{a}\quad k_S=(s,n).\]}
        \visible<7->{\item Šifrování a~dešifrování pro $x<n$:
        \[\markdarkred{y=}\;\mathbf{E}(x, k_V)=x^{v}\bmod n\quad\text{a}\quad\mathbf{D}(y, k_S)=y^{s}\bmod n.\]}
    \end{enumerate}}
\end{frame}

\begin{frame}{Schéma RSA}
    \obrazek[0.4\textheight]{images/schema-rsa.tex}
\end{frame}

\subsection{Malá Fermatova věta}

\begin{frame}{Malá Fermatova věta}

    \visible<2->{\begin{thmblock}[Malá Fermatova věta]
        Nechť \(p\) je prvočíslo a~nechť \(a\in\mathbb{Z}\) není dělitelné číslem \(p\). Potom platí
        \[
            a^{p-1} \bmod p = 1.
        \]
    \end{thmblock}

    \vspace{0.25cm}}

    \visible<3->{\begin{tnoticeframe}
        Ekvivalentně lze psát, že pro každé \(a\in\mathbb{Z}\) platí
        \[
            a^{p} \bmod p = a \bmod p.
        \]
    \end{tnoticeframe}}

\end{frame}

\subsection{Čínská věta o~zbytcích}

\begin{frame}{Čínská věta o~zbytcích}

    \visible<2->{\begin{thmblock}[Čínská věta o~zbytcích]
        Nechť \(m_1,m_2,\ldots,m_k\in\N\) jsou po dvou nesoudělná.

        Potom pro libovolná celá čísla \(a_1,a_2,\ldots,a_k\) má soustava kongruencí
        \begin{align*}
            x\bmod m_1 &= a_1,\\
            x\bmod m_2 &= a_2,\\
            &\vdots\\
            x\bmod m_k &= a_k
        \end{align*}
        právě jedno řešení.
    \end{thmblock}}

    \visible<3->{\begin{tnoticeframe}
        Toto řešení je určeno jednoznačně modulo $M=m_1m_2\cdots m_k$.
    \end{tnoticeframe}}

\end{frame}

\begin{frame}{Důsledek čínské věty o~zbytcích}

    \visible<2->{\begin{corblock}
        Nechť $m_1,m_2\in\N$ jsou nesoudělná. Pak pro každé $x,a\in\N$ platí
        \[\left.\begin{aligned}
            x\bmod m_1&=a\\
            x\bmod m_2&=a
        \end{aligned}\right\}\iff x\bmod m_1m_2=a.\]
    \end{corblock}}
\end{frame}

\subsection{Korektnost RSA}

\begin{frame}{Korektnost RSA}
    \footnotesize

    \visible<2->{Chceme ukázat, že $\mathbf{D}(\mathbf{E}(x, k_V), k_S)=x$, neboli že
    \[\left(x^{v}\bmod n\right)^{s}\bmod n=x^{vs}\bmod n=x.\]}

    \visible<3->{Důkaz rozdělíme na dva případy:
    \begin{enumerate}

        \visible<4->{\item \textbf{$p$ dělí číslo $x$.} Pak platí
        \[x^{vs}\bmod p = 0 = x\bmod p.\]}
        \visible<5->{\item \textbf{$p$ nedělí číslo $x$.} Protože platí $vs\bmod(p-1)(q-1)=1$, existuje $t\in\N$ takové, že
        \[vs=1+t(p-1)(q-1).\]}
        \visible<6->{Pak dostáváme
        \begin{align*}
            x^{vs}\bmod p&=x^{1+t(p-1)(q-1)}\bmod p\stackrel{\text{MFV}}{=}x(x^{p-1})^{t(q-1)}\bmod p\\
            &=x\cdot 1^{t(q-1)}\bmod p=x\bmod p.
        \end{align*}}

        \visible<7->{Podobně pro $q$ dostaneme, že též platí $x^{vs}\bmod q=x\bmod q$.}
    \end{enumerate}}

\end{frame}

\begin{frame}{Korektnost RSA}

    \visible<2->{Tedy víme, že platí
    \begin{align*}
        x\bmod p&=x^{vs}\bmod p\\
        x\bmod q&=x^{vs}\bmod q.
    \end{align*}}

    \visible<3->{Podle \markdarkred{důsledku čínské věty o~zbytcích} platí
    \[x^{vs}\bmod pq=x\bmod pq=x\bmod n\stackrel{x<n}{=}x.\]}
\end{frame}

\begin{frame}{Příklad}{Algoritmus RSA pro malá prvočísla}

    \begin{itemize}

        \visible<2->{\item Volme prvočísla $p=5$ a~$q=11$ a~exponent
        \[
        s=27,\quad\text{který je nesoudělný s~}(p-1)(q-1)=40.
        \]}

        \visible<3->{\item Pak $n=pq=55$. Spočítáme hodnotu $v$ tak, aby platilo
        \[vs\bmod (p-1)(q-1)=27v\bmod 40=1.\]
        Tuto rovnost splňuje např. $v=3$.}

        \visible<4->{\item Máme tedy veřejný klíč $k_V=(3, 55)$ a~soukromý klíč $k_S=(27,55)$.}

        \visible<5->{\item Alice zašifruje zprávu \(x=7\):
        \[
        c=x^v \bmod n = 7^3 \bmod 55 = 343 \bmod 55 = 13.
        \]
        Bob dešifruje:
        \[
        x=c^s \bmod n = 13^{27}\bmod 55 = 7.
        \]}
    \end{itemize}

\end{frame}

\subsection{Bezpečnost RSA}

\begin{frame}{Bezpečnost RSA}

    \visible<2->{U~RSA je veřejný klíč $ k_V=(v,n)$, kde
    \[
        n=pq
    \]
    je součin dvou \markdarkred{velkých prvočísel}.}

    \visible<3->{\vspace{0.2cm}

    \begin{talertframe}
        Pokud útočník zná rozklad
        \[
            n=pq,
        \]
        může z~rovnosti
        \[vs\bmod(p-1)(q-1)=1\]
        snadno dopočítat $s$.
    \end{talertframe}}
\end{frame}

\begin{frame}{Faktorizace}
    \small

    \visible<2->{\begin{talertbox}{Základní předpoklad bezpečnosti}
        Je snadné spočítat součin \(n=pq\), ale je těžké z~čísla \(n\)
        zpětně zjistit prvočísla \(p\) a~\(q\).
    \end{talertbox}}

    \visible<3->{\decproblem{Faktorizace ($\FACTOR$)}{
        Přirozená čísla $n,b\in\N,b\leqslant n$.
    }{
        Existuje $d\in\N$ takové, že $1<d<b$ a~$d$ dělí $n$?
    }}

    \visible<4->{Protože pracujeme s~binárními hodnotami $n$, platí
    \[|\binary{n}|=\lceil\log_2{n}\rceil.\]}

    \visible<5->{Naivní algoritmus by musel mít složitost $2^{|\binary{n}|}$.}
\end{frame}

\begin{frame}{Problém $\FACTOR$ patří do $\classNP$}

    \visible<2->{\begin{propblock}
        Platí $\FACTOR\in\classNP$.
    \end{propblock}}

    \visible<3->{\begin{itemize}
        \item Na vstupu mějme dvojici $(n,b)$ a~číslo $d\in\N$.
        \visible<4->{\item Celková délka vstupu je $L_n=|\binary{n}|+|\binary{b}|$.}
        \visible<5->{\item Mějme algoritmus níže.}
        \visible<6->{\begin{center}
            \begin{minipage}{0.7\textwidth}
                \begin{algorithm}[H]
                    \caption{Verifikátor}
                    \If{\textup{$d\leqslant 1\lor d\geqslant b\lor n\bmod d\neq 0$}}{
                        \Return{\textup{\textbf{false}}}
                    }
                    \Return{\textup{\textbf{true}}}
                \end{algorithm}
            \end{minipage}
        \end{center}}

        \visible<7->{\item Porovnání binárně zapsaných čísel i~výpočet zbytku po dělení lze provést v~čase polynomiálním vzhledem k~počtu bitů vstupu $L_n$.}
        \visible<8->{\item Ověřovač tedy běží v~polynomiálním čase.}
    \end{itemize}}

\end{frame}

\begin{frame}{Co víme o~faktorizaci?}

    \visible<2->{\begin{tnoticeframe}
        \begin{center}
            Víme, že $\FACTOR\in\classNP$.
        \end{center}
    \end{tnoticeframe}}

    \visible<3->{\begin{talertbox}{Otevřený problém}
        \begin{center}
            Není známo, zdali $\FACTOR$ je $\classNP$-úplný problém nebo $\FACTOR\in\classP$.
        \end{center}
    \end{talertbox}}

    \visible<4->{Kdyby $\FACTOR\in\classP$, pak ze znalosti čísla $n$ bychom mohli nalézt prvočísla $p,q$ a~odtud odvodit $s$.}
\end{frame}

\section{Pseudonáhodné generátory a~jednosměrné funkce}
% =================================================================

\begin{titleframeenv}{\huge Pseudonáhodné generátory\\ a~jednosměrné funkce}

\end{titleframeenv}

\subsection{Pseudonáhodné generátory}

\begin{frame}{Pseudonáhodný generátor}

    \visible<2->{Funkce
    \[
        G:\set{0,1}^n \to \set{0,1}^{\ell(n)}
    \]
    je \markdarkred{pseudonáhodný generátor (PRG)}, pokud:}
    \visible<3->{\begin{itemize}
        \item \(G\) je vyčíslitelná deterministickým polynomiálním algoritmem,
        \visible<4->{\item \(\ell(n)>n\) pro každé \(n\in\mathbb{N}\) (\markdarkred{stretch}),}
        \visible<5->{\item žádný \markdarkred{pravděpodobnostní polynomiální algoritmus $\mathcal{A}$} neumí "rozlišit"
        \(G(y_n)\) od \(y_{\ell(n)}\), kromě zanedbatelné chyby.}
    \end{itemize}}
    \visible<6->{Vstup \(s\in\{0,1\}^n\), tzv. \markdarkred{seed}, se "natáhne" na delší řetězec \(G(s)\).}

\end{frame}

\subsection{Nerozlišitelnost výstupu PRG}

\begin{frame}{Nerozlišitelnost výstupu PRG}
    \small

    \visible<2->{\begin{tinfobox}{Třetí podmínka}
        Pro každý pravděpodobnostní polynomiální algoritmus \(\mathcal{A}\) platí:
        \[
        \left|
        \Pr_{y\in\{0,1\}^{\ell(n)}}[\mathcal{A}(y)=1]
        -
        \Pr_{y\in\{0,1\}^{n}}[\mathcal{A}(G(y))=1]
        \right|
        \leqslant \varepsilon(n),
        \]
        kde \(\varepsilon(n)\) je \textbf{zanedbatelná} funkce, tj. pro každé
        \(k\in\mathbb{N}\) existuje \(n_k\) takové, že pro každé \(n>n_k\) platí \(\varepsilon(n) < 1/n^k\).
    \end{tinfobox}}

    \visible<3->{\begin{tnoticeframe}
        Algoritmus \(\mathcal{A}\) dostane řetězec délky \(\ell(n)\) a~snaží se
        rozhodnout, zda je skutečně náhodný, nebo vznikl jako výstup \(G\).
    \end{tnoticeframe}}

\end{frame}

\subsection{Existence PRG, když P=NP}

\begin{frame}{Žádné PRG, pokud $\classP=\classNP$}
    \small

    \visible<2->{\begin{timportantframe}
        Pokud by platilo $\classP=\classNP$, pak by žádné pseudonáhodné generátory neexistovaly.
    \end{timportantframe}

    \vspace{0.2cm}}

    \visible<3->{\textbf{Důvod:} Pro daný výstup \(y\in\set{0,1}^{n}\) bychom uměli efektivně rozhodnout, zda existuje seed \(s\) takový, že
    \[
        G(s)=y.
    \]}

    \visible<4->{\begin{itemize}
        \item Uvažujme jazyk
        \[
            L_G =\set{y : \exists s\in\{0,1\}^n: G(s)=y}.
        \]
        Tento jazyk patří do \(\classNP\), protože certifikátem (nápovědou) je seed~\(s\).
        \visible<5->{\item Rozlišovač \(\mapping{\mathcal{D}}{\set{0,1}^{\ell(n)}}{\set{0,1}}\) se vstupem \(y\) udělá:
        \[
            \mathcal{D}(y)=1
            \quad\Longleftrightarrow\quad
            y\in L_G.
        \]}
    \end{itemize}}

\end{frame}

\begin{frame}{Žádné PRG, pokud $\classP=\classNP$}

    \visible<2->{\begin{itemize}
        \item Pokud \(y=G(s)\), pak vždy platí $\mathcal{D}(y)=1$ a~tedy
        \[\Pr[\mathcal{D}(y)=1]=1.\]
        \visible<3->{\item Pokud je \(y\) skutečně náhodný řetězec délky \(\ell(n)\), pak pravděpodobnost, že leží v~\(L_G\), je nejvýše
        \[
            \frac{2^n}{2^{\ell(n)}}=2^{n-\ell(n)}.
        \]}
        \visible<4->{\item Pak dostáváme
        \begin{align*}
             &\left|\Pr_{y\in\{0,1\}^{\ell(n)}}[\mathcal{D}(y)=1]-\Pr_{y\in\{0,1\}^{n}}[\mathcal{D}(G(y))=1]\right|\\
             &=\left|2^{n-\ell(n)}-1\right|=1-2^{n-\ell(n)}\geqslant 1-\frac{1}{2}=\frac{1}{2},
        \end{align*}
        což není zanedbatelná funkce.}
    \end{itemize}}
\end{frame}

\subsection{Jednosměrné funkce}

\begin{frame}{Jednosměrná funkce}

    \visible<2->{\[
        \mapping{f}{\set{0,1}^{*}}{\set{0,1}^{*}}
    \]
    je \markdarkred{jednosměrná funkce (OWF)}, pokud}
    \visible<3->{\begin{itemize}
        \item $f(x)$ lze spočítat v~polynomiálním čase pro každé $x\in\set{0,1}^{*}$ a
        \visible<4->{\item je \markdarkred{"těžko" invertovatelná}.}
    \end{itemize}}

    \visible<5->{Druhá podmínka formálně říká, že pro každý pravděpodobnostní polynomiální algoritmus $\mathcal{A}$ existuje zanedbatelná funkce $\varepsilon(n)$ taková, že pro každé $n\in\N$ 
    \[\Pr_{x\in\set{0,1}^{n}}\left[f(\mathcal{A}(f(x)))=f(x)\right]\leqslant\varepsilon(n).\]}

    \visible<6->{\begin{timportantframe}
        Když náhodně vybereme $x$ a~útočníkovi dáme pouze $y=f(x)$, pak má \textbf{zanedbatelně malou šanci}, že najde nějaký vzor~$x'$ takový, že $f(x')=y$.
    \end{timportantframe}}

\end{frame}

\begin{frame}{Idea jednosměrné funkce}

    \obrazek[0.4\textheight]{images/idea-jednosmerne-funkce.tex}

\end{frame}

\begin{frame}{Možní kandidáti na jednosměrné funkce}

    \visible<2->{\begin{talertframe}
        Dodnes není známé, zdali nějaké jednosměrné funkce existují.
    \end{talertframe}}

    \visible<3->{Existují kandidáti na jednosměrné funkce:}

    \visible<4->{\begin{itemize}
        \item Násobení prvočísel: $f(p,q)=pq$.
        \visible<5->{\item Součet podmnožiny:
        \[f(x_1,\dots,x_n,J)=\left(x_1,\dots,x_n,\left(\sum_{j\in J}x_j\right)\bmod 2^{n}\right),\]
        kde $x_1,\dots,x_n\in\set{0,1}^{n}$ a~$J\subseteq\set{1,\dots,n}$.}
    \end{itemize}}

\end{frame}

\begin{frame}{Souvislost OWF a~PRG}

    \visible<2->{\begin{itemize}
        \item Existence jednosměrných funkcí je~úzce spojena s~existencí pseudonáhodných generátorů.
        \visible<3->{\item Ve skutečnosti platí:
        \[
            \text{existují OWF}
            \quad\Longleftrightarrow\quad
            \text{existují PRG}.
        \]}
        \visible<4->{\item Jednosměrná funkce poskytuje výpočetní těžkost. Z~této těžkosti lze konstruovat bity, které vypadají náhodně.}
    \end{itemize}}

\end{frame}

\subsection{Souvislost PRG s~kryptografií}

\begin{frame}{Souvislost s~kryptografií}

    \visible<2->{\begin{itemize}
        \item V~praxi nechceme používat opravdu náhodný klíč délky celé zprávy.
        \visible<3->{\item Místo toho použijeme krátký tajný seed \(s\) a~z něj vytvoříme dlouhý proud bitů $G(s)$.}
        \visible<4->{\item Potom šifrujeme podobně jako u~jednorázové šifry:
        \[
            c=m\oplus G(s).
        \]}
    \end{itemize}}

    \visible<5->{\vspace{0.2cm}

    \begin{talertframe}
        Bezpečnost pak nestojí na dokonalé náhodnosti, ale na tom, že
        \(G(s)\) je \textbf{nerozlišitelný od náhodného řetězce}.
    \end{talertframe}}

\end{frame}

\subsection{Definice a~vlastnosti hešovací funkce}

\begin{frame}{Hešovací funkce}

    \visible<2->{\begin{defblock}
        \markdarkred{Hešovací funkce} je jakákoliv funkce z~$\set{0,1}^{*}$ do~$\set{0,1}^{n}$. Je-li $H$ hešovací funkce a~$x\in\set{0,1}^{*}$, pak $H(x)$ se nazývá \markdarkred{hash} nebo též \markdarkred{otisk}.
    \end{defblock}}

    \visible<3->{Požadavky na hešovací funkci:}

    \visible<4->{\begin{enumerate}
        \item otisk je \textbf{rychle vyčíslitelný} a
        \visible<5->{\item malá změna vstupu má \textbf{výrazně} změnit výstup.}
    \end{enumerate}}
\end{frame}

\begin{frame}{Kolize jsou nevyhnutelné}

    \visible<2->{Funkce
    \[\mapping{H}{\set{0,1}^{*}}{\set{0,1}^{n}}\]
    uvažuje vstupy z~nekonečné množiny do množiny s~$2^{n}$ prvky.}

    \visible<3->{\begin{talertframe}
        \begin{center}
            Musí existovat dvojice $x,y\in\set{0,1}^{*}$ taková, že $H(x)=H(y)$.
        \end{center}
    \end{talertframe}}


    
    \visible<4->{\obrazek[0.4\textheight]{images/kolize-jsou-nevyhnutelne.tex}}

    \visible<5->{\textbf{Nový požadavek:} Chceme, aby bylo "výpočetně složité" nějakou kolizi najít.} 
\end{frame}

\subsection{Narozeninový paradox a~hledání kolizí}

\begin{frame}{Narozeninový paradox}
    \footnotesize

    \visible<2->{\begin{tinfobox}{Otázka}
        Jaká je pravděpodobnost, že mezi \(n\) lidmi existují alespoň dva,
        kteří mají narozeniny ve stejný den?
    \end{tinfobox}}

    \visible<3->{Nechť \(C\) značí počet dvojic lidí, které \markdarkred{mají narozeniny ve stejný den}.}

    \visible<4->{\begin{align*}
        \Pr[C>0]&=1-\Pr[C=0]\\
        \Pr[C=0]&=\frac{365}{365}\cdot\frac{364}{365}\cdot\frac{363}{365}\cdots\frac{365-n+1}{365}\\
        \Pr[C>0]&=1-\prod_{i=0}^{n-1}\frac{365-i}{365}
    \end{align*}}

    \visible<5->{\begin{tnoticeframe}
        Už pro \(n=23\) vychází pravděpodobnost přibližně \(50{,}7\ \%\).
    \end{tnoticeframe}}

\end{frame}

\begin{frame}{Kolik vzniká dvojic?}

    \obrazek[0.4\textheight]{images/kolik-vznika-dvojic.tex}

\end{frame}

\begin{frame}{Narozeninový paradox a~kolize hashů}
    \small

    \visible<2->{\begin{itemize}
        \item Uvažujme hešovací funkci $H_r:\set{0,1}^* \to \set{0,1}^r$. Možných hashů je~$2^r$.
        \visible<3->{\item Mějme celkově $k$ různých zpráv
        \[x_1,\dots,x_k\;\text{a jejich hashe}\;H_r(x_1),\dots,H_r(x_k).\]}

        \visible<4->{\item Označme si počet vzniklých kolizí opět jako $C$. Zajímá nás $\Pr[C>0]$.}
        \visible<5->{\item Postupujeme podobně:
        \begin{align*}
            \Pr[C>0]&=1-\Pr[C=0]=1-\prod_{j=1}^{k-1}\left(1-\frac{j}{2^{r}}\right)\stackrel{1-x\approx e^{-x}}{\approx}1-\prod_{j=1}^{k-1}e^{-j/2^{r}}\\
            &=1-e^{-1/2^{r}}\cdot e^{-2/2^{r}}\cdots e^{-(k-1)/2^{r}}=1-e^{-\frac{1}{2^{r}}\sum_{j=1}^{k-1}j}=1-e^{-\frac{k(k-1)}{2^{r+1}}}
        \end{align*}}
    \end{itemize}}

\end{frame}

\begin{frame}{Narozeninový paradox a~kolize hashů}
    \small

    \visible<2->{\obrazek[0.18\textheight]{images/narozeninovy-paradox-a-kolize-hashu.tex}}

    \visible<3->{\begin{itemize}
        \item Pokud budeme předpokládat, že $k$ je "velké", potom řádově porovnáme
        \[\binom{k}{2}=\frac{k(k-1)}{2}=\frac{k^{2}-k}{2}\approx \frac{k^{2}}{2}\]
        dvojic.
        \visible<4->{\item Kolize začne být pravděpodobná, pokud $\frac{k^{2}}{2}\approx 2^{r}$, neboli $k\approx 2^{r/2}$. Pak totiž
        \[\Pr[C>0]=1-\exp\left(-\frac{2^{r/2}(2^{r/2}-1)}{2^{r+1}}\right)\approx 1-e^{-\frac{2^{r}}{2^{r+1}}}=1-e^{-1/2}\approx 0{,}3935.\]}
    \end{itemize}}

    \visible<5->{\begin{talertframe}
        \begin{center}
            Kolize v~$r$-bitovém hashi lze hrubou silou hledat přibližně v~čase $2^{r/2}$.
        \end{center}
    \end{talertframe}}
\end{frame}

\subsection{Bezpečnostní vlastnosti hešovacích funkcí}

\begin{frame}{Bezpečnostní vlastnosti hešovacích funkcí}
    \small

    \visible<2->{Uvažujme hešovací funkci
    \[
        \mapping{H_n}{\set{0,1}^{*}}{\set{0,1}^{n}}
    \]}

    \visible<3->{Dále nechť $\varepsilon$ značí nějakou zanedbatelnou funkci a~$\mathcal{A}$ pravděpodobnostní polynomiální algoritmus (útočníka).}

    \visible<4->{\begin{center}
        \begin{tabular}{@{}p{0.25\textwidth}@{\hspace{0.5em}}p{0.7\textwidth}@{}}
            \markdarkred{Odolnost vůči nalezení vzoru} &
            Z~hashe \(y\) je těžké najít nějaké \(x\), pro které
            \(H_n(x)=y\), tzn. 
            \[
                \Pr_{y\in\set{0,1}^{n}}\left[
                H_n(\mathcal{A}(y))=y
                \right]\leqslant \varepsilon(n).
            \]
            pro každé $n\in\N$.\\
            \visible<5->{\markdarkred{Odolnost vůči kolizím}} &
            \visible<5->{Je těžké najít libovolnou dvojici \(x\neq y\), pro kterou $H_n(x)=H_n(y)$, tj.
            \[
                \Pr\left[
                (x,y)\gets\mathcal{A}(n):\;
                x\neq y \land H_n(x)=H_n(y)
                \right]\leqslant \varepsilon(n).
            \]
            pro každé $n\in\N$.}

        \end{tabular}
    \end{center}}

\end{frame}

% 1
\begin{frame}{Sponge function}

    \visible<2->{Sponge function (SF) pracuje s~vnitřním stavem délky
    \[
        b=\underbrace{r}_{\text{rate}}+\overbrace{c}^{\text{capacity}}.
    \]

    \vspace{0.15cm}}

    \visible<3->{Zpráva $m$ se nejprve doplní paddingem a~rozdělí na bloky délky \(r\):
    \[
        M \longrightarrow p_1p_2\ldots p_t.
    \]}

    \visible<4->{Každý blok \(p_i\) má délku \(r\) bitů.}

\end{frame}

\begin{frame}{Fáze SF}

    \visible<2->{% Fáze 1
    \obrazek[0.4\textheight]{images/faze-sf-1.tex}}

    \visible<3->{\begin{itemize}
        \item Začneme s~nulovým stavem $S_0=0^b$.
        \visible<4->{\item Pro každý blok \(p_i\) položíme
        \[
            S_i=f\left(S_{i-1}\oplus(p_i 0^c)\right),
        \]
        kde $\mapping{f}{\set{0,1}^{b}}{\set{0,1}^{b}}$ je \markdarkred{pseudonáhodná permutace}.}
    \end{itemize}}

    \visible<5->{% Fáze 2
    \obrazek[0.4\textheight]{images/faze-sf-2.tex}}

    \visible<6->{\begin{itemize}
        \item Z~výsledného stavu \(S_t\) čteme prvních \(r\) bitů.
        \visible<7->{\item Pokud potřebujeme delší výstup, opakujeme:
        \[
            S_{k+1}=f(S_k)\;,\; k>t.
        \]}
    \end{itemize}}

\end{frame}

\begin{frame}{Je protokol bezpečný?}

\begin{frame}{Proč tento protokol nestačí?}
    \small

    \visible<2->{\begin{talertbox}{Hlavní problém}
        Bob sice obdrží zprávu, kterou dokáže dešifrovat svým soukromým klíčem,
        ale nemá jistotu, že zpráva skutečně pochází od Alice.
    \end{talertbox}

    \vspace{0.2cm}}

    \visible<3->{\begin{itemize}
        \item Záškodník může zprávu zachytit, změnit připojený podpis a~vydávat se za jiného odesílatele.
        \visible<4->{\item Bob potom přečte správnou zprávu \(x\), ale mylně se domnívá, že ji poslal někdo jiný.}

    \visible<4->{\vspace{0.15cm}}

    \visible<5->{\begin{timportantbox}{Co zde chybí?}
        Podpis musí být kryptograficky "svázán" se zprávou.
        \begin{center}
            \markdarkred{Podpis závisí na obsahu zprávy i~na~soukromém klíči odesílatele.}
        \end{center}
    \end{timportantbox}}

\end{frame}

\subsection{Komutativita klíčů}

\begin{frame}{Komutativita klíčů}

    \visible<2->{\begin{itemize}
        \item Chceme používat \textbf{stejnou dvojici klíčů} jak pro šifrování, tak pro podpis.
        \visible<3->{\item Proto potřebujeme, aby platilo nejen
        \[
            \mathbf{D}\left(\mathbf{E}(x,k_V),k_S\right)=x,\quad\text{ale také}\quad\mathbf{E}\left(\mathbf{D}(x,k_S),k_V\right)=x.
        \]}
    \end{itemize}}

\begin{frame}{Podpis hashe}
    \small

    \visible<2->{Podepisovat přímo celou zprávu
    \[
        \sigma=\mathbf{D}(m,k_S^A)
    \]
    není praktické.}

    \visible<3->{\begin{talertframe}
        Zpráva může být dlouhá a~asymetrické operace jsou pomalé.
    \end{talertframe}

    \vspace{0.2cm}}

    \visible<4->{Mějme libovolnou \markdarkred{hešovací funkci $\mapping{H_n}{\set{0,1}^{*}}{\set{0,1}^{n}}$}}

    \visible<5->{\begin{itemize}
        \item Alice nejprve spočítá krátký otisk zprávy $h=H_n(m)$.
        \visible<6->{\item Podepíše až tento hash: $\sigma=\mathbf{D}\left(H_n(m),k_S^A\right)$.}
        \visible<7->{\item Bob spočítá hash přijaté zprávy a~porovná: $\mathbf{E}\left(\sigma,k_V^A\right)\stackrel{?}{=}H_n(m)$.}
        \visible<8->{\item Pokud se zpráva změní, změní se i~její hash a~podpis přestane sedět.}
    \end{itemize}}

\end{frame}
