\subsection{Jednorázová šifra}

\begin{frame}{Jednorázová šifra}
    \pause
    \begin{itemize}
        \item Anglicky \emph{one-time pad}.\pause
        \item Alice i~Bob sdílí náhodný tajný klíč $k \in \set{0,1}^n$. Zprávu $m \in \set{0,1}^n$ Alice zašifruje pomocí operace XOR:
        \[
            c = m \oplus k.
        \]
        \pause
        \item Bob zprávu \markdarkred{dešifruje stejným klíčem}:
        \[
            c \oplus k=(m \oplus k)\oplus k=m.
        \]
        \pause
        \item Použita během 2. světové války, studené války, \dots
    \end{itemize}

\end{frame}

\begin{frame}{Bezpečnost jednorázové šifry}
    \pause
    \begin{itemize}
        \item Jednorázová šifra splňuje:
        \[
            \Pr(M=m \mid C=c)=\Pr(M=m).
        \]
        Zachycený šifrový text tedy útočníkovi neposkytuje žádnou informaci
        o~původní zprávě.
        \pause
        \item Pro každou zprávu \(m\) a každý šifrový text \(c\) existuje právě jeden klíč $k = m \oplus c$.\pause
        \item Protože \(k\) je vybrán rovnoměrně náhodně z \(2^n\) možností,
        \[
            \Pr(C=c \mid M=m)=\Pr(K=m\oplus c)=\frac{1}{2^n}.
        \]
        Výsledná pravděpodobnost je \markdarkred{nezávislá na původní zprávě $m$}.
    \end{itemize}
    \pause
    \begin{talertbox}{Důležité podmínky}
        Klíč $k$ musí být \textbf{náhodný}, \textbf{stejně dlouhý} jako zpráva $m$ a~\textbf{použitý pouze jednou}.
    \end{talertbox}

\end{frame}

\begin{frame}{Proč se klíč nesmí použít vícekrát?}
    \pause
    \begin{itemize}
        \item Nechť Alice zašifruje dvě zprávy stejným klíčem \(k\):
        \[
            c_1 = m_1 \oplus k,
            \qquad
            c_2 = m_2 \oplus k.
        \]
        \pause
        \item Útočník může spočítat:
        \[
            c_1 \oplus c_2
            =
            (m_1 \oplus k)\oplus(m_2 \oplus k)
            =
            m_1 \oplus m_2.
        \]
        Klíč se tedy vyruší.
    \end{itemize}
    \pause
    \begin{timportantbox}{Důsledek}
        Útočník nezíská přímo \(m_1\) a \(m_2\), ale získá informaci
        o~ jejich vzájemném vztahu. Pokud jednu zprávu zná nebo odhadne,
        může snadno získat i druhou.
    \end{timportantbox}

\end{frame}

\begin{frame}{Nevýhoda jednorázové šifry}
    \pause
    \begin{talertframe}
        Klíč $k$ musí být stejně dlouhý jako zpráva $m$. Bezpečně předat klíč délky \(n\) může být stejně obtížné
        jako bezpečně předat samotnou zprávu délky \(n\).
    \end{talertframe}
    \pause
    To představuje problém, pokud klíč $k$ nemůžeme druhé straně předat bezpečně.
\end{frame}

\begin{frame}{Asymetrická kryptografie}
    \pause
    Každý uživatel má dvojici klíčů:
    \[
        \text{veřejný klíč}\;k_V
        \qquad
        \text{a}
        \qquad
        \text{soukromý klíč}\;k_S.
    \]
    \pause
    Veřejný klíč \markdarkred{může znát kdokoliv}. Soukromý klíč si uživatel \markdarkred{musí

    \begin{tinfobox}{Princip}
        Chce-li Alice poslat zprávu Bobovi, použije Bobův veřejný klíč:
        \[
            c = \mathbf{E}(m, k_V).
        \]
        Bob zprávu dešifruje svým soukromým klíčem $m = \mathbf{D}(c, k_P)$.
    \end{tinfobox}

\begin{frame}{Algoritmus RSA}
    \pause
    Zkratka \textbf{RSA} pochází ze jmen autorů Ron \textbf{R}ivest, Adi \textbf{S}hamir a~Leonard \textbf{A}dleman.
    \pause
    \begin{enumerate}
        \item Vygeneruj \markdarkred{tři velká prvočísla}: největší označíme $s$ a~zbylá $p,q$.\pause
        \item Spočtáme $n=p\cdot q$.\pause
        \item Spočítáme číslo $v$ tak, aby
        \[v\cdot s\bmod (p-1)(q-1)=1.\]\pause
        \item Položíme veřejný a soukromý klíč uspořádaným dvojicím
        \[k_V=(v,n)\quad\text{a}\quad k_P=(s,n).\]\pause

        \[\markdarkred{y=}\;\mathbf{E}(x, k_V)=x^{v}\bmod n\quad\text{a}\quad\mathbf{D}(y, k_P)=y^{s}\bmod n.\]
    \end{enumerate}
\end{frame}

\subsection{Malá Fermatova věta}

\begin{frame}{Malá Fermatova věta}
    \pause
    \begin{thmblock}[Malá Fermatova věta]
        Nechť \(p\) je prvočíslo a nechť \(a\in\mathbb{Z}\) není dělitelné číslem \(p\). Potom platí
        \[
            a^{p-1} \bmod p = 1.
        \]
    \end{thmblock}

    \vspace{0.25cm}
    \pause
    \begin{tnoticeframe}
        Ekvivalentně lze psát, že pro každé \(a\in\mathbb{Z}\) platí
        \[
            a^{p} \bmod p = a.
        \]
    \end{tnoticeframe}

\end{frame}

\subsection{Čínská věta o zbytcích}

\begin{frame}{Čínská věta o zbytcích}
    \pause
    \begin{thmblock}[Čínská věta o zbytcích]
        Nechť \(m_1,m_2,\ldots,m_k\in\N\) jsou po dvou nesoudělná.

        Potom pro libovolná celá čísla \(a_1,a_2,\ldots,a_k\) má soustava kongruencí
        \begin{align*}
            x\bmod m_1 &= a_1,\\
            x\bmod m_2 &= a_2,\\
            &\vdots\\
            x\bmod m_k &= a_k
        \end{align*}
        právě jedno řešení.
    \end{thmblock}
    \pause
    \begin{tnoticeframe}
        Toto řešení je určeno jednoznačně modulo $M=m_1m_2\cdots m_k$.
    \end{tnoticeframe}

\end{frame}

\begin{frame}{Důsledek čínské věty o zbytcích}
    \pause
    \begin{corblock}
        Nechť $m_1,m_2\in\N$ jsou nesoudělná. Pak pro každé $x,a\in\N$ platí
        \[\left.\begin{aligned}
            x\bmod m_1&=a\\
            x\bmod m_2&=a
        \end{aligned}\right\}\iff x\bmod m_1m_2=a.\]
    \end{corblock}
\end{frame}

\subsection{Korektnost RSA}

\begin{frame}{Korektnost RSA}
    \pause
    Chceme ukázat, že $\mathbf{D}(\mathbf{E}(x, k_V), k_S)=x$, neboli že
    \[\left(x^{v}\bmod n\right)^{s}\bmod n=x^{vs}\bmod n=x.\]
    \pause
    Důkaz rozdělíme na dva případy:
    \begin{enumerate}
        \pause
        \item \textbf{$p$ dělí číslo $x$.} Pak platí
        \[x^{vs}\bmod p = 0 = x\bmod p.\]\pause
        \item \textbf{$p$ nedělí číslo $x$.} Protože platí $vs\bmod(p-1)(q-1)=1$, existuje $t\in\N$ takové, že
        \[vs=1+t(p-1)(q-1).\]\pause
        Pak dostáváme
        \begin{align*}
            x^{vs}\bmod p&=x^{1+t(p-1)(q-1)}\bmod p\stackrel{\text{MFV}}{=}x(x^{p-1})^{t(q-1)}\bmod p\\
            &=x\cdot 1^{t(q-1)}\bmod p=x\bmod p.
        \end{align*}
        \pause
        Podobně pro $q$ dostaneme, že též platí $x^{vs}\bmod q=x\bmod q$.
    \end{enumerate}

\end{frame}

\begin{frame}{Korektnost RSA}
    \pause
    Tedy víme, že platí
    \begin{align*}
        x\bmod p&=x^{vs}\bmod p\\
        x\bmod q&=x^{vs}\bmod q.
    \end{align*}
    \pause
    Podle \markdarkred{důsledku čínské věty o~zbytcích} platí
    \[x^{vs}\bmod pq=x\bmod pq=x\bmod n\stackrel{x<n}{=}x.\]
\end{frame}

\begin{frame}{Příklad}{Algoritmus RSA pro malá prvočísla}

    \begin{itemize}
        \pause
        \item Volme prvočísla
        \[
        s=27,\quad p=5,\quad q=11.
        \]
        \pause
        \item Pak $n=pq=55$. Spočítáme hodnotu $v$ tak, aby platilo
        \[vs\bmod (p-1)(q-1)=27v\bmod 40=1.\]
        Tuto rovnost splňuje např. $v=3$.
        \pause
        \item Máme tedy veřejný klíč $k_V=(3, 55)$ a soukromý klíč $k_S=(27,55)$.
        \pause
        \item Alice zašifruje zprávu \(x=7\):
        \[
        c=x^v \bmod n = 7^3 \bmod 55 = 343 \bmod 55 = 13.
        \]
        Bob dešifruje:
        \[
        x=c^s \bmod n = 13^{27}\bmod 55 = 7.
        \]
    \end{itemize}

\end{frame}

\subsection{Bezpečnost RSA}

\begin{frame}{Bezpečnost RSA}
    \pause
    U RSA je veřejný klíč $ k_V=(v,n)$, kde
    \[
        n=pq
    \]
    je součin dvou \markdarkred{velkých prvočísel}.
    \pause
    \vspace{0.2cm}

    \begin{talertframe}
        Pokud útočník zná rozklad
        \[
            n=pq,
        \]
        může z rovnosti
        \[vs\bmod(p-1)(q-1)=1\]
        snadno dopočítat $s$.
    \end{talertframe}
\end{frame}

\begin{frame}{Faktorizace}
    \pause
    \begin{talertbox}{Základní předpoklad bezpečnosti}
        Je snadné spočítat součin \(n=pq\), ale je těžké z čísla \(n\)
        zpětně zjistit prvočísla \(p\) a \(q\).
    \end{talertbox}
    \pause
    \decproblem{Faktorizace ($\FACTOR$)}{
        Přirozená čísla $n,b\in\N,b\leqslant n$.
    }{
        Existuje $d\in\N$ takové, že $1<d<b$ a $d$ dělí $n$?
    }
    \pause
    Protože pracujeme s binárními hodnotami $n$, platí
    \[|\binary{n}|=\lceil\log_2{n}\rceil.\]
    \pause
    Naivní algoritmus by musel mít složitost $2^{|\binary{n}|}$.
\end{frame}

\begin{frame}{Problém $\FACTOR$ patří do $\classNP$}
    \pause
    \begin{propblock}
        Platí $\FACTOR\in\classNP$.
    \end{propblock}
    \pause
    \begin{itemize}
        \item Na vstupu mějme dvojici $(n,b)$ a číslo $d\in\N$.\pause
        \item Celková délka vstupu je $L_n=|\binary{n}|+|\binary{b}|$.\pause
        \item Mějme algoritmus níže.\pause
        \begin{center}
            \begin{minipage}{0.7\textwidth}
                \begin{algorithm}[H]
                    \caption{Verifikátor}
                    \If{\textup{$(d=1\lor d\geqslant b)\land n\bmod d\neq 0$}}{
                        \Return{\textup{\textbf{false}}}
                    }
                    \Return{\textup{\textbf{true}}}
                \end{algorithm}
            \end{minipage}
        \end{center}
        \pause
        \item Porovnání binárně zapsaných čísel i~výpočet zbytku po dělení lze provést v~čase polynomiálním vzhledem k~počtu bitů vstupu $L_n$.\pause
        \item Ověřovač tedy běží v~polynomiálním čase.
    \end{itemize}

\end{frame}

\begin{frame}{Co víme o faktorizaci?}
    \pause
    \begin{tnoticeframe}[.5\textwidth]
        \begin{center}
            Víme, že $\FACTOR\in\classNP$.
        \end{center}
    \end{tnoticeframe}
    \pause
    \begin{talertbox}[.8\textwidth]{Otevřený problém}
        \begin{center}
            Není známo, zdali $\FACTOR$ je $\classNP$-úplný problém.
        \end{center}
    \end{talertbox}
    \pause
    Kdyby $\FACTOR\in\classP$, pak ze znalosti čísla $n$ bychom mohli nalézt prvočísla $p,q$ a~odtud odvodit $s$.
\end{frame}

\section{Pseudonáhodné generátory a jednosměrné funkce}

\titleframe{\huge Pseudonáhodné generátory\\ a~jednosměrné funkce}

\subsection{Pseudonáhodné generátory}

\begin{frame}{Pseudonáhodný generátor}
    \pause
    Funkce
    \[
        G:\{0,1\}^n \to \{0,1\}^{\ell(n)}
    \]
    je \markdarkred{pseudonáhodný generátor (PRG)}, pokud:\pause
    \begin{itemize}
        \item \(G\) je vyčíslitelná deterministickým polynomiálním algoritmem,\pause
        \item \(\ell(n)>n\) pro každé \(n\in\mathbb{N}\) (\markdarkred{stretch}),\pause
        \item žádný \markdarkred{pravděpodobnostní polynomiální algoritmus $\mathcal{A}$} neumí "rozlišit"
        \(G(y_n)\) od \(y_{\ell(n)}\), kromě zanedbatelné chyby.\pause
    \end{itemize}
    Vstup \(s\in\{0,1\}^n\), tzv. \markdarkred{seed}, se "natáhne" na delší řetězec \(G(s)\).

\end{frame}

\subsection{Nerozlišitelnost výstupu PRG}

\begin{frame}{Nerozlišitelnost výstupu PRG}
    \pause
    \begin{tinfobox}{Třetí podmínka}
        Pro každý pravděpodobnostní polynomiální algoritmus \(\mathcal{A}\) platí:
        \[
        \left|
        \Pr_{y\in\{0,1\}^{\ell(n)}}[\mathcal{A}(y)=1]
        -
        \Pr_{y\in\{0,1\}^{n}}[\mathcal{A}(G(y))=1]
        \right|
        \leqslant \varepsilon(n),
        \]
        kde \(\varepsilon(n)\) je zanedbatelná funkce. Funkce \(\varepsilon(n)\) je \textbf{zanedbatelná}, pokud pro každé
        \(k\in\mathbb{N}\) existuje \(n_k\) takové, že pro každé \(n>n_k\) platí
        \[
            \varepsilon(n) < \frac{1}{n^k}.
        \]
    \end{tinfobox}
    \pause
    \begin{tnoticeframe}
        Algoritmus \(\mathcal{A}\) dostane řetězec délky \(\ell(n)\) a snaží se
        rozhodnout, zda je skutečně náhodný, nebo vznikl jako výstup \(G\).
    \end{tnoticeframe}

\end{frame}

\subsection{Existence PRG, když P=NP}

\begin{frame}{Žádné PRG, pokud $\classP=\classNP$}
    \pause
    \begin{timportantframe}
        Pokud by platilo $\classP=\classNP$, pak by žádné pseudonáhodné generátory neexistovaly.
    \end{timportantframe}

    \vspace{0.2cm}
    \pause
    \textbf{Důvod:} Pro daný výstup \(y\in\set{0,1}^{n}\) bychom uměli efektivně rozhodnout, zda existuje seed \(s\) takový, že
    \[
        G(s)=y.
    \]
    \pause
    \begin{itemize}
        \item Uvažujme jazyk
        \[
            L_G =\set{y : \exists s\in\{0,1\}^n: G(s)=y}.
        \]
        Tento jazyk patří do \(\classNP\), protože certifikátem (nápovědou) je seed~\(s\).\pause
        \item Rozlišovač \(\mapping{\mathcal{D}}{\set{0,1}^{\ell(n)}}{\set{0,1}}\) se vstupem \(y\) udělá:
        \[
            \mathcal{D}(y)=1
            \quad\Longleftrightarrow\quad
            y\in L_G.
        \]
    \end{itemize}

\end{frame}

\begin{frame}{Žádné PRG, pokud $\classP=\classNP$}
    \pause
    \begin{itemize}
        \item Pokud \(y=G(s)\), pak vždy platí $\mathcal{D}(y)=1$ a tedy
        \[\Pr[\mathcal{D}(y)=1]=1.\]\pause
        \item Pokud je \(y\) skutečně náhodný řetězec délky \(\ell(n)\), pak pravděpodobnost, že leží v~\(L_G\), je nejvýše
        \[
            \frac{2^n}{2^{\ell(n)}}=2^{n-\ell(n)}.
        \]\pause
        \item Pak dostáváme
        \begin{align*}
             &\left|\Pr_{y\in\{0,1\}^{\ell(n)}}[\mathcal{D}(y)=1]-\Pr_{y\in\{0,1\}^{n}}[\mathcal{D}(G(y))=1]\right|\\
             &=\left|2^{n-\ell(n)}-1\right|=1-2^{n-\ell(n)}\geqslant 1-\frac{1}{2}=\frac{1}{2},
        \end{align*}
        což není zanedbatelná funkce.
    \end{itemize}
\end{frame}

\subsection{Jednosměrné funkce}

\begin{frame}{Jednosměrná funkce}
    \pause
    \[
        \mapping{f}{\set{0,1}^{*}}{\set{0,1}^{*}}
    \]
    je \markdarkred{jednosměrná funkce (OWF)}, pokud\pause
    \begin{itemize}
        \item $f(x)$ lze spočítat v polynomiálním čase pro každé $x\in\set{0,1}^{*}$ a\pause
        \item je \markdarkred{"těžko" invertovatelná}.
    \end{itemize}
    \pause
    Druhá podmínka formálně říká, že pro každý pravděpodobnostní polynomiální algoritmus $\mathcal{A}$ existuje zanedbatelná funkce $\varepsilon(n)$ taková, že pro každé $n\in\N$ 
    \[\Pr_{y\in\set{0,1}^{n}}\left[f(\mathcal{A}(y))=y\right]\leqslant\varepsilon(n).\]
    \pause
    \begin{timportantframe}
        Když náhodně vybereme $y$ a dáme jej útočníkovi, pak útočník má \textbf{zanedbatelně malou šanci}, že se mu podaří najít vstup~$x$ takový, že $y=f(x)$.
    \end{timportantframe}

\end{frame}

\begin{frame}{Možní kandidáti na jednosměrné funkce}
    \pause
    \begin{talertframe}[.8\textwidth]
        Dodnes není známé, zdali nějaké jednosměrné funkce existují.
    \end{talertframe}
    \pause
    Existují kandidáti na jednosměrné funkce:
    \pause
    \begin{itemize}
        \item Násobení prvočísel: $f(p,q)=pq$.\pause
        \item Součet podmnožiny:
        \[f(x_1,\dots,x_n,J)=\left(x_1,\dots,x_n,\left(\sum_{j\in J}x_j\right)\bmod 2^{n}\right),\]
        kde $x_1,\dots,x_n\in\set{0,1}^{n}$ a $J\subseteq\set{1,\dots,n}$.
    \end{itemize}

\end{frame}

\begin{frame}{Souvislost OWF a PRG}
    \pause
    \begin{itemize}
        \item Existence jednosměrných funkcí je~úzce spojena s~existencí pseudonáhodných generátorů.\pause
        \item Ve skutečnosti platí:
        \[
            \text{existují OWF}
            \quad\Longleftrightarrow\quad
            \text{existují PRG}.
        \]\pause
        \item Jednosměrná funkce poskytuje výpočetní těžkost. Z této těžkosti lze konstruovat bity, které vypadají náhodně.
    \end{itemize}

\end{frame}

\subsection{Souvislost PRG s kryptografií}

\begin{frame}{Souvislost s kryptografií}
    \pause
    \begin{itemize}
        \item V praxi nechceme používat opravdu náhodný klíč délky celé zprávy.\pause
        \item Místo toho použijeme krátký tajný seed \(s\) a z něj vytvoříme dlouhý proud bitů $G(s)$.\pause
        \item Potom šifrujeme podobně jako u jednorázové šifry:
        \[
            c=m\oplus G(s).
        \]
    \end{itemize}
    \pause
    \vspace{0.2cm}

    \begin{talertframe}
        Bezpečnost pak nestojí na dokonalé náhodnosti, ale na tom, že
        \(G(s)\) je \textbf{nerozlišitelný od náhodného řetězce}.
    \end{talertframe}

\end{frame}

\subsection{Definice a vlastnosti hešovací funkce}

\begin{frame}{Hešovací funkce}
    \pause
    \begin{defblock}
        \markdarkred{Hešovací funkce} je jakákoliv funkce z~$\set{0,1}^{*}$ do~$\set{0,1}^{n}$. Je-li $H$ hešovací funkce a $x\in\set{0,1}^{*}$, pak $H(x)$ se nazývá \markdarkred{hash} nebo též \markdarkred{otisk}.
    \end{defblock}
    \pause
    Požadavky na hešovací funkci:
    \pause
    \begin{enumerate}
        \item otisk je \textbf{rychle vyčíslitelný} a\pause
        \item malá změna vstupu má \textbf{výrazně} změnit výstup.
    \end{enumerate}
\end{frame}

\begin{frame}{Kolize jsou nevyhnutelné}
    Funkce
    \[\mapping{H}{\set{0,1}^{*}}{\set{0,1}^{n}}\]
    uvažuje vstupy z nekonečné množiny do množiny s~$2^{n}$ prvky.

            Musí existovat dvojice $x,y\in\set{0,1}^{*}$ taková, že $H(x)=H(y)$.

\subsection{Bezpečnostní vlastnosti hešovacích funkcí}

\begin{frame}{Bezpečnostní vlastnosti hešovacích funkcí}

    Uvažujme hešovací funkci
    \[
        \mapping{H_n}{\set{0,1}^{*}}{\set{0,1}^{n}}
    \]
    Dále nechť $\varepsilon$ značí nějakou zanedbatelnou funkci a $\mathcal{A}$ pravděpodobnostní polynomiální algoritmus.
    \begin{center}
        \begin{tabular}{@{}p{0.25\textwidth}@{\hspace{0.5em}}p{0.7\textwidth}@{}}

            \markdarkred{Odolnost vůči nalezení vzoru} &
            Z hashe \(y\) je těžké najít nějaké \(x\), pro které
            \(H_n(x)=y\), tzn. 
            \[
                \Pr_{y\in\set{0,1}^{n}}\left[
                H_n(\mathcal A(y))=y
                \right]\leqslant \varepsilon(n).
            \]
            pro každé $n\in\N$.\\

            \markdarkred{Odolnost vůči kolizím} &
            Je těžké najít libovolnou dvojici \(x\neq y\), pro kterou $H_n(x)=H_n(y)$, tj.
            \[
                \Pr_{x,y\in\set{0,1}^{n}}\left[
                x\neq y \land H_n(x)=H_n(y)
                \right]\leqslant \varepsilon(n).
            \]
            pro každé $n\in\N$.

        \end{tabular}
    \end{center}

\end{frame}
