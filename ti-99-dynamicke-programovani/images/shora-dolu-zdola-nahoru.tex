% Táž soustava podproblémů, dvojí směr výpočtu.
% Vlevo rekurze (shora dolů), vpravo dynamické programování (zdola nahoru).
\begin{tikzpicture}[x=1cm,y=1cm,font=\small]

    % ------------------------------------------------ vlevo: shora dolů
    \foreach \k in {0,...,3}{
        \foreach \i in {0,...,\k}{
            \node[vertex, minimum size=5mm, fill=white]
                (L\k\i) at ({-2.1+(\i-\k/2)*0.85}, {-\k*0.90}) {};
        }
    }
    \foreach \k in {0,1,2}{
        \foreach \i in {0,...,\k}{
            \pgfmathtruncatemacro{\kk}{\k+1}
            \pgfmathtruncatemacro{\ii}{\i+1}
            \draw[diredge, darkred] (L\k\i) -- (L\kk\i);
            \draw[diredge, darkred] (L\k\i) -- (L\kk\ii);
        }
    }
    \node[font=\small\bfseries, darkred] at (-2.1,0.85) {shora dolů};
    \node[font=\scriptsize] at (-2.1,0.45) {rekurze};
    \node[font=\scriptsize, anchor=north, align=center, text width=3.6cm]
        at (-2.1,-3.05) {celý problém rozkládáme\\ na menší podproblémy};
    \draw[diredge, darkred, line width=1.2pt] (-4.15,0.25) -- (-4.15,-2.85);

    % ---------------------------------------------- vpravo: zdola nahoru
    \foreach \k [evaluate=\k as \op using int(10+12*\k)] in {0,...,3}{
        \foreach \i in {0,...,\k}{
            \node[vertex, minimum size=5mm, fill=darkgreen!\op]
                (P\k\i) at ({2.1+(\i-\k/2)*0.85}, {-\k*0.90}) {};
        }
    }
    \foreach \k in {0,1,2}{
        \foreach \i in {0,...,\k}{
            \pgfmathtruncatemacro{\kk}{\k+1}
            \pgfmathtruncatemacro{\ii}{\i+1}
            \draw[diredge, darkgreen] (P\kk\i) -- (P\k\i);
            \draw[diredge, darkgreen] (P\kk\ii) -- (P\k\i);
        }
    }
    \node[font=\small\bfseries, darkgreen] at (2.1,0.85) {zdola nahoru};
    \node[font=\scriptsize] at (2.1,0.45) {dynamické programování};
    \node[font=\scriptsize, anchor=north, align=center, text width=3.6cm]
        at (2.1,-3.05) {od nejmenších podproblémů\\ skládáme celý problém};
    \draw[diredge, darkgreen, line width=1.2pt] (4.15,-2.85) -- (4.15,0.25);

\end{tikzpicture}
