% cena podle ceníku, pod tyčí zvolené dělení a celková utržená částka.
\begin{tikzpicture}[x=1cm,y=1cm,font=\small,
                    dil/.style={thick, fill=maincolor!15, rounded corners=1pt},
                    dilopt/.style={thick, fill=darkgreen!30, rounded corners=1pt}]

    % ---- horní řada: 4 | 1+3 | 2+2 | 3+1
    \foreach \x/\len/\cena in {0.00/4/9,

        \draw[dil] (\x,0) rectangle ({\x+\len*0.42},0.34);
        \node[font=\scriptsize, anchor=south] at ({\x+\len*0.21},0.32) {\cena};
    }
    \foreach \x/\len/\cena in {4.90/2/5, 5.74/2/5}{
        \draw[dilopt] (\x,0) rectangle ({\x+\len*0.42},0.34);
        \node[font=\scriptsize, anchor=south] at ({\x+\len*0.21},0.32) {\cena};
    }

    \node[font=\scriptsize, anchor=north west] at (0.00,-0.10) {$4$: $9$};
    \node[font=\scriptsize, anchor=north west] at (2.45,-0.10) {$1{+}3$: $9$};

    \node[font=\scriptsize, anchor=north west] at (0.00,-1.55) {$1{+}1{+}2$: $7$};
