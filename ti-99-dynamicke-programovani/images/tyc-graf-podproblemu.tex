% Graf podproblémů úlohy o dělení tyče pro $n=4$: vrcholy stromu rekurze
% se stejným popiskem jsou slepené do jediného.
\begin{tikzpicture}[x=1cm,y=1cm,font=\small,
                    cutnode/.style={vertex, minimum size=6.5mm, fill=white}]

    \foreach \k/\x in {4/0.0,3/1.7,2/3.4,1/5.1,0/6.8}{
        \node[cutnode] (p\k) at (\x,0) {$\k$};
    }

    \foreach \a/\b in {4/3,3/2,2/1,1/0}{
        \draw[diredge] (p\a) -- (p\b);
    }
    \foreach \a/\b in {4/2,3/1,2/0}{
        \draw[diredge] (p\a) to[bend left=38] (p\b);
    }
    \foreach \a/\b in {4/1,3/0}{
        \draw[diredge] (p\a) to[bend right=38] (p\b);
    }
    \draw[diredge] (p4) to[bend left=58] (p0);

    \node[font=\scriptsize, anchor=north, align=center] at (3.4,-1.55)
        {$5$ vrcholů a $10$ hran místo $16$ vrcholů stromu rekurze};

\end{tikzpicture}
