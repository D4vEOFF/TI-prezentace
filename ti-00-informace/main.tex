    \subsection{Tématické výstupy}
    
    \begin{frame}{Tématické výstupy}
        \begin{itemize}
            \onslide<2->{\item Během školního roku se budou odevzdávat tzv. \textbf{tématické výstupy}}
            \onslide<3->{\item V průběhu školního roku se bude postupně zpracovávat několik samostatných výstupů, z nichž každý bude věnován jednomu většímu tematickému celku.}
            \onslide<4->{\item Tyto výstupy budou obvykle obsahovat
            \begin{itemize}
                \item vypracování zadaných úloh,
                \item stručná vysvětlení pojmů,
                \item práci s příklady,
                \item případně i samostatné nastudování vybraného dílčího tématu.
            \end{itemize}}
            \onslide<5->{\item Úlohy, které budou součástí výstupů, budou zadávány \textbf{postupně}.}
            \onslide<6->{\item Tématické výstupy se budou odevzdávat v elektronické podobě v přísluěném modulu v Moodle kurzu.}
        \end{itemize}
    \end{frame}

    \section{Sylabus}
    
    \begin{frame}{Sylabus}
        \begin{enumerate}
            \onslide<2->{\item \textbf{Grafové algoritmy}
            \begin{itemize}
                \item Opakování základních pojmů z~teorie grafů
                \item Algoritmy BFS, DFS
                \item Binární halda
                \item Dijkstrův algoritmus a~algoritmus A$^{*}$
            \end{itemize}}
            \onslide<3->{\item \textbf{Algoritmicky těžké problémy}
            \begin{itemize}
                \item Opakování výrokové logiky
                \item Problémy $\SAT$, $\UNSAT$
                \item Algoritmus DPLL
                \item Problém nezávislé množiny a~kliky v~grafu
                \item Třídy složitosti $\classP$, $\classNP$ a~$\classNP$-úplné problémy
                \item Cookova-Levinova věta
            \end{itemize}}
            \onslide<4->{\item \textbf{Konečné automaty}
            \begin{itemize}
                \item Formální jazyky
                \item Deterministický a~nedeterministický konečný automat
                \item Regulární výrazy
                \item Formální gramatiky
            \end{itemize}}
        \end{enumerate}
    \end{frame}

    \begin{frame}{Sylabus}
        \begin{enumerate}

            \end{itemize}}
