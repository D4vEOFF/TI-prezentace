    \begin{frame}{Tématické výstupy}
        \begin{itemize}
            \onslide<2->{\item Během školního roku se budou odevzdávat tzv. \textbf{tématické výstupy}}
            \onslide<3->{\item V průběhu školního roku se bude postupně zpracovávat několik samostatných výstupů, z nichž každý bude věnován jednomu většímu tematickému celku.}
            \onslide<4->{\item Tyto výstupy budou obvykle obsahovat
            \begin{itemize}
                \item vypracování zadaných úloh,
                \item stručná vysvětlení pojmů,
                \item práci s příklady,
                \item případně i samostatné nastudování vybraného dílčího tématu.
            \end{itemize}}
            \onslide<5->{\item Úlohy, které budou součástí výstupů, budou zadávány \textbf{postupně}.}
            \onslide<6->{\item Tématické výstupy se budou odevzdávat v elektronické podobě v přísluěném modulu v Moodle kurzu.}
            \onslide<7->{\item Konkrétní požadavky a způsob hodnocení tématických výstupů jsou uvedeny v Moodle kurzu.}
        \end{itemize}
    \end{frame}
