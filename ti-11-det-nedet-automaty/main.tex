    \subsection{Definice jazyka}

    \begin{frame}{Formální jazyky}
        \visible<2->{\begin{defblock}
            \visible<2->{Mějme konečnou množinu znaků $\Sigma\neq\emptyset$, kterou budeme nazývat \markdarkred{abeceda}. Pak}
            \begin{itemize}
                \visible<3->{\item \markdarkred{slovo (řetězec)} je libovolná posloupnost znaků z~abecedy $\Sigma$.}
                \visible<4->{\item \markdarkred{prázdné slovo} je slovo neobsahující žádné znaky. Značíme jej $\lambda$.}
                \visible<5->{\item \markdarkred{množinu všech slov v~abecedě $\Sigma$} značíme $\Sigma^{*}$.}
                \visible<6->{\item \markdarkred{množina všech neprázdných slov v~abecedě $\Sigma$} se značí $\Sigma^{+}=\Sigma^{*}\setminus\set{\lambda}$.}
                \visible<7->{\item \markdarkred{jazyk nad abecedou $\Sigma$} je libovolná podmnožina slov $L\subseteq\Sigma^{*}$.}
                \visible<8->{\item \markdarkred{délka slova $w$} se značí $|w|$ (počet znaků ve slově $w$).}
                \visible<9->{\item \markdarkred{počet výskytů znaku $z$ ve slově $w$} značíme $|w|_z$.}
            \end{itemize}
        \end{defblock}}
    \end{frame}

    \subsection{Operace nad slovy}

    \begin{frame}{Operace nad slovy}
        \visible<2->{\begin{defblock}[Operace nad slovy]
            \visible<2->{Nechť $u=u_1\dots u_n,v=v_1\dots v_m$ jsou slova nad abecedou $\Sigma$. Pak definujeme}
            \begin{itemize}
                \visible<3->{\item \markdarkred{konkatenaci (zřetězení)} slov $u$ a~$v$ jako slovo
                \[uv=u_1\dots u_nv_1\dots v_m.\]}
                \visible<4->{\item \markdarkred{mocninu slova $u$} následovně:
                \begin{itemize}
                    \item $u^{0}=\lambda$,
                    \item $u^{k+1}=u^{k}u$,
                \end{itemize}
                kde $k\in\N_0$. Tzn. $u^{k}=\overbrace{\underbrace{u_1\dots u_n}_u\underbrace{u_1\dots u_n}_u\dots\underbrace{u_1\dots u_n}_u}^{\text{$k$-krát}}$.}
            \end{itemize}
        \end{defblock}}
    \end{frame}

    \begin{frame}{Formální jazyky}{Příklady}
        \small
        \visible<2->{Mějme abecedu znaků $\Sigma=\set{\texttt{a},\texttt{b}}$.}
        \begin{itemize}
            \visible<3->{\item Slovo $w=\texttt{abba}$ má délku $|w|=4$.}
            \visible<4->{\item $w^{5}=\texttt{abba} \texttt{abba} \texttt{abba} \texttt{abba} \texttt{abba}$.}
            \visible<5->{\item Množina všech slov $\Sigma^{*}=\set{\lambda,\texttt{a},\texttt{b},\texttt{aa},\texttt{ab},\texttt{ba},\texttt{bb},\texttt{aaa},\texttt{aab},\texttt{aba},\texttt{abb},\texttt{baa},\dots}$.}
            \visible<6->{\item Příklady \textbf{triviálních} jazyků nad abecedou: $\emptyset$, $\set{\lambda}$, $\Sigma^{*}$, $\set{\texttt{aa},\texttt{bb}}$.}
        \end{itemize}
    \end{frame}

    \begin{frame}{Formální jazyky}{Příklady jazyků}
        \small
        \begin{itemize}
            \visible<2->{\item Další příklady jazyků nad $\Sigma=\set{\texttt{a},\texttt{b}}$:}
            \begin{align*}
                \visible<3->{L_1&=\set{\texttt{a}^{2n}: n\in\N_0}=\set{\lambda,\texttt{aa},\texttt{aaaa},\texttt{aaaaaa},\dots}}\\
                \visible<4->{L_2&=\set{w\in\Sigma^{*}: \text{$w$ končí na $\texttt{ab}$}}=\set{w\in\Sigma^{*}: \exists u\in\Sigma^{*}: w=u\texttt{ab}}\\
                &=\set{\texttt{\markdarkred{ab}},\texttt{a\markdarkred{ab}},\texttt{b\markdarkred{ab}},\texttt{aa\markdarkred{ab}},\texttt{ab\markdarkred{ab}},\texttt{ba\markdarkred{ab}},\dots}}\\
                \visible<5->{L_3&=\set{w\in\Sigma^{*}: \text{$w$ neobsahuje jako podslovo $\texttt{bb}$}}\\
                &=\set{w\in\Sigma^{*}: \forall u,v\in\Sigma^{*}: w\neq u\texttt{bb}v}\\
                &=\set{\lambda,\texttt{a},\texttt{b},\texttt{aa},\texttt{ab},\texttt{ba},\xcancel{\texttt{\markdarkred{bb}}},\texttt{aaa},\texttt{aab},\texttt{aba},\xcancel{\texttt{a\markdarkred{bb}}},\dots}}
            \end{align*}
        \end{itemize}
    \end{frame}

    \subsection{Rozhodovací problém vs. jazyk}

    \begin{frame}{Poznámka k~rozhodovacím problémům}
        \small
        \visible<2->{U~rozhodovacích problémů jsme kódovali vstupy pomocí $0$ a~$1$.}
        \begin{itemize}
            \visible<3->{\item Mějme \textbf{binární} abecedu $\Sigma=\set{0,1}$.}
            \visible<4->{\item Rozhodovací problémy lze chápat jako jazyky nad $\set{0,1}^{*}$.}
            \visible<5->{\item Označme si $\binary{x}$ binární zakódování libovolného objektu $x$.}
            \visible<6->{\item Pak např. $\SAT$ lze chápat jako množinu všech \textbf{binárních řetězců} reprezentující takové logické formule $\varphi$, které jsou v~CNF a~jsou splnitelné. Tzn.
            \[\SAT=\set{\binary{\varphi}: \text{$\varphi$ je v CNF a je splnitelná}}.\]}
            \visible<7->{\item Podobně:}
            \begin{align*}
                \visible<8->{\THREESAT&=\set{\binary{\varphi}: \text{$\varphi$ je v~3-CNF a~je splnitelná}},\\}
                \visible<9->{\NZMNA&=\set{\binary{(G,k)}: \text{V~grafu $G$ existuje nez. mn. velikosti aspoň $k$}},\\}
                \visible<10->{\KLIKA&=\set{\binary{(G,k)}: \text{V~grafu $G$ existuje klika velikosti aspoň $k$}}.}
            \end{align*}
        \end{itemize}
    \end{frame}

    \subsection{Úvod k~NFA}

    \begin{frame}{Deterministický konečný automat}
        \visible<2->{\begin{defblock}[Deterministický konečný automat]
            \markdarkred{Deterministický konečný automat (DFA)} je uspořádaná pětice $A=(Q,\Sigma,\delta,q_0,F)$, kde
            \begin{itemize}
                \visible<3->{\item $Q$ je \textbf{množina stavů},}
                \visible<4->{\item $\Sigma$ je \textbf{abeceda}, z~níž sestávají slova na vstupu,}
                \visible<5->{\item $\mapping{\delta}{Q\times\Sigma}{Q}$ je tzv. \textbf{přechodová funkce} značící přechody mezi stavy,}
                \visible<6->{\item $q_0\in Q$ je \textbf{počáteční stav} a}
                \visible<7->{\item $F\subseteq Q$ je \textbf{množina přijímajících stavů}.}
            \end{itemize}
        \end{defblock}}
        \visible<8->{Automaty zakreslujeme jako orientovaný graf, kde}
        \begin{itemize}
            \visible<9->{\item \textbf{vrcholy} představují jednotlivé stavy a}
            \visible<10->{\item \textbf{hrany} přechody mezi nimi.}
        \end{itemize}
    \end{frame}

    \begin{frame}{Deterministický konečný automat}{Příklad}
        \small
        \visible<2->{\obrazek[0.25\textheight]{images/det-automat-1.tex}}
        \begin{itemize}
            \visible<3->{\item \textbf{Počáteční stav $q_0$} značíme šipkou vedoucí "odnikud".}
            \visible<4->{\item \textbf{Přijímající stav(y)} zvýrazňujeme dvojitým okrajem.}
        \end{itemize}
        \visible<5->{\begin{minipage}{0.45\textwidth}
            \begin{align*}
                \delta(q_0,0)&=q_1\\
                \delta(q_1,1)&=q_2\\
                \delta(q_2,0)&=q_2\\
                \delta(q_2,1)&=q_2
            \end{align*}
        \end{minipage}}
        \qquad
        \visible<6->{\begin{minipage}{0.45\textwidth}
            \begin{table}
                \centering
                \begin{tabular}{c|cc}
                            & $0$   & $1$   \\\hline
                    $q_0$   & $q_1$ &       \\
                    $q_1$   &       & $q_2$ \\
                    $q_2$   & $q_2$ & $q_2$
                \end{tabular}
            \end{table}
        \end{minipage}}
    \end{frame}

    \subsection{Výpočet DFA}

    \begin{frame}{Výpočet deterministického automatu}
        \visible<2->{Automat $A=(Q,\Sigma,\delta,q_0,F)$ dostane na vstupu slovo $w\in\Sigma^{*}$.}
        \begin{enumerate}
            \visible<3->{\item Na začátku se automat nachází v~počátečním stavu $q_0$.}
            \visible<4->{\item Vstupní slovo $w$ čte $A$ \textbf{zleva doprava} po jednotlivých znacích.}
            \visible<5->{\item Nachází-li se obecně automat ve stavu $q\in Q$, pak podle přečteného znaku $a$ určí podle přechodové funkce stav $s$, do kterého se má přesunout:
            \[\delta(q,a)=s.\]}
            \visible<6->{\item Pokud pro přečtený znak neexistuje žádný odpovídající stav, kam se má výpočet automatu přesunout, pak automat slovo $w$ \markdarkred{odmítá}.}
            \visible<7->{\item Pokud automat po přečtení celého slova $w$ skončil v~přijímajícím stavu, pak slovo $w$ \markdarkred{přijímá}. V~opačném případě jej odmítá.}
        \end{enumerate}
    \end{frame}

    \begin{frame}{Co všechno automat přijme?}
        \visible<2->{Pro obecný automat $A$ dává smysl se ptát, jak vypadají \textbf{všechna slova}, která automat přijme.}
        \visible<3->{\begin{defblock}[Jazyk automatu]
            Nechť je dán automat $A$ nad abecedou $\Sigma$. Pak \markdarkred{jazykem automatu $A$} rozumíme množinu slov
            \[L(A)=\set{w\in\Sigma^{*}: \text{$A$ přijme slovo $w$}}.\]
        \end{defblock}}
        \visible<4->{\begin{tnoticeframe}
            \begin{center}
                Jak vypadá $L(A)$ automatu z~minulého příkladu?
            \end{center}
        \end{tnoticeframe}}
    \end{frame}

    \begin{frame}{Jazyk automatu $A$}
        \visible<2->{\obrazek{images/det-automat-1-bez-pozn.tex}}
        \begin{itemize}
            \visible<3->{\item Abychom se dostali do přijímajícího stavu $q_2$, musíme nejprve na začátku přečíst znaky $0$ a~$1$.}
            \visible<4->{\item Když se výpočet dostane do $q_2$, pak už na zbytku slova nezáleží (smyčka pro oba znaky $0,1$).}
            \visible<5->{\item Tzn. vstupní slovo $w$ musí obsahovat na začátku znaky $01$.}
        \end{itemize}
        \visible<6->{Celkově tedy jazyk automatu $A$ je množina
        \begin{align*}
            L(A)&=\set{w\in\set{0,1}^{*}: \text{$w$ začíná na $01$}}\\
            &=\set{w\in\set{0,1}^{*}: \exists u\in\set{0,1}^{*}: w=01u}.
        \end{align*}}
    \end{frame}

    \subsection{Definice NFA}

    \begin{frame}{Nedeterministické konečné automaty}
        \begin{itemize}
            \visible<2->{\item Ne vždy je výhodné pro nás konstruovat automat pro nějaký jazyk přímo.}
            \visible<3->{\item Pokud bychom chtěli sestrojit automat např. pro rozpoznání slov $w$ obsahující podslovo $u$, pak jednodušší přístup by mohl být následující.}
            \visible<4->{\begin{timportantframe}
                \begin{center}
                    V~každém kroku výpočtu budeme "hádat", zdali jsme už nalezli \textbf{první znak} podslova $u$.
                \end{center}
            \end{timportantframe}}
            \visible<5->{\item Co to znamená? V~jednom stavu budeme mít \textbf{více přechodů} pro jeden znak.}
            \visible<6->{\begin{talertframe}
                \begin{center}
                    Pokud budeme mít více vyhovujících přechodů pro jeden znak, automat spustí \textbf{více paralelních výpočtů najednou}.
                \end{center}
            \end{talertframe}}
        \end{itemize}
    \end{frame}

    \subsection{Hledání podslova pomocí NFA}

    \begin{frame}{Hledáme podslovo $u$}
        \begin{itemize}
            \visible<2->{\item Navrhnout přechody pro takový automat nemusí být úplně jednoduché.}
            \visible<3->{\item Zkusíme proto následující strategii:}
            \begin{itemize}
                \visible<4->{\item Označme si první znak (pod)slova $u$ jako $a\in\Sigma$.}
                \visible<5->{\item Pokud uvidíme znak $a$, spustíme dva paralelní výpočty: první bude odpovídat situaci, kdy jsme \textbf{nenašli podslovo $u$} a~druhý \textbf{situaci, kdy ano}.}
                \visible<6->{\item Pokud jeden z~těchto výpočtů přijme slovo $w$ na vstupu, pak výpočet \textbf{prohlásíme za přijímající}.}
            \end{itemize}
        \end{itemize}
    \end{frame}

    \subsection{Jak NFA funguje?}
    
    \begin{frame}{Jak nedeterministický automat funguje?}
        \begin{itemize}
            \visible<2->{\item Pro jeden znak $a\in\Sigma$ můžeme mít ze stavu $q$ \textbf{více přechodů}.}
            \visible<3->{\item V~případě, že existuje pro $a$ více přechodů, automat spustí \textbf{více paralelních výpočtů najednou} (každý pro jeden přechod).}
            \visible<4->{\item Nedeterministický konečný automat budeme značit \markdarkred{NFA} (\textit{nondeterministic finite automaton}).}
        \end{itemize}
        \visible<5->{\begin{tnoticeframe}
            \begin{center}
                Pokud \textbf{alespoň jeden} výpočet přijal vstupní slovo $w$, pak automat $w$ \markdarkred{přijímá}.
            \end{center}
        \end{tnoticeframe}}
        \visible<6->{\begin{talertframe}
            \begin{center}
                Pokud všechny výpočty slovo $w$ odmítnou, pak automat $w$ také \markdarkred{odmítá}.
            \end{center}
        \end{talertframe}}
    \end{frame}

    \subsection{Definice NFA}

    \begin{frame}{Formální definice NFA}
        \visible<2->{\begin{defblock}[Nedeterministický konečný automat]
            \markdarkred{Nedeterministický konečný automat (NFA)} je pětice $A = (Q,\Sigma,\delta,q_0,F)$, kde
            \begin{itemize}
                \item $Q$ je konečná množina stavů,
                \item $\Sigma$ je vstupní abeceda,
                \item $\mapping{\delta}{Q\times\Sigma}{\powset(Q)}$ je přechodová funkce,
                \item $q_0 \in Q$ je počáteční stav a
                \item $F \subseteq Q$ je množina přijímajících stavů.
            \end{itemize}
        \end{defblock}}
        \visible<3->{$\powset(X)$ = potenční množina množiny $X$. Obsahuje \textbf{všechny podmnožiny} množiny $X$.}
        \visible<4->{\[\powset(X)=\set{A: A\subseteq X}\]}
    \end{frame}

    \begin{frame}{Čísla s~konečným desetinným rozvojem}
        \visible<2->{Mějme automat $D$ níže nad abecedou $\Sigma=\set{0,1,\dots,9,.}$:}
        \visible<3->{\obrazek{images/des-rozvoj.tex}}
        \visible<4->{\begin{align*}
            L(D)&=\set{d_1\dots d_k: k\geqslant 1,d_i\in\set{0,\dots,9}}\\
            &\cup\set{d_1\dots d_k.d_{k+1}\dots d_n:k\geqslant 1,n>k,d_i\in\set{0,\dots,9}}.
        \end{align*}}
    \end{frame}

    \subsection{Vztah DFA a~NFA}

    \begin{frame}{Vztah DFA a~NFA}
        \begin{itemize}
            \item NFA $N$ \textbf{lze vždy převést} na DFA $A$ takový, že
            \[L(N)=L(A),\]
            tzn. $A$ a~$N$ přijímají stejné jazyky.
            \item Triviálně každý DFA je zároveň i~NFA.
        \end{itemize}
        \begin{tnoticeframe}
            \begin{center}
                Algoritmus pro převod se nazývá \textbf{podmnožinová konstrukce}, ale tu pro jednoduchost vynecháme.
            \end{center}
        \end{tnoticeframe}
    \end{frame}

    \subsection{"Rozpoznávací" schopnost DFA}

    \begin{frame}{Co všechno DFA "rozpozná"?}
        \visible<2->{\begin{tnoticeframe}
            \begin{center}
                Lze pro každý jazyk sestrojit deterministický automat?
            \end{center}
        \end{tnoticeframe}}
        \visible<3->{\begin{talertframe}
            \textbf{Bohužel ne.} Dokonce existují takové jazyky, které nelze zpracovat žádným výpočetním modelem.
        \end{talertframe}}
    \end{frame}

    \begin{frame}{DFA neumí počítat}
        \visible<2->{Uvažujme jazyk
        \[L_\text{eq}=\set{w\in\set{0,1}^{*}: |w|_0=|w|_1}.\]}
        \visible<3->{\begin{talertframe}
            \begin{center}
                Pro jazyk $L_\text{eq}$ nelze sestrojit žádný DFA, který by jej přijímal.
            \end{center}
        \end{talertframe}}
    \end{frame}

    \subsection{Důkaz neexistence DFA}

    \begin{frame}{Pro jazyk $L_\text{eq}$ neexistuje DFA}
        \begin{itemize}
            \visible<2->{\item Zkusme pro spor předpokládat, že pro jazyk $L_\text{eq}$ existuje nějaký DFA $A$, tzn.
            \[L(A)=L_\text{eq}.\]}
            \visible<3->{\begin{talertframe}
                \begin{center}
                    Tento předpoklad povede ke sporu.
                \end{center}
            \end{talertframe}}
            \visible<4->{\item Označme si \textbf{počet stavů} automatu $A$ jako $m=|Q|$.}
            \visible<5->{\item Pokusíme se najít takové slovo $w$, které automat $A$ přijme, \textbf{i~když by neměl}.}
        \end{itemize}
    \end{frame}

    \begin{frame}{Pro jazyk $L_\text{eq}$ neexistuje DFA}
        \begin{itemize}
            \visible<2->{\item Uvažujme $m+1$ slov tvaru
            \[\lambda,0,00,\dots,0^{m}.\]}
            \visible<3->{\item Protože těchto slov je $m+1$ a~stavů automatu $A$ je jen $m$, musí nutně z~těchto slov existovat slova
            \[0^{i}\quad\text{a}\quad 0^{j},\]
            kde $0\leqslant i<j\leqslant m$, po jejichž přečtení skončí automat ve stejném stavu $q\in Q$.}
            \visible<4->{\item Uvažujme slova
            \[u=0^{i}1^{i}\quad\text{a}\quad v=0^{j}1^{i}.\]}
        \end{itemize}
    \end{frame}

    \begin{frame}{Pro jazyk $L_\text{eq}$ neexistuje DFA}
        \centering
        \visible<2->{\input{images/pro-jazyk-l-eq-neexistuje-dfa.tex}}
        \begin{itemize}
            \visible<3->{\item Automat $A$ skončí po přečtení prvních $i$ znaků slova $u$ ve stavu $q$.}
            \visible<4->{\item Podobně po přečtení prvních $j$ znaků slova $v$ skončí ve stejném stavu $q$.}
            \visible<5->{\item To znamená, že po přečtení podslova $1^{i}$ by měl pro $u$ a~$v$ na konci výpočtu být $A$ ve stejném stavu $r\in Q$.}
            \visible<6->{\item Protože $u\in L(A)$, tak $A$ slovo $u$ musí přijmout a~tedy přijme i~slovo $v$.}
        \end{itemize}
        \visible<7->{\begin{talertframe}
            \begin{center}
                Jenže $v\notin L(A)$, což znamená, že $A$ přijal slovo, \textbf{které není v~jazyce $L_\text{eq}$!}
            \end{center}
        \end{talertframe}}
    \end{frame}
