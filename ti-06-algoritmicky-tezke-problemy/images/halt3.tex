% Spor při dosazení D(D) jako vývojový diagram
% (makra \flow... z assets/macros.tex).
\begin{tikzpicture}[font=\small]

% definice D
\flowstep[text width=5.9cm]{diagdef}{(0,0)}
{
\small program $D(P)$\\[0.4mm]
\scriptsize pokud $\HALT(P,P)=\text{ANO}$, pak cyklí\\
\scriptsize pokud $\HALT(P,P)=\text{NE}$, pak skončí
}

% dosazení
\flowstep[text width=1.6cm]{self}{(0,-1.75)}{$D(D)$}

\flowarrowlabel{diagdef}{right}{\scriptsize dosadíme $P=D$}{self}

% rozhodnutí
\flowdecision{haltdd}{(0,-3.15)}{\scriptsize $\HALT(D,D)$?}

\flowarrow{self}{haltdd}

% levá větev
\flowstep[text width=3.2cm]{yes}{(-3.45,-4.55)}
{
\scriptsize $\HALT(D,D)=\text{ANO}$\\
\scriptsize tedy $D(D)$ se zastaví
}

\flowio{yescontr}{(-3.45,-5.8)}
{
\scriptsize ale podle definice $D$\\
\scriptsize pak $D(D)$ cyklí
}

% pravá větev
\flowstep[text width=3.2cm]{no}{(3.45,-4.55)}
{
\scriptsize $\HALT(D,D)=\text{NE}$\\
\scriptsize tedy $D(D)$ se nezastaví
}

\flowio{nocontr}{(3.45,-5.8)}
{
\scriptsize ale podle definice $D$\\
\scriptsize pak $D(D)$ skončí
}

% šipky
\flowbranch[flowhv]{haltdd.west}{pos=0.25, above}{\scriptsize ANO}{yes.north}
\flowbranch[flowhv]{haltdd.east}{pos=0.25, above}{\scriptsize NE}{no.north}

\flowarrow{yes}{yescontr}
\flowarrow{no}{nocontr}

% závěr
\node[align=center] at (0,-6.95)
{
V obou případech vznikne spor $\implies$ Algoritmus pro \HALT{} neexistuje.
};

\end{tikzpicture}
