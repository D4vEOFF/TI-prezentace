}

%---------------------------------
% makra
%---------------------------------
\newcommand{\diskrect}[5]{%
  \draw[draw=black, fill=#4, rounded corners=1pt]
    ({#1-#3/2},{#2}) rectangle ({#1+#3/2},{#2+0.30});
  \node at (#1,{#2+0.15}) {\scriptsize #5};
}

% trojúhelníkové znázornění věže 1,...,n-1
% #1 = x středu, #2 = y spodní hrany
\newcommand{\triangletower}[2]{%
  \draw[draw=black, fill=blue!20]
    ({#1-0.52},{#2}) --
    ({#1+0.52},{#2}) --
    ({#1},{#2+1.42}) -- cycle;
  \node at (#1,{#2+0.43}) {\scriptsize $n-1$};
}

% sada tří kolíků
\newcommand{\pegsystem}{%
  \draw[base] (-1.35,0) -- (1.35,0);
  \draw[peg] (-0.75,0) -- (-0.75,1.90);
  \draw[peg] ( 0.00,0) -- ( 0.00,1.90);
  \draw[peg] ( 0.75,0) -- ( 0.75,1.90);
  \node[below] at (-0.75,-0.08) {$A$};
  \node[below] at ( 0.00,-0.08) {$B$};
  \node[below] at ( 0.75,-0.08) {$C$};
}

%---------------------------------
% Krok 2
%---------------------------------
\begin{scope}[shift={(0,0)}]
  \node[align=center] at (0,2.55) {přesuň prvních $n-1$ disků\\ z $A$ na $C$};
  \pegsystem
  \diskrect{-0.75}{0.00}{1.15}{orange!40}{$n$}
  \triangletower{-0.75}{0.30}
\end{scope}

% šipka 2 -> 3
\draw[->,thick] (1.65,0.95) -- (2.45,0.95);

  \pegsystem
  \diskrect{-0.75}{0.00}{1.15}{orange!40}{$n$}
  \triangletower{0.75}{0.00}
\end{scope}
