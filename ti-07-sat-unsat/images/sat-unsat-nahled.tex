% Náhled k~sekci o~SAT a~UNSAT: existuje ohodnocení, které formuli splní?
\begin{tikzpicture}[x=1cm, y=1cm]

    \node[draw, thick, rounded corners=3pt, inner sep=5pt, fill=maincolor!8]
        (f) at (0,0) {$\varphi=(x\lor \neg y)\land(\neg x\lor y)\land(x\lor y)$};

    \node[darkgreen] (ano) at (-2.6,-2.2) {$x=1,\ y=1$};
    \node[darkred]   (ne)  at ( 2.6,-2.2) {$x=0,\ y=1$};

    \draw[->, >=stealth, thick, darkgreen] (f.south) -- (ano.north);
    \draw[->, >=stealth, thick, darkred]   (f.south) -- (ne.north);

    \node[darkgreen] at (-2.6,-3.0) {$\varphi=1$ \cmark};
    \node[darkred]   at ( 2.6,-3.0) {$\varphi=0$ \xmark};

    \node[font=\small] at (0,-4.0)
        {Stačí \textbf{jedno} splňující ohodnocení, a~formule je splnitelná.};

\end{tikzpicture}
