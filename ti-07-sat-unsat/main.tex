    \subsection{Logické operace}

    \begin{frame}{Logické operace}
        \visible<2->{Mějme logické proměnné $x,y\in\set{0,1}$.}
        \begin{itemize}
            \visible<3->{\item $x \land y$ = \textbf{konjunkce (AND)}}
            \visible<4->{\item $x\lor y$ = \textbf{disjunkce (OR)}}
            \visible<5->{\item $x\implies y$ = \textbf{implikace}}
            \visible<6->{\item $x\iff y$ = \textbf{ekvivalence}}
        \end{itemize}
    \end{frame}

    \subsection{Tabulka logických operací}

    \begin{frame}{Tabulka logických operací}

    \subsection{Základní vztahy}

    \begin{frame}{Základní vztahy}
        \begin{enumerate}
            \visible<2->{\item \textbf{De Morganovy zákony}}
            \begin{itemize}
                \visible<3->{\item $\neg (x\land y) = \left[\neg x\lor \neg y\right]$
                \item $\neg (x_1\land x_2\land \dots\land x_n)=\neg x_1 \lor \neg x_2 \lor \dots \lor \neg x_n$}
                \visible<4->{\item $\neg (x\lor y)=\neg x\land \neg y$
                \item $\neg (x_1\lor x_2\lor \dots\lor x_n)=\neg x_1 \land \neg x_2 \land \dots \land \neg x_n$}
            \end{itemize}
            \visible<5->{\item \textbf{Vztahy mezi implikací, konjunkcí a disjunkcí}}
            \begin{itemize}
                \visible<6->{\item $x\implies y=\neg x\lor y$}
                \visible<7->{\item $x\iff y=(x\implies y)\land (y\implies x)$, nebo-li
                \[x\iff y=(\neg x\lor y)\land (\neg y\lor x)\]}
            \end{itemize}
        \end{enumerate}
        \visible<8->{\begin{talertframe}[0.9\textwidth]
            \begin{center}
                Implikaci a ekvivalenci můžeme nahradit pomocí $\land$ a $\lor$.
            \end{center}
        \end{talertframe}}
    \end{frame}

    \begin{frame}{Logická formule}
        \begin{itemize}
            \visible<2->{\item \markdarkred{Logická formule} je pro nás libovolný logický výraz obsahující \textbf{volné logické proměnné}.}
            \visible<3->{\item Např.
            \[\varphi(x,y)=(x\land y)\lor \neg y.\]}
        \end{itemize}
    \end{frame}
